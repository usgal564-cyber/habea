   348→   348→
   349→   349→function PaymentGate({
   350→   350→  quiz,
   351→   351→  onPaid,
   352→   352→  onBack,
   353→   353→}: {
   354→   354→  quiz: QuizListItem;
   355→   355→  onPaid: () => void;
   356→   356→  onBack: () => void;
   357→   357→}) {
   358→   358→  const { token } = useAuthStore();
   359→   359→  const [paying, setPaying] = useState(false);
   360→   360→
   361→   361→  const handlePay = async () => {
   362→   362→    setPaying(true);
   363→   363→    try {
   364→   364→      const res = await fetch("/api/quiz/payment", {
   365→   365→        method: "POST",
   366→   366→        headers: {
   367→   367→          "Content-Type": "application/json",
   368→   368→          Authorization: `Bearer ${token}`,
   369→   369→        },
   370→   370→        body: JSON.stringify({ quizId: quiz.id }),
   371→   371→      });
   372→   372→      const data = await res.json();
   373→   373→      if (res.ok) {
   374→   374→        toast.success("Төлбөр амжилттай");
   375→   375→        onPaid();
   376→   376→      } else {
   377→   377→        toast.error("Төлбөр амжилтгүй", { description: data.error || "Дахин оролдоно уу" });
   378→   378→      }
   379→   379→    } catch {
   380→   380→      toast.error("Холболтын алдаа");
   381→   381→    } finally {
   382→   382→      setPaying(false);
   383→   383→    }
   384→   384→  };
   385→   385→
   386→   386→  return (
   387→   387→    <Dialog open onOpenChange={() => onBack()}>
   388→   388→      <DialogContent className="sm:max-w-md bg-white">
   389→   389→        <DialogHeader>
   390→   390→          <DialogTitle className="text-brand-900 font-bold text-xl flex items-center gap-2">
