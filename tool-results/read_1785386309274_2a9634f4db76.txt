   366→   366→      // First register for the course
   367→   367→      const res = await fetch(`/api/courses/${course.id}/register`, {
   368→   368→        method: "POST",
   369→   369→        headers: {
   370→   370→          Authorization: `Bearer ${token}`,
   371→   371→          "Content-Type": "application/json",
   372→   372→        },
   373→   373→      });
   374→   374→      const data = await res.json();
   375→   375→
   376→   376→      if (res.ok) {
   377→   377→        toast.success("Бүртгэл амжилттай!");

   383→   383→      toast.error("Холболтын алдаа");

   385→   385→      setLoading(false);
   386→   386→    }

   391→   391→    setLoading(true);

   394→   394→        method: "POST",
   395→   395→        headers: {
   396→   396→          Authorization: `Bearer ${token}`,
   397→   397→          "Content-Type": "application/json",
   398→   398→        },
   399→   399→        body: JSON.stringify({ courseId: course.id }),
   400→   400→      });
   401→   401→      const data = await res.json();
   402→   402→
   403→   403→      if (res.ok) {
   404→   404→        toast.success("Төлбөр амжилттай!");
   405→   405→        setStep("confirmed");
   406→   406→      } else {
   407→   407→        toast.error("Төлбөр амжилтгүй", { description: data.error });
   408→   408→      }
   409→   409→    } catch {
   410→   410→      toast.error("Холболтын алдаа");

   413→   413→    }
   414→   414→  };

   421→   421→  ];

   426→   426→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 backdrop-blur-sm p-4">
   427→   427→      <motion.div
   428→   428→        initial={{ opacity: 0, scale: 0.95 }}
   429→   429→        animate={{ opacity: 1, scale: 1 }}
