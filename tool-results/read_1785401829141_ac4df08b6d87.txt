    12→        quizSeeded bool

    16→type QuizPaymentRequest struct {
    17→        QuizID string `json:"quizId" binding:"required"`
    18→}

    20→func GetQuizzesHandler(c *gin.Context) {
    21→        quizMu.Lock()
    22→        defer quizMu.Unlock()
    23→
    24→        if !quizSeeded {
    25→                seedQuizzes()
    26→                quizSeeded = true
    27→        }
    28→
    29→        var quizzes []Quiz
    30→        if err := DB.Find(&quizzes).Error; err != nil {
    31→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to fetch quizzes"})
    32→                return
    33→        }
    34→
    35→        c.JSON(http.StatusOK, gin.H{"quizzes": quizzes})
    36→}
    37→
    38→func GetQuizQuestionsHandler(c *gin.Context) {
    39→        quizID := c.Query("quizId")
    40→        if quizID == "" {
    41→                c.JSON(http.StatusBadRequest, gin.H{"error": "quizId query parameter required"})
    42→                return
    43→        }
    44→
    45→        var questions []QuizQuestion
    46→        if err := DB.Where("quiz_id = ?", quizID).Order("question_order ASC").Find(&questions).Error; err != nil {
    47→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to fetch questions"})
    48→                return
    49→        }
    50→
    51→        var quiz Quiz
    52→        DB.Select("id, title").Where("id = ?", quizID).First(&quiz)
    53→
    54→        type SafeQuestion struct {
    55→                ID            string `json:"id"`
    56→                QuizID        string `json:"quizId"`
    57→                QuestionText  string `json:"questionText"`
    58→                OptionA       string `json:"optionA"`
    59→                OptionB       string `json:"optionB"`
    60→                OptionC       string `json:"optionC"`
    61→                OptionD       string `json:"optionD"`
    62→                QuestionOrder int    `json:"questionOrder"`
    63→        }
    64→
    65→        safeQuestions := make([]SafeQuestion, len(questions))
    66→        for i, q := range questions {
    67→                safeQuestions[i] = SafeQuestion{
    68→                        ID:            q.ID,
    69→                        QuizID:        q.QuizID,
    70→                        QuestionText:  q.QuestionText,
    71→                        OptionA:       q.OptionA,
    72→                        OptionB:       q.OptionB,
    73→                        OptionC:       q.OptionC,
    74→                        OptionD:       q.OptionD,
    75→                        QuestionOrder: q.QuestionOrder,
    76→                }
    77→        }
    78→
    79→        c.JSON(http.StatusOK, gin.H{
    80→                "quiz":      quiz,
    81→                "questions": safeQuestions,
    82→        })
    83→}

    85→func QuizPaymentHandler(c *gin.Context) {
    86→        var req QuizPaymentRequest
    87→        if err := c.ShouldBindJSON(&req); err != nil {
    88→                c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
    89→                return
    90→        }
    91→
    92→        userID, _ := c.Get("userId")
    93→
    94→        attempt := QuizAttempt{
    95→                ID:     uuid.New().String(),
    96→                QuizID: req.QuizID,
    97→                UserID: userID.(string),
    98→                Score:  0,
    99→                Total:  0,
   100→                Passed: false,
   101→        }
   102→
   103→        if err := DB.Create(&attempt).Error; err != nil {
   104→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to create quiz attempt"})
   105→                return
   106→        }
   107→
   108→        c.JSON(http.StatusOK, gin.H{"attemptId": attempt.ID})
   109→}

   112→        title       string
   113→        description string
   114→        slug        string

   118→func seedQuizzes() {
   119→        var count int64
   120→        DB.Model(&Quiz{}).Count(&count)
   121→        if count > 0 {
   122→                return
   123→        }
   124→
   125→        allQuizData := getAllQuizData()
   126→
   127→        for _, qd := range allQuizData {
   128→                quiz := Quiz{
   129→                        ID:            uuid.New().String(),
   130→                        Title:         qd.title,
   131→                        Description:   qd.description,
   132→                        Slug:          qd.slug,
   133→                        QuestionCount: len(qd.questions),
   134→                        Price:         5000,
   135→                }
   136→                DB.Create(&quiz)
   137→
   138→                for i, q := range qd.questions {
   139→                        correctIdx := 0
   140→                        switch q[5] {
   141→                        case "A":
   142→                                correctIdx = 0
   143→                        case "B":
   144→                                correctIdx = 1
   145→                        case "C":
   146→                                correctIdx = 2
   147→                        case "D":
   148→                                correctIdx = 3
   149→                        }
   150→                        qq := QuizQuestion{
   151→                                ID:            uuid.New().String(),
   152→                                QuizID:        quiz.ID,
   153→                                QuestionText:  q[0],
   154→                                OptionA:       q[1],
   155→                                OptionB:       q[2],
   156→                                OptionC:       q[3],
   157→                                OptionD:       q[4],
   158→                                CorrectIndex:  correctIdx,
   159→                                QuestionOrder: i + 1,
   160→                        }
   161→                        DB.Create(&qq)
   162→                }
   163→        }
   164→}
