   241→/* ─── Step 2: Quiz Taking ───────────────────────────────── */
   242→
   243→function QuizTaking({
   244→  quiz,
   245→  onFinish,
   246→  onBack,
   247→}: {
   248→  quiz: QuizListItem;
   249→  onFinish: (answers: number[]) => void;
   250→  onBack: () => void;
   251→}) {
   252→  const [currentPage, setCurrentPage] = useState(1);
   253→  const [totalPages, setTotalPages] = useState(1);
   254→  const [questions, setQuestions] = useState<Question[]>([]);
   255→  const [answers, setAnswers] = useState<Record<string, number>>({});
   256→  const [isLoading, setIsLoading] = useState(true);
   257→  const [isSubmitting, setIsSubmitting] = useState(false);
   258→
   259→  const fetchQuestions = useCallback(
   260→    async (page: number) => {
   261→      setIsLoading(true);
   262→      try {
   263→        const res = await fetch(
   264→          `/api/quiz?slug=${quiz.slug}&page=${page}&limit=${QUESTIONS_PER_PAGE}`
   265→        );
   266→        const data = await res.json();
   267→        if (data.quiz && data.questions) {
   268→          setQuestions(data.questions);
   269→          setTotalPages(data.pagination.totalPages);
   270→        } else {
   271→          toast.error("Алдаа", { description: "Асуултуудыг ачааллахад алдаа гарлаа." });
   272→        }
   273→      } catch {
   274→        toast.error("Холболтын алдаа");
   275→      } finally {
   276→        setIsLoading(false);
   277→      }
   278→    },
   279→    [quiz.slug]
   280→  );
   281→
   282→  // Fetch first page
   283→  useEffect(() => {
   284→    fetchQuestions(1);
   285→  }, [fetchQuestions]);
   286→
   287→  const handleSelectAnswer = (questionId: string, optionIndex: number) => {
   288→    setAnswers((prev) => ({ ...prev, [questionId]: optionIndex }));
   289→  };
   290→
   291→  const handlePageChange = (page: number) => {
   292→    setCurrentPage(page);
   293→    fetchQuestions(page);
   294→    window.scrollTo({ top: 0, behavior: "smooth" });
   295→  };
   296→
   297→  const handleFinish = async () => {
   298→    setIsSubmitting(true);
   299→    // Build answers array ordered by questionOrder across ALL pages
   300→    // We need to fetch all questions to build the full answer array
   301→    try {
   302→      const allAnswers: number[] = [];
   303→      // For each page, fetch if not already cached, then map answers
   304→      for (let p = 1; p <= totalPages; p++) {
   305→        const res = await fetch(
   306→          `/api/quiz?slug=${quiz.slug}&page=${p}&limit=${QUESTIONS_PER_PAGE}`
   307→        );
   308→        const data = await res.json();
   309→        if (data.questions) {
   310→          for (const q of data.questions) {
   311→            allAnswers.push(answers[q.id] ?? -1);
   312→          }
   313→        }
   314→      }
   315→      onFinish(allAnswers);
   316→    } catch {
   317→      toast.error("Алдаа", { description: "Хариултуудыг илгээхэд алдаа гарлаа." });
   318→    } finally {
   319→      setIsSubmitting(false);
   320→    }
   321→  };
   322→
   323→  const answeredOnPage = questions.filter((q) => answers[q.id] !== undefined).length;
   324→  const progressPercent = totalPages > 0 ? ((currentPage - 1) / totalPages) * 100 : 0;
   325→
   326→  return (
   327→    <motion.div
   328→      initial={{ opacity: 0, y: 20 }}
   329→      animate={{ opacity: 1, y: 0 }}
   330→      transition={{ duration: 0.4 }}
   331→    >
   332→      {/* Top Bar */}
   333→      <div className="mb-6 flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   334→        <div className="flex items-center gap-3">
   335→          <Button
   336→            variant="ghost"
   337→            size="icon"
   338→            className="text-brand-600 hover:text-brand-800 hover:bg-brand-50"
   339→            onClick={onBack}
   340→          >
   341→            <ArrowLeft className="size-5" />
   342→          </Button>
   343→          <div>
   344→            <h2 className="text-xl font-bold text-brand-900 sm:text-2xl">
   345→              {quiz.title}
   346→            </h2>
   347→            <p className="text-sm text-muted-foreground">{quiz.description}</p>
   348→          </div>
   349→        </div>
   350→        <Badge className="bg-brand-50 text-brand-700 hover:bg-brand-100 w-fit">
   351→          <ListChecks className="mr-1 size-3.5" />
   352→          Хуудас {currentPage} / {totalPages}
   353→        </Badge>
   354→      </div>
   355→
   356→      {/* Progress Bar */}
   357→      <div className="mb-6 space-y-2">
   358→        <div className="flex items-center justify-between text-sm">
   359→          <span className="text-muted-foreground">
   360→            Хуудасны дэвсгэр: {currentPage} / {totalPages}
   361→          </span>
   362→          <span className="font-medium text-brand-700">
   363→            Энэ хуудас: {answeredOnPage}/{questions.length} хариулсан
   364→          </span>
   365→        </div>
   366→        <Progress value={progressPercent + (answeredOnPage / (questions.length || 1) / totalPages) * 100} className="h-2.5 bg-brand-100" />
   367→      </div>
   368→
   369→      {/* Questions */}
   370→      {isLoading ? (
   371→        <div className="space-y-6">
   372→          {Array.from({ length: 5 }).map((_, i) => (
   373→            <Card key={i} className="p-6">
   374→              <Skeleton className="h-5 w-3/4 mb-4" />
   375→              <div className="space-y-3">
   376→                {Array.from({ length: 4 }).map((_, j) => (
   377→                  <Skeleton key={j} className="h-10 w-full" />
   378→                ))}
   379→              </div>
   380→            </Card>
   381→          ))}
   382→        </div>
   383→      ) : (
   384→        <AnimatePresence mode="wait">
   385→          <motion.div
   386→            key={currentPage}
   387→            variants={pageTransition}
   388→            initial="initial"
   389→            animate="animate"
   390→            exit="exit"
   391→            className="space-y-4"
   392→          >
   393→            {questions.map((question, qIdx) => {
   394→              const globalIndex = (currentPage - 1) * QUESTIONS_PER_PAGE + qIdx + 1;
   395→              const selectedAnswer = answers[question.id];
   396→
   397→              return (
   398→                <motion.div
   399→                  key={question.id}
   400→                  variants={itemVariants}
   401→                  initial="hidden"
   402→                  animate="visible"
   403→                >
   404→                  <Card className="overflow-hidden border-brand-100 bg-white">
   405→                    <CardContent className="p-5 sm:p-6">
   406→                      {/* Question Number & Text */}
   407→                      <div className="mb-4 flex items-start gap-3">
   408→                        <span className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-brand-600 text-xs font-bold text-white">
   409→                          {globalIndex}
   410→                        </span>
   411→                        <p className="text-base font-medium leading-relaxed text-foreground sm:text-lg">
   412→                          {question.questionText}
   413→                        </p>
   414→                      </div>
   415→
   416→                      {/* Options */}
   417→                      <div className="ml-10 space-y-2.5">
   418→                        {OPTION_KEYS.map((key, optIdx) => {
   419→                          const isSelected = selectedAnswer === optIdx;
   420→                          return (
   421→                            <button
   422→                              key={key}
   423→                              type="button"
   424→                              onClick={() =>
   425→                                handleSelectAnswer(question.id, optIdx)
   426→                              }
   427→                              className={`
   428→                                flex w-full items-start gap-3 rounded-lg border-2 px-4 py-3 text-left text-sm transition-all duration-200
   429→                                sm:text-base
   430→                                ${
   431→                                  isSelected
   432→                                    ? "border-brand-500 bg-brand-50 text-brand-900 shadow-sm ring-1 ring-brand-200"
   433→                                    : "border-transparent bg-muted/50 text-foreground hover:border-brand-200 hover:bg-brand-50/50"
   434→                                }
   435→                              `}
   436→                            >
   437→                              <span
   438→                                className={`
   439→                                  flex h-6 w-6 shrink-0 items-center justify-center rounded-full border-2 text-xs font-bold transition-colors
   440→                                  ${
   441→                                    isSelected
   442→                                      ? "border-brand-600 bg-brand-600 text-white"
   443→                                      : "border-muted-foreground/30 text-muted-foreground"
   444→                                  }
   445→                                `}
   446→                              >
   447→                                {OPTION_LABELS[optIdx]}
   448→                              </span>
   449→                              <span className="pt-0.5 leading-relaxed">
   450→                                {question[key]}
   451→                              </span>
   452→                            </button>
   453→                          );
   454→                        })}
   455→                      </div>
   456→                    </CardContent>
   457→                  </Card>
   458→                </motion.div>
   459→              );
   460→            })}
   461→          </motion.div>
   462→        </AnimatePresence>
   463→      )}
   464→
   465→      {/* Page Navigation */}
   466→      <div className="mt-8 flex flex-col items-center gap-4 sm:flex-row sm:justify-between">
   467→        {/* Previous Button */}
   468→        <Button
   469→          variant="outline"
   470→          className="border-brand-200 text-brand-700 hover:bg-brand-50 hover:text-brand-800"
   471→          onClick={() => handlePageChange(currentPage - 1)}
   472→          disabled={currentPage <= 1 || isLoading}
   473→        >
   474→          <ChevronLeft className="size-4" />
   475→          Өмнөх
   476→        </Button>
   477→
   478→        {/* Page Numbers */}
   479→        <div className="flex items-center gap-1.5">
   480→          {Array.from({ length: totalPages }, (_, i) => i + 1).map((page) => (
   481→            <button
   482→              key={page}
   483→              type="button"
   484→              onClick={() => handlePageChange(page)}
   485→              disabled={isLoading}
   486→              className={`
   487→                flex h-9 w-9 items-center justify-center rounded-md text-sm font-medium transition-all
   488→                ${
   489→                  page === currentPage
   490→                    ? "bg-brand-600 text-white shadow-sm"
   491→                    : "text-muted-foreground hover:bg-brand-50 hover:text-brand-700"
   492→                }
   493→                disabled:opacity-50
   494→              `}
   495→            >
   496→              {page}
   497→            </button>
   498→          ))}
   499→        </div>
   500→
   501→        {/* Next / Finish Button */}
   502→        {currentPage < totalPages ? (
   503→          <Button
   504→            className="bg-brand-600 text-white hover:bg-brand-700"
   505→            onClick={() => handlePageChange(currentPage + 1)}
   506→            disabled={isLoading}
   507→          >
   508→            Дараах
   509→            <ChevronRight className="size-4" />
   510→          </Button>
   511→        ) : (
   512→          <Button
   513→            className="bg-brand-600 text-white hover:bg-brand-700"
   514→            onClick={handleFinish}
   515→            disabled={isSubmitting || isLoading}
   516→          >
   517→            {isSubmitting ? (
   518→              <Loader2 className="size-4 animate-spin" />
   519→            ) : (
   520→              <CheckCircle2 className="size-4" />
   521→            )}
   522→            Дуусгах
   523→          </Button>
   524→        )}
   525→      </div>
   526→    </motion.div>
   527→  );
   528→}

   530→/* ─── Step 3: Results ───────────────────────────────────── */
   531→
   532→function QuizResults({
   533→  quiz,
   534→  result,
   535→  onRetry,
   536→  onBackToList,
   537→}: {
   538→  quiz: QuizListItem;
   539→  result: QuizResult;
   540→  onRetry: () => void;
   541→  onBackToList: () => void;
   542→}) {
   543→  const percentage = Math.round((result.score / result.total) * 100);
   544→  const isPassed = result.passed;
   545→
   546→  return (
   547→    <motion.div
   548→      initial={{ opacity: 0, scale: 0.95 }}
   549→      animate={{ opacity: 1, scale: 1 }}
   550→      transition={{ duration: 0.5, ease: "easeOut" }}
   551→      className="mx-auto max-w-lg"
   552→    >
   553→      <Card className="overflow-hidden border-2 bg-white">
   554→        {/* Top Accent */}
   555→        <div
   556→          className={`h-2 ${
   557→            isPassed
   558→              ? "bg-gradient-to-r from-brand-500 to-brand-700"
   559→              : "bg-gradient-to-r from-red-400 to-red-600"
   560→          }`}
   561→        />
   562→
   563→        <CardContent className="p-8 sm:p-10">
   564→          <div className="flex flex-col items-center text-center">
   565→            {/* Icon Animation */}
   566→            <motion.div
   567→              initial={{ scale: 0 }}
   568→                  animate={{ scale: 1 }}
   569→                  transition={{
   570→                    type: "spring",
   571→                    stiffness: 200,
   572→                    damping: 15,
   573→                    delay: 0.2,
   574→                  }}
   575→                  className={`
   576→                    mb-6 flex h-24 w-24 items-center justify-center rounded-full
   577→                    ${
   578→                      isPassed
   579→                        ? "bg-brand-100 text-brand-600"
   580→                        : "bg-red-100 text-red-500"
   581→                    }
   582→                  `}
   583→                >
   584→                  {isPassed ? (
   585→                    <Trophy className="size-12" />
   586→                  ) : (
   587→                    <XCircle className="size-12" />
   588→                  )}
   589→                </motion.div>
   590→

   729→    /* ─── Main Component ──────────────────────────────────── */
   730→    
   731→    type Step = "selection" | "taking" | "results";
   732→    
   733→    export default function QuizPage() {
   734→      const [step, setStep] = useState<Step>("selection");
   735→      const [selectedQuiz, setSelectedQuiz] = useState<QuizListItem | null>(null);
   736→      const [result, setResult] = useState<QuizResult | null>(null);
   737→      const [isSubmitting, setIsSubmitting] = useState(false);
   738→    
   739→      const handleSelectQuiz = useCallback((quiz: QuizListItem) => {
   740→        setSelectedQuiz(quiz);
   741→        setResult(null);
   742→        setStep("taking");
   743→      }, []);
   744→    
   745→      const handleFinish = useCallback(
   746→        async (answers: number[]) => {
   747→          if (!selectedQuiz) return;
   748→    
   749→          setIsSubmitting(true);
   750→          try {
   751→            const res = await fetch("/api/quiz", {
   752→              method: "POST",
   753→              headers: { "Content-Type": "application/json" },
   754→              body: JSON.stringify({
   755→                quizId: selectedQuiz.id,
   756→                answers,
   757→              }),
   758→            });
   759→    
   760→            const data = await res.json();
   761→    
   762→            if (res.ok) {
   763→              setResult({
   764→                attemptId: data.attemptId,
   765→                score: data.score,
   766→                total: data.total,
   767→                passed: data.passed,
   768→              });
   769→              setStep("results");
   770→    
   771→              if (data.passed) {
   772→                toast.success("Баяр хүргэе!", {
   773→                  description: `Та ${data.score}/${data.total} оноо авч тэнцлээ!`,
   774→                });
   775→              } else {
   776→                toast.error("Тэнцээгүй", {
   777→                  description: `Та ${data.score}/${data.total} оноо авсан байна. 70% шаардлагатай.`,
   778→                });
   779→              }
   780→            } else {
   781→              toast.error("Алдаа", {
   782→                description: data.error || "Хариултуудыг илгээхэд алдаа гарлаа.",
   783→              });
   784→            }
   785→          } catch {
   786→            toast.error("Холболтын алдаа");
   787→          } finally {
   788→            setIsSubmitting(false);
   789→          }
   790→        },
   791→        [selectedQuiz]
   792→      );
   793→    
   794→      const handleRetry = useCallback(() => {
   795→        setResult(null);
   796→        setStep("taking");
   797→      }, []);
   798→    
   799→      const handleBackToList = useCallback(() => {
   800→        setSelectedQuiz(null);
   801→        setResult(null);
   802→        setStep("selection");
   803→      }, []);
   804→    
   805→      return (
   806→        <section className="w-full">
   807→          {step === "selection" && (
   808→            <QuizSelection onSelect={handleSelectQuiz} />
   809→          )}
   810→    
   811→          {step === "taking" && selectedQuiz && (
   812→            <QuizTaking
   813→              quiz={selectedQuiz}
   814→              onFinish={handleFinish}
   815→              onBack={handleBackToList}
   816→            />
   817→          )}
   818→    
   819→          {step === "results" && selectedQuiz && result && (
   820→            <QuizResults
   821→              quiz={selectedQuiz}
   822→              result={result}
   823→              onRetry={handleRetry}
   824→              onBackToList={handleBackToList}
   825→            />
   826→          )}
   827→    
   828→          {/* Submit overlay */}
   829→          {isSubmitting && (
   830→            <motion.div
   831→              initial={{ opacity: 0 }}
   832→              animate={{ opacity: 1 }}
   833→              className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 backdrop-blur-sm"
   834→            >
   835→              <div className="flex flex-col items-center gap-4 rounded-2xl bg-white p-8 shadow-2xl">
   836→                <Loader2 className="size-10 animate-spin text-brand-600" />
   837→                <p className="text-lg font-medium text-brand-900">
   838→                  Хариултуудыг шалгаж байна...
   839→                </p>
   840→              </div>
   841→            </motion.div>
   842→          )}
   843→        </section>
   844→      );
   845→    }
