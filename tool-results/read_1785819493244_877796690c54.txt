   152→func generateCode() string {
   153→        const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"
   154→        code := make([]byte, 6)
   155→        for i := range code {
   156→                n, _ := rand.Int(rand.Reader, big.NewInt(int64(len(chars))))
   157→                code[i] = chars[n.Int64()]
   158→        }
   159→        return string(code)
   160→}

   177→func RegisterHandler(c *gin.Context) {
   178→        var req RegisterRequest
   179→        if err := c.ShouldBindJSON(&req); err != nil {
   180→                c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
   181→                return
   182→        }
   183→        var existing User
   184→        if DB.Where("email = ?", req.Email).First(&existing).Error == nil {
   185→                c.JSON(http.StatusConflict, gin.H{"error": "Бүртгэлтэй имэйл байна"})
   186→                return
   187→        }
   188→        hash, _ := bcrypt.GenerateFromPassword([]byte(req.Password), bcrypt.DefaultCost)
   189→        user := User{ID: uuid.New().String(), FirstName: req.FirstName, LastName: req.LastName, Email: req.Email, Phone: req.Phone, Password: string(hash), Address: req.Address, SecondaryPhone: req.SecondaryPhone, Role: "USER"}
   190→        if err := DB.Create(&user).Error; err != nil {
   191→                c.JSON(500, gin.H{"error": "Failed to create user"})
   192→                return
   193→        }
   194→        token, _ := GenerateToken(user.ID, user.Email, user.Role, user.FirstName+" "+user.LastName)
   195→        c.JSON(201, gin.H{"user": userResponse(user), "token": token})
   196→}

   320→func QuizSubmitHandler(c *gin.Context) {
   321→        var req QuizSubmitRequest
   322→        if err := c.ShouldBindJSON(&req); err != nil {
   323→                c.JSON(400, gin.H{"error": err.Error()})
   324→                return
   325→        }
   326→        userID, _ := c.Get("userId")
   327→        var questions []QuizQuestion
   328→        DB.Where("quiz_id = ?", req.QuizID).Order("index asc").Find(&questions)
   329→
   330→        score := 0
   331→        for _, ans := range req.Answers {
   332→                for _, q := range questions {
   333→                        if q.ID == ans.QuestionID && q.Correct == ans.SelectedOption {
   334→                                score++
   335→                        }
   336→                }
   337→        }
   338→
   339→        ansJSON, _ := json.Marshal(req.Answers)
   340→        attempt := QuizAttempt{ID: uuid.New().String(), UserID: userID.(string), QuizID: req.QuizID, Score: score, Total: len(questions), Answers: string(ansJSON)}
   341→        DB.Create(&attempt)
   342→
   343→        c.JSON(200, gin.H{"score": score, "total": len(questions), "passed": score >= int(float64(len(questions))*0.8), "attemptId": attempt.ID})
   344→}

   348→func ExamHandler(c *gin.Context) {
   349→        var req ExamActionRequest
   350→        if err := c.ShouldBindJSON(&req); err != nil {
   351→                c.JSON(400, gin.H{"error": err.Error()})
   352→                return
   353→        }
   354→
   355→        if req.Action == "verify" {
   356→                var exam Exam
   357→                if DB.Where("code = ? AND is_active = ?", req.Code, true).First(&exam).Error != nil {
   358→                        c.JSON(404, gin.H{"error": "Идэвхтэй шалгалт олдсонгүй эсвэл код буруу байна"})
   359→                        return
   360→                }
   361→                c.JSON(200, gin.H{"exam": gin.H{"id": exam.ID, "title": exam.Title, "questionCount": exam.QuestionCount, "duration": exam.Duration}})
   362→                return
   363→        }
   364→
   365→        if req.Action == "submit" {
   366→                userID, _ := c.Get("userId")
   367→                // Check if already took
   368→                var existing ExamAttempt
   369→                if DB.Where("exam_id = ? AND user_id = ?", req.ExamID, userID).First(&existing).Error == nil {
   370→                        c.JSON(403, gin.H{"error": "Та энэ шалгалтад өмнө нь орсон байна. Дахин өгөх боломжгүй."})
   371→                        return
   372→                }
   373→
   374→                var questions []ExamQuestion
   375→                DB.Where("exam_id = ?", req.ExamID).Order("index asc").Find(&questions)
   376→
   377→                score := 0
   378→                for i, ans := range req.Answers {
   379→                        if i < len(questions) && ans == questions[i].Correct {
   380→                                score++
   381→                        }
   382→                }
   383→                passed := len(questions) > 0 && score >= int(float64(len(questions))*0.8)
   384→
   385→                ansJSON, _ := json.Marshal(req.Answers)
   386→                attempt := ExamAttempt{ID: uuid.New().String(), UserID: userID.(string), ExamID: req.ExamID, Score: score, Total: len(questions), Passed: passed, Answers: string(ansJSON)}
   387→                DB.Create(&attempt)
   388→
   389→                c.JSON(200, gin.H{"score": score, "total": len(questions), "passed": passed})
   390→                return
   391→        }
   392→
   393→        c.JSON(400, gin.H{"error": "Invalid action"})
   394→}

   429→func ExportExamHandler(c *gin.Context) {
   430→        attemptID := c.Param("attemptId")
   431→        var attempt ExamAttempt
   432→        if DB.Where("id = ?", attemptID).First(&attempt).Error != nil {
   433→                c.JSON(404, gin.H{"error": "Not found"})
   434→                return
   435→        }
   436→        var exam Exam
   437→        DB.Where("id = ?", attempt.ExamID).First(&exam)
   438→        var questions []ExamQuestion
   439→        DB.Where("exam_id = ?", attempt.ExamID).Order("index asc").Find(&questions)
   440→
   441→        var answers []int
   442→        json.Unmarshal([]byte(attempt.Answers), &answers)
   443→
   444→        c.Header("Content-Type", "text/csv; charset=utf-8")
   445→        c.Header("Content-Disposition", fmt.Sprintf("attachment; filename=exam_%s_result.csv", exam.Code))
   446→        writer := csv.NewWriter(c.Writer)
   447→        writer.Write([]string{"Шалгалт", exam.Title, "", "Код", exam.Code, "", "Огноо", attempt.CreatedAt.Format("2006-01-02 15:04:05")})
   448→        writer.Write([]string{"#", "Асуулт", "A", "B", "C", "D", "Таны хариулт", "Зөв хариулт", "Дүн"})
   449→        for i, q := range questions {
   450→                ansStr := "-"
   451→                correctStr := string([]string{"A", "B", "C", "D"}[q.Correct])
   452→                if i < len(answers) && answers[i] >= 0 {
   453→                        ansStr = string([]string{"A", "B", "C", "D"}[answers[i]])
   454→                }
   455→                status := "Буруу"
   456→                if i < len(answers) && answers[i] == q.Correct {
   457→                        status = "Зөв"
   458→                }
   459→                writer.Write([]string{fmt.Sprintf("%d", i+1), q.Question, q.OptionA, q.OptionB, q.OptionC, q.OptionD, ansStr, correctStr, status})
   460→        }
   461→        writer.Write([]string{"", "", "", "", "", "", "", "", ""})
   462→        writer.Write([]string{"Нийт", "", "", "", "", "", fmt.Sprintf("%d/%d", attempt.Score, attempt.Total), "", fmt.Sprintf("%v", attempt.Passed)})
   463→        writer.Flush()
   464→}

   547→func GetSurveyResultsHandler(c *gin.Context) {
   548→        var responses []SurveyResponse
   549→        DB.Order("created_at desc").Find(&responses)
   550→
   551→        type Stat struct {
   552→                QuestionID   string      `json:"questionId"`
   553→                Ratings      []int       `json:"ratings"`
   554→                AvgRating    float64     `json:"avgRating"`
   555→                YesCount     int         `json:"yesCount"`
   556→                NoCount      int         `json:"noCount"`
   557→                TextAnswers  []string    `json:"textAnswers"`
   558→        }
   559→
   560→        // Aggregate
   561→        questionStats := make(map[string]*Stat)
   562→        totalResponses := len(responses)
   563→
   564→        for _, r := range responses {
   565→                var parsed map[string]interface{}
   566→                json.Unmarshal([]byte(r.Responses), &parsed)
   567→                for qID, val := range parsed {
   568→                        if questionStats[qID] == nil {
   569→                                questionStats[qID] = &Stat{QuestionID: qID}
   570→                        }
   571→                        stat := questionStats[qID]
   572→                        switch v := val.(type) {
   573→                        case float64:
   574→                                stat.Ratings = append(stat.Ratings, int(v))
   575→                        case string:
   576→                                if v == "yes" {
   577→                                        stat.YesCount++
   578→                                } else if v == "no" {
   579→                                        stat.NoCount++
   580→                                } else {
   581→                                        stat.TextAnswers = append(stat.TextAnswers, v)
   582→                                }
   583→                        }
   584→                }
   585→        }
   586→
   587→        var stats []*Stat
   588→        for _, s := range questionStats {
   589→                if len(s.Ratings) > 0 {
   590→                        sum := 0
   591→                        for _, r := range s.Ratings { sum += r }
   592→                        s.AvgRating = float64(sum) / float64(len(s.Ratings))
   593→                }
   594→                stats = append(stats, s)
   595→        }
   596→
   597→        c.JSON(200, gin.H{"totalResponses": totalResponses, "stats": stats, "responses": responses})
   598→}
