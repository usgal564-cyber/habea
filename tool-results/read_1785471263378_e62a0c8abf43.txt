   126→func RegisterHandler(c *gin.Context) {
   127→        var req RegisterRequest
   128→        if err := c.ShouldBindJSON(&req); err != nil {
   129→                c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
   130→                return
   131→        }
   132→
   133→        var existing User
   134→        if err := DB.Where("email = ?", req.Email).First(&existing).Error; err == nil {
   135→                c.JSON(http.StatusConflict, gin.H{"error": "Email already registered"})
   136→                return
   137→        }
   138→
   139→        hashedPassword, err := bcrypt.GenerateFromPassword([]byte(req.Password), bcrypt.DefaultCost)
   140→        if err != nil {
   141→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to hash password"})
   142→                return
   143→        }
   144→
   145→        user := User{
   146→                ID:             uuid.New().String(),
   147→                FirstName:      req.FirstName,
   148→                LastName:       req.LastName,
   149→                Email:          req.Email,
   150→                Phone:          req.Phone,
   151→                Password:       string(hashedPassword),
   152→                Address:        req.Address,
   153→                SecondaryPhone: req.SecondaryPhone,
   154→                Role:           "USER",
   155→        }
   156→
   157→        if err := DB.Create(&user).Error; err != nil {
   158→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to create user"})
   159→                return
   160→        }
   161→
   162→        token, err := GenerateToken(user.ID, user.Email, user.Role)
   163→        if err != nil {
   164→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to generate token"})
   165→                return
   166→        }
   167→
   168→        c.JSON(http.StatusCreated, gin.H{
   169→                "user": gin.H{
   170→                        "userId": user.ID,
   171→                        "email":  user.Email,
   172→                        "role":   user.Role,
   173→                },
   174→                "token": token,
   175→        })
   176→}

   178→func LoginHandler(c *gin.Context) {

   204→                        "userId": user.ID,
   205→                        "email":  user.Email,

   294→func QuizSubmitHandler(c *gin.Context) {
   295→        var req QuizSubmitRequest
   296→        if err := c.ShouldBindJSON(&req); err != nil {
   297→                c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
   298→                return
   299→        }
   300→
   301→        var questions []QuizQuestion
   302→        if err := DB.Where("quiz_id = ?", req.QuizID).Order("index ASC").Find(&questions).Error; err != nil {
   303→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to fetch questions"})
   304→                return
   305→        }
   306→
   307→        if len(questions) == 0 {
   308→                c.JSON(http.StatusNotFound, gin.H{"error": "Quiz has no questions"})
   309→                return
   310→        }
   311→
   312→        questionMap := make(map[string]QuizQuestion)
   313→        for _, q := range questions {
   314→                questionMap[q.ID] = q
   315→        }
   316→
   317→        correct := 0
   318→        total := len(questions)
   319→        for _, ans := range req.Answers {
   320→                if q, ok := questionMap[ans.QuestionID]; ok {
   321→                        if ans.SelectedOption == q.Correct {
   322→                                correct++
   323→                        }
   324→                }
   325→        }
   326→
   327→        answersJSON, _ := json.Marshal(req.Answers)
   328→        userID, _ := c.Get("userId")
   329→
   330→        attempt := QuizAttempt{
   331→                ID:      uuid.New().String(),
   332→                UserID:  userID.(string),
   333→                QuizID:  req.QuizID,
   334→                Score:   correct,
   335→                Total:   total,
   336→                Answers: string(answersJSON),
   337→        }
   338→        DB.Create(&attempt)
   339→
   340→        percentage := 0
   341→        if total > 0 {
   342→                percentage = (correct * 100) / total
   343→        }
   344→
   345→        c.JSON(http.StatusOK, gin.H{
   346→                "score":      correct,
   347→                "total":      total,
   348→                "correct":    correct,
   349→                "percentage": percentage,
   350→        })
   351→}

   357→func ExamHandler(c *gin.Context) {
   358→        var req ExamActionRequest
   359→        if err := c.ShouldBindJSON(&req); err != nil {
   360→                c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
   361→                return
   362→        }
   363→
   364→        switch req.Action {
   365→        case "verify":
   366→                verifyExamCode(c, req.Code)
   367→        case "submit":
   368→                submitExam(c, req)
   369→        default:
   370→                c.JSON(http.StatusBadRequest, gin.H{"error": "Invalid action"})
   371→        }
   372→}
   373→
   374→func verifyExamCode(c *gin.Context, code string) {
   375→        var exam Exam
   376→        if err := DB.Where("code = ? AND is_active = ?", code, true).First(&exam).Error; err != nil {
   377→                c.JSON(http.StatusNotFound, gin.H{"error": "Invalid or inactive exam code"})
   378→                return
   379→        }
   380→
   381→        c.JSON(http.StatusOK, gin.H{
   382→                "exam": gin.H{
   383→                        "id":            exam.ID,
   384→                        "title":         exam.Title,
   385→                        "duration":      exam.Duration,
   386→                        "questionCount": exam.QuestionCount,
   387→                },
   388→        })
   389→}
   390→
   391→func submitExam(c *gin.Context, req ExamActionRequest) {
   392→        userID, _ := c.Get("userId")
   393→
   394→        var exam Exam
   395→        if err := DB.Where("id = ?", req.ExamID).First(&exam).Error; err != nil {
   396→                c.JSON(http.StatusNotFound, gin.H{"error": "Exam not found"})
   397→                return
   398→        }
   399→
   400→        if !exam.IsActive {
   401→                c.JSON(http.StatusBadRequest, gin.H{"error": "Exam is not active"})
   402→                return
   403→        }
   404→
   405→        var questions []QuizQuestion
   406→        DB.Where("quiz_id = ?", req.ExamID).Order("index ASC").Find(&questions)
   407→
   408→        score := 0
   409→        total := len(questions)
   410→        for i, answer := range req.Answers {
   411→                if i < len(questions) && answer == questions[i].Correct {
   412→                        score++
   413→                }
   414→        }
   415→
   416→        passed := false
   417→        if total > 0 {
   418→                passed = float64(score)/float64(total) >= 0.7
   419→        }
   420→
   421→        answersJSON, _ := json.Marshal(req.Answers)
   422→
   423→        attempt := ExamAttempt{
   424→                ID:      uuid.New().String(),
   425→                ExamID:  req.ExamID,
   426→                UserID:  userID.(string),
   427→                Score:   score,
   428→                Total:   total,
   429→                Passed:  passed,
   430→                Answers: string(answersJSON),
   431→        }
   432→
   433→        if err := DB.Create(&attempt).Error; err != nil {
   434→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to save attempt"})
   435→                return
   436→        }
   437→
   438→        c.JSON(http.StatusOK, gin.H{
   439→                "score":  score,
   440→                "total":  total,
   441→                "passed": passed,
   442→        })
   443→}

  1591→// GetQuizQuestionsHandler returns questions for a specific quiz (for quiz taking)
  1592→func GetQuizQuestionsHandler(c *gin.Context) {
  1593→        quizID := c.Query("quizId")
  1594→        if quizID == "" {
  1595→                c.JSON(http.StatusBadRequest, gin.H{"error": "quizId query parameter required"})
  1596→                return
  1597→        }
  1598→
  1599→        var questions []QuizQuestion
  1600→        if err := DB.Where("quiz_id = ?", quizID).Order("index ASC").Find(&questions).Error; err != nil {
  1601→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to fetch questions"})
  1602→                return
  1603→        }
  1604→
  1605→        type SafeQuestion struct {

  1607→                QuizID   string   `json:"quizId"`

  1609→                Options  []string `json:"options"`
  1610→                Index    int      `json:"index"`
  1611→        }
  1612→
  1613→        safeQuestions := make([]SafeQuestion, len(questions))

  1616→                json.Unmarshal([]byte(q.Options), &opts)
  1617→
  1618→                safeQuestions[i] = SafeQuestion{
  1619→                        ID:       q.ID,
  1620→                        QuizID:   q.QuizID,
  1621→                        Question: q.Question,
  1622→                        Options:  opts,
  1623→                        Index:    q.Index,
  1624→                }
  1625→        }
  1626→
  1627→        var quiz Quiz
  1628→        DB.Select("id, title").Where("id = ?", quizID).First(&quiz)
  1629→
  1630→        c.JSON(http.StatusOK, gin.H{
  1631→                "quiz":     gin.H{"id": quiz.ID, "title": quiz.Title},
  1632→                "questions": safeQuestions,
  1633→        })
