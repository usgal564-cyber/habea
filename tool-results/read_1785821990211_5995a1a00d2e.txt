   153→func generateCode() string {
   154→        const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"
   155→        code := make([]byte, 6)
   156→        for i := range code {
   157→                n, _ := rand.Int(rand.Reader, big.NewInt(int64(len(chars))))
   158→                code[i] = chars[n.Int64()]
   159→        }
   160→        return string(code)
   161→}

   178→func RegisterHandler(c *gin.Context) {
   179→        var req RegisterRequest
   180→        if err := c.ShouldBindJSON(&req); err != nil {
   181→                c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
   182→                return
   183→        }
   184→        var existing User
   185→        if DB.Where("email = ?", req.Email).First(&existing).Error == nil {
   186→                c.JSON(http.StatusConflict, gin.H{"error": "Бүртгэлтэй имэйл байна"})
   187→                return
   188→        }
   189→        hash, _ := bcrypt.GenerateFromPassword([]byte(req.Password), bcrypt.DefaultCost)
   190→        user := User{ID: uuid.New().String(), FirstName: req.FirstName, LastName: req.LastName, Email: req.Email, Phone: req.Phone, Password: string(hash), Address: req.Address, SecondaryPhone: req.SecondaryPhone, Role: "USER"}
   191→        if err := DB.Create(&user).Error; err != nil {
   192→                c.JSON(500, gin.H{"error": "Failed to create user"})
   193→                return
   194→        }
   195→        token, _ := GenerateToken(user.ID, user.Email, user.Role, user.FirstName+" "+user.LastName)
   196→        c.JSON(201, gin.H{"user": userResponse(user), "token": token})
   197→}

   375→func QuizSubmitHandler(c *gin.Context) {
   376→        var req QuizSubmitRequest
   377→        if err := c.ShouldBindJSON(&req); err != nil {
   378→                c.JSON(400, gin.H{"error": err.Error()})
   379→                return
   380→        }
   381→        userID, _ := c.Get("userId")
   382→        var questions []QuizQuestion
   383→        DB.Where("quiz_id = ?", req.QuizID).Order("index asc").Find(&questions)
   384→
   385→        score := 0
   386→        for _, ans := range req.Answers {
   387→                for _, q := range questions {
   388→                        if q.ID == ans.QuestionID && q.Correct == ans.SelectedOption {
   389→                                score++
   390→                        }
   391→                }
   392→        }
   393→
   394→        ansJSON, _ := json.Marshal(req.Answers)
   395→        attempt := QuizAttempt{ID: uuid.New().String(), UserID: userID.(string), QuizID: req.QuizID, Score: score, Total: len(questions), Answers: string(ansJSON)}
   396→        DB.Create(&attempt)
   397→
   398→        c.JSON(200, gin.H{"score": score, "total": len(questions), "passed": score >= int(float64(len(questions))*0.8), "attemptId": attempt.ID})
   399→}

   403→func ExamHandler(c *gin.Context) {
   404→        var req ExamActionRequest
   405→        if err := c.ShouldBindJSON(&req); err != nil {
   406→                c.JSON(400, gin.H{"error": err.Error()})
   407→                return
   408→        }
   409→
   410→        if req.Action == "verify" {
   411→                var exam Exam
   412→                if DB.Where("code = ? AND is_active = ?", req.Code, true).First(&exam).Error != nil {
   413→                        c.JSON(404, gin.H{"error": "Идэвхтэй шалгалт олдсонгүй эсвэл код буруу байна"})
   414→                        return
   415→                }
   416→                c.JSON(200, gin.H{"exam": gin.H{"id": exam.ID, "title": exam.Title, "questionCount": exam.QuestionCount, "duration": exam.Duration}})
   417→                return
   418→        }
   419→
   420→        if req.Action == "submit" {
   421→                userID, _ := c.Get("userId")
   422→                // Check if already took
   423→                var existing ExamAttempt
   424→                if DB.Where("exam_id = ? AND user_id = ?", req.ExamID, userID).First(&existing).Error == nil {
   425→                        c.JSON(403, gin.H{"error": "Та энэ шалгалтад өмнө нь орсон байна. Дахин өгөх боломжгүй."})
   426→                        return
   427→                }
   428→
   429→                var questions []ExamQuestion
   430→                DB.Where("exam_id = ?", req.ExamID).Order("index asc").Find(&questions)
   431→
   432→                score := 0
   433→                for i, ans := range req.Answers {
   434→                        if i < len(questions) && ans == questions[i].Correct {
   435→                                score++
   436→                        }
   437→                }
   438→                passed := len(questions) > 0 && score >= int(float64(len(questions))*0.8)
   439→
   440→                ansJSON, _ := json.Marshal(req.Answers)
   441→                attempt := ExamAttempt{ID: uuid.New().String(), UserID: userID.(string), ExamID: req.ExamID, Score: score, Total: len(questions), Passed: passed, Answers: string(ansJSON), TimeSpent: req.TimeSpent}
   442→                DB.Create(&attempt)
   443→
   444→                c.JSON(200, gin.H{"score": score, "total": len(questions), "passed": passed, "timeSpent": req.TimeSpent})
   445→                return
   446→        }
   447→
   448→        c.JSON(400, gin.H{"error": "Invalid action"})
   449→}

   484→func ExportExamHandler(c *gin.Context) {
   485→        attemptID := c.Param("attemptId")
   486→        var attempt ExamAttempt
   487→        if DB.Where("id = ?", attemptID).First(&attempt).Error != nil {
   488→                c.JSON(404, gin.H{"error": "Not found"})
   489→                return
   490→        }
   491→        var exam Exam
   492→        DB.Where("id = ?", attempt.ExamID).First(&exam)
   493→        var questions []ExamQuestion
   494→        DB.Where("exam_id = ?", attempt.ExamID).Order("index asc").Find(&questions)
   495→
   496→        var answers []int
   497→        json.Unmarshal([]byte(attempt.Answers), &answers)
   498→
   499→        c.Header("Content-Type", "text/csv; charset=utf-8")
   500→        c.Header("Content-Disposition", fmt.Sprintf("attachment; filename=exam_%s_result.csv", exam.Code))
   501→        writer := csv.NewWriter(c.Writer)
   502→        writer.Write([]string{"Шалгалт", exam.Title, "", "Код", exam.Code, "", "Огноо", attempt.CreatedAt.Format("2006-01-02 15:04:05")})
   503→        writer.Write([]string{"#", "Асуулт", "A", "B", "C", "D", "Таны хариулт", "Зөв хариулт", "Дүн"})
   504→        for i, q := range questions {
   505→                ansStr := "-"
   506→                correctStr := string([]string{"A", "B", "C", "D"}[q.Correct])
   507→                if i < len(answers) && answers[i] >= 0 {
   508→                        ansStr = string([]string{"A", "B", "C", "D"}[answers[i]])
   509→                }
   510→                status := "Буруу"
   511→                if i < len(answers) && answers[i] == q.Correct {
   512→                        status = "Зөв"
   513→                }
   514→                writer.Write([]string{fmt.Sprintf("%d", i+1), q.Question, q.OptionA, q.OptionB, q.OptionC, q.OptionD, ansStr, correctStr, status})
   515→        }
   516→        writer.Write([]string{"", "", "", "", "", "", "", "", ""})
   517→        writer.Write([]string{"Нийт", "", "", "", "", "", fmt.Sprintf("%d/%d", attempt.Score, attempt.Total), "", fmt.Sprintf("%v", attempt.Passed)})
   518→        writer.Flush()
   519→}

   602→func GetSurveyResultsHandler(c *gin.Context) {
   603→        var responses []SurveyResponse
   604→        DB.Order("created_at desc").Find(&responses)
   605→
   606→        type Stat struct {
   607→                QuestionID   string      `json:"questionId"`
   608→                Ratings      []int       `json:"ratings"`
   609→                AvgRating    float64     `json:"avgRating"`
   610→                YesCount     int         `json:"yesCount"`
   611→                NoCount      int         `json:"noCount"`
   612→                TextAnswers  []string    `json:"textAnswers"`
   613→        }
   614→
   615→        // Aggregate
   616→        questionStats := make(map[string]*Stat)
   617→        totalResponses := len(responses)
   618→
   619→        for _, r := range responses {
   620→                var parsed map[string]interface{}
   621→                json.Unmarshal([]byte(r.Responses), &parsed)
   622→                for qID, val := range parsed {
   623→                        if questionStats[qID] == nil {
   624→                                questionStats[qID] = &Stat{QuestionID: qID}
   625→                        }
   626→                        stat := questionStats[qID]
   627→                        switch v := val.(type) {
   628→                        case float64:
   629→                                stat.Ratings = append(stat.Ratings, int(v))
   630→                        case string:
   631→                                if v == "yes" {
   632→                                        stat.YesCount++
   633→                                } else if v == "no" {
   634→                                        stat.NoCount++
   635→                                } else {
   636→                                        stat.TextAnswers = append(stat.TextAnswers, v)
   637→                                }
   638→                        }
   639→                }
   640→        }
   641→
   642→        var stats []*Stat
   643→        for _, s := range questionStats {
   644→                if len(s.Ratings) > 0 {
   645→                        sum := 0
   646→                        for _, r := range s.Ratings { sum += r }
   647→                        s.AvgRating = float64(sum) / float64(len(s.Ratings))
   648→                }
   649→                stats = append(stats, s)
   650→        }
   651→
   652→        c.JSON(200, gin.H{"totalResponses": totalResponses, "stats": stats, "responses": responses})
   653→}
