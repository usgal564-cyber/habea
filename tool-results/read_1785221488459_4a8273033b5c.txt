   219→/* ─── Step 2: Quiz Taking ───────────────────────────────── */
   220→
   221→function QuizTaking({
   222→  quiz,
   223→  onFinish,
   224→  onBack,
   225→}: {
   226→  quiz: QuizListItem;
   227→  onFinish: (answers: number[]) => void;
   228→  onBack: () => void;
   229→}) {
   230→  const [currentPage, setCurrentPage] = useState(1);
   231→  const [totalPages, setTotalPages] = useState(1);
   232→  const [questions, setQuestions] = useState<Question[]>([]);
   233→  const [answers, setAnswers] = useState<Record<string, number>>({});
   234→  const [isLoading, setIsLoading] = useState(true);
   235→  const [isSubmitting, setIsSubmitting] = useState(false);
   236→
   237→  const fetchQuestions = useCallback(
   238→    async (page: number) => {
   239→      setIsLoading(true);
   240→      try {
   241→        const res = await fetch(
   242→          `/api/quiz?slug=${quiz.slug}&page=${page}&limit=${QUESTIONS_PER_PAGE}`
   243→        );
   244→        const data = await res.json();
   245→        if (data.quiz && data.questions) {
   246→          setQuestions(data.questions);
   247→          setTotalPages(data.pagination.totalPages);
   248→        } else {
   249→          toast.error("Алдаа", {
   250→            description: "Асуултуудыг ачааллахад алдаа гарлаа.",
   251→          });
   252→        }
   253→      } catch {
   254→        toast.error("Холболтын алдаа");
   255→      } finally {
   256→        setIsLoading(false);
   257→      }
   258→    },
   259→    [quiz.slug]
   260→  );
   261→
   262→  // Fetch first page
   263→  useEffect(() => {
   264→    fetchQuestions(1);
   265→  }, [fetchQuestions]);
   266→
   267→  const handleSelectAnswer = (questionId: string, optionIndex: number) => {
   268→    setAnswers((prev) => ({ ...prev, [questionId]: optionIndex }));
   269→  };
   270→
   271→  const handlePageChange = (page: number) => {
   272→    setCurrentPage(page);
   273→    fetchQuestions(page);
   274→    window.scrollTo({ top: 0, behavior: "smooth" });
   275→  };
   276→
   277→  const handleFinish = async () => {
   278→    setIsSubmitting(true);
   279→    try {
   280→      const allAnswers: number[] = [];
   281→      // Fetch all pages to build ordered answer array
   282→      for (let p = 1; p <= totalPages; p++) {
   283→        const res = await fetch(
   284→          `/api/quiz?slug=${quiz.slug}&page=${p}&limit=${QUESTIONS_PER_PAGE}`
   285→        );
   286→        const data = await res.json();
   287→        if (data.questions) {
   288→          for (const q of data.questions) {
   289→            allAnswers.push(answers[q.id] ?? -1);
   290→          }
   291→        }
   292→      }
   293→      onFinish(allAnswers);
   294→    } catch {
   295→      toast.error("Алдаа", {
   296→        description: "Хариултуудыг илгээхэд алдаа гарлаа.",
   297→      });
   298→    } finally {
   299→      setIsSubmitting(false);
   300→    }
   301→  };
   302→
   303→  const answeredOnPage = questions.filter(
   304→    (q) => answers[q.id] !== undefined
   305→  ).length;
   306→  const progressPercent =
   307→    totalPages > 0
   308→      ? ((currentPage - 1) / totalPages) * 100 +
   309→        (answeredOnPage / (questions.length || 1) / totalPages) * 100
   310→      : 0;
   311→
   312→  return (
   313→    <motion.div
   314→      initial={{ opacity: 0, y: 20 }}
   315→      animate={{ opacity: 1, y: 0 }}
   316→      transition={{ duration: 0.4 }}
   317→    >
   318→      {/* Top Bar */}
   319→      <div className="mb-6 flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   320→        <div className="flex items-center gap-3">
   321→          <Button
   322→            variant="ghost"
   323→            size="icon"
   324→            className="text-brand-600 hover:bg-brand-50 hover:text-brand-800"
   325→            onClick={onBack}
   326→          >
   327→            <ArrowLeft className="size-5" />
   328→          </Button>
   329→          <div>
   330→            <h2 className="text-xl font-bold text-brand-900 sm:text-2xl">
   331→              {quiz.title}
   332→            </h2>
   333→            <p className="text-sm text-muted-foreground">{quiz.description}</p>
   334→          </div>
   335→        </div>
   336→        <Badge className="w-fit bg-brand-50 text-brand-700 hover:bg-brand-100">
   337→          <ListChecks className="mr-1 size-3.5" />
   338→          Хуудас {currentPage} / {totalPages}
   339→        </Badge>
   340→      </div>
   341→
   342→      {/* Progress Bar */}
   343→      <div className="mb-6 space-y-2">
   344→        <div className="flex items-center justify-between text-sm">
   345→          <span className="text-muted-foreground">
   346→            Хуудасны дэвсгэр: {currentPage} / {totalPages}
   347→          </span>
   348→          <span className="font-medium text-brand-700">
   349→            Энэ хуудас: {answeredOnPage}/{questions.length} хариулсан
   350→          </span>
   351→        </div>
   352→        <Progress
   353→          value={progressPercent}
   354→          className="h-2.5 bg-brand-100"
   355→        />
   356→      </div>
   357→
   358→      {/* Questions */}
   359→      {isLoading ? (
   360→        <div className="space-y-6">
   361→          {Array.from({ length: 5 }).map((_, i) => (
   362→            <Card key={i} className="p-6">
   363→              <Skeleton className="mb-4 h-5 w-3/4" />
   364→              <div className="space-y-3">
   365→                {Array.from({ length: 4 }).map((_, j) => (
   366→                  <Skeleton key={j} className="h-10 w-full" />
   367→                ))}
   368→              </div>
   369→            </Card>
   370→          ))}
   371→        </div>
   372→      ) : (
   373→        <AnimatePresence mode="wait">
   374→          <motion.div
   375→            key={currentPage}
   376→            variants={pageTransition}
   377→            initial="initial"
   378→            animate="animate"
   379→            exit="exit"
   380→            className="space-y-4"
   381→          >
   382→            {questions.map((question, qIdx) => {
   383→              const globalIndex =
   384→                (currentPage - 1) * QUESTIONS_PER_PAGE + qIdx + 1;
   385→              const selectedAnswer = answers[question.id];
   386→
   387→              return (
   388→                <motion.div
   389→                  key={question.id}
   390→                  variants={itemVariants}
   391→                  initial="hidden"
   392→                  animate="visible"
   393→                >
   394→                  <Card className="overflow-hidden border-brand-100 bg-white">
   395→                    <CardContent className="p-5 sm:p-6">
   396→                      {/* Question Number & Text */}
   397→                      <div className="mb-4 flex items-start gap-3">
   398→                        <span className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-brand-600 text-xs font-bold text-white">
   399→                          {globalIndex}
   400→                        </span>
   401→                        <p className="text-base font-medium leading-relaxed text-foreground sm:text-lg">
   402→                          {question.questionText}
   403→                        </p>
   404→                      </div>
   405→
   406→                      {/* Options */}
   407→                      <div className="ml-10 space-y-2.5">
   408→                        {OPTION_KEYS.map((key, optIdx) => {
   409→                          const isSelected = selectedAnswer === optIdx;
   410→                          return (
   411→                            <button
   412→                              key={key}
   413→                              type="button"
   414→                              onClick={() =>
   415→                                handleSelectAnswer(question.id, optIdx)
   416→                              }
   417→                              className={[
   418→                                "flex w-full items-start gap-3 rounded-lg border-2 px-4 py-3 text-left text-sm transition-all duration-200 sm:text-base",
   419→                                isSelected
   420→                                  ? "border-brand-500 bg-brand-50 text-brand-900 shadow-sm ring-1 ring-brand-200"
   421→                                  : "border-transparent bg-muted/50 text-foreground hover:border-brand-200 hover:bg-brand-50/50",
   422→                              ].join(" ")}
   423→                            >
   424→                              <span
   425→                                className={[
   426→                                  "flex h-6 w-6 shrink-0 items-center justify-center rounded-full border-2 text-xs font-bold transition-colors",
   427→                                  isSelected
   428→                                    ? "border-brand-600 bg-brand-600 text-white"
   429→                                    : "border-muted-foreground/30 text-muted-foreground",
   430→                                ].join(" ")}
   431→                              >
   432→                                {OPTION_LABELS[optIdx]}
   433→                              </span>
   434→                              <span className="pt-0.5 leading-relaxed">
   435→                                {question[key]}
   436→                              </span>
   437→                            </button>
   438→                          );
   439→                        })}
   440→                      </div>
   441→                    </CardContent>
   442→                  </Card>
   443→                </motion.div>
   444→              );
   445→            })}
   446→          </motion.div>
   447→        </AnimatePresence>
   448→      )}
   449→
   450→      {/* Page Navigation */}
   451→      <div className="mt-8 flex flex-col items-center gap-4 sm:flex-row sm:justify-between">
   452→        {/* Previous Button */}
   453→        <Button
   454→          variant="outline"
   455→          className="border-brand-200 text-brand-700 hover:bg-brand-50 hover:text-brand-800"
   456→          onClick={() => handlePageChange(currentPage - 1)}
   457→          disabled={currentPage <= 1 || isLoading}
   458→        >
   459→          <ChevronLeft className="size-4" />
   460→          Өмнөх
   461→        </Button>
   462→
   463→        {/* Page Numbers */}
   464→        <div className="flex items-center gap-1.5">
   465→          {Array.from({ length: totalPages }, (_, i) => i + 1).map((page) => (
   466→            <button
   467→              key={page}
   468→              type="button"
   469→              onClick={() => handlePageChange(page)}
   470→              disabled={isLoading}
   471→              className={[
   472→                "flex h-9 w-9 items-center justify-center rounded-md text-sm font-medium transition-all disabled:opacity-50",
   473→                page === currentPage
   474→                  ? "bg-brand-600 text-white shadow-sm"
   475→                  : "text-muted-foreground hover:bg-brand-50 hover:text-brand-700",
   476→              ].join(" ")}
   477→            >
   478→              {page}
   479→            </button>
   480→          ))}
   481→        </div>
   482→
   483→        {/* Next / Finish Button */}
   484→        {currentPage < totalPages ? (
   485→          <Button
   486→            className="bg-brand-600 text-white hover:bg-brand-700"
   487→            onClick={() => handlePageChange(currentPage + 1)}
   488→            disabled={isLoading}
   489→          >
   490→            Дараах
   491→            <ChevronRight className="size-4" />
   492→          </Button>
   493→        ) : (
   494→          <Button
   495→            className="bg-brand-600 text-white hover:bg-brand-700"
   496→            onClick={handleFinish}
   497→            disabled={isSubmitting || isLoading}
   498→          >
   499→            {isSubmitting ? (
   500→              <Loader2 className="size-4 animate-spin" />
   501→            ) : (
   502→              <CheckCircle2 className="size-4" />
   503→            )}
   504→            Дуусгах
   505→          </Button>
   506→        )}
   507→      </div>
   508→    </motion.div>
   509→  );
   510→}

   512→/* ─── Step 3: Results ───────────────────────────────────── */
   513→
   514→function QuizResults({
   515→  quiz,
   516→  result,
   517→  onRetry,
   518→  onBackToList,
   519→}: {
   520→  quiz: QuizListItem;
   521→  result: QuizResult;
   522→  onRetry: () => void;
   523→  onBackToList: () => void;
   524→}) {
   525→  const percentage = Math.round((result.score / result.total) * 100);
   526→  const isPassed = result.passed;
   527→
   528→  return (
   529→    <motion.div
   530→      initial={{ opacity: 0, scale: 0.95 }}
   531→      animate={{ opacity: 1, scale: 1 }}
   532→      transition={{ duration: 0.5, ease: "easeOut" }}
   533→      className="mx-auto max-w-lg"
   534→    >
   535→      <Card className="overflow-hidden border-2 bg-white">
   536→        {/* Top Accent */}
   537→        <div
   538→          className={[
   539→            "h-2",
   540→            isPassed
   541→              ? "bg-gradient-to-r from-brand-500 to-brand-700"
   542→              : "bg-gradient-to-r from-red-400 to-red-600",
   543→          ].join(" ")}
   544→        />
   545→
   546→        <CardContent className="p-8 sm:p-10">
   547→          <div className="flex flex-col items-center text-center">
   548→            {/* Icon Animation */}
   549→            <motion.div
   550→              initial={{ scale: 0 }}
   551→              animate={{ scale: 1 }}
   552→              transition={{
   553→                type: "spring",
   554→                stiffness: 200,
   555→                damping: 15,
   556→                delay: 0.2,
   557→              }}
   558→              className={[
   559→                "mb-6 flex h-24 w-24 items-center justify-center rounded-full",
   560→                isPassed
   561→                  ? "bg-brand-100 text-brand-600"
   562→                  : "bg-red-100 text-red-500",
   563→              ].join(" ")}
   564→            >
   565→              {isPassed ? (
   566→                <Trophy className="size-12" />
   567→              ) : (
   568→                <XCircle className="size-12" />
   569→              )}
   570→            </motion.div>
   571→
   572→            {/* Title */}
   573→            <motion.h2
   574→              initial={{ opacity: 0, y: 10 }}
   575→              animate={{ opacity: 1, y: 0 }}
   576→              transition={{ delay: 0.4 }}
   577→              className={[
   578→                "text-2xl font-bold sm:text-3xl",
   579→                isPassed ? "text-brand-800" : "text-red-700",
   580→              ].join(" ")}
   581→            >
   582→              {isPassed ? "Тэнцлээ!" : "Тэнцээгүй"}
   583→            </motion.h2>

   712→/* ─── Main Component ──────────────────────────────────────── */
   713→
   714→type Step = "selection" | "taking" | "results";
   715→
   716→export default function QuizPage() {
   717→  const [step, setStep] = useState<Step>("selection");
   718→  const [selectedQuiz, setSelectedQuiz] = useState<QuizListItem | null>(null);
   719→  const [result, setResult] = useState<QuizResult | null>(null);
   720→  const [isSubmitting, setIsSubmitting] = useState(false);
   721→
   722→  const handleSelectQuiz = useCallback((quiz: QuizListItem) => {
   723→    setSelectedQuiz(quiz);
   724→    setResult(null);
   725→    setStep("taking");
   726→  }, []);
   727→
   728→  const handleFinish = useCallback(
   729→    async (answers: number[]) => {
   730→      if (!selectedQuiz) return;
   731→
   732→      setIsSubmitting(true);
   733→      try {
   734→        const res = await fetch("/api/quiz", {
   735→          method: "POST",
   736→          headers: { "Content-Type": "application/json" },
   737→          body: JSON.stringify({
   738→            quizId: selectedQuiz.id,
   739→            answers,
   740→          }),
   741→        });
   742→
   743→        const data = await res.json();
   744→
   745→        if (res.ok) {
   746→          setResult({
   747→            attemptId: data.attemptId,
   748→            score: data.score,
   749→            total: data.total,
   750→            passed: data.passed,
   751→          });
   752→          setStep("results");
   753→
   754→          if (data.passed) {
   755→            toast.success("Баяр хүргэе!", {
   756→              description: `Та ${data.score}/${data.total} оноо авч тэнцлээ!`,
   757→            });
   758→          } else {
   759→            toast.error("Тэнцээгүй", {
   760→              description: `Та ${data.score}/${data.total} оноо авсан байна. 70% шаардлагатай.`,
   761→            });
   762→          }
   763→        } else {
   764→          toast.error("Алдаа", {
   765→            description: data.error || "Хариултуудыг илгээхэд алдаа гарлаа.",
   766→          });
   767→        }
   768→      } catch {
   769→        toast.error("Холболтын алдаа");
   770→      } finally {
   771→        setIsSubmitting(false);
   772→      }
   773→    },
   774→    [selectedQuiz]
   775→  );
   776→
   777→  const handleRetry = useCallback(() => {
   778→    setResult(null);
   779→    setStep("taking");
   780→  }, []);
   781→
   782→  const handleBackToList = useCallback(() => {
   783→    setSelectedQuiz(null);
   784→    setResult(null);
   785→    setStep("selection");
   786→  }, []);
   787→
   788→  return (
   789→    <section className="w-full">
   790→      {step === "selection" && <QuizSelection onSelect={handleSelectQuiz} />}
   791→
   792→      {step === "taking" && selectedQuiz && (
   793→        <QuizTaking
   794→          quiz={selectedQuiz}
   795→          onFinish={handleFinish}
   796→          onBack={handleBackToList}
   797→        />
   798→      )}
   799→
   800→      {step === "results" && selectedQuiz && result && (
   801→        <QuizResults
   802→          quiz={selectedQuiz}
   803→          result={result}
   804→          onRetry={handleRetry}
   805→          onBackToList={handleBackToList}
   806→        />
   807→      )}
   808→
   809→      {/* Submit overlay */}
   810→      <AnimatePresence>
   811→        {isSubmitting && (
   812→          <motion.div
   813→            initial={{ opacity: 0 }}
   814→            animate={{ opacity: 1 }}
   815→            exit={{ opacity: 0 }}
   816→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 backdrop-blur-sm"
   817→          >
   818→            <div className="flex flex-col items-center gap-4 rounded-2xl bg-white p-8 shadow-2xl">
   819→              <Loader2 className="size-10 animate-spin text-brand-600" />
   820→              <p className="text-lg font-medium text-brand-900">
   821→                Хариултуудыг шалгаж байна...
   822→              </p>
   823→            </div>
   824→          </motion.div>
   825→        )}
   826→      </AnimatePresence>
   827→    </section>
   828→  );
   829→}
