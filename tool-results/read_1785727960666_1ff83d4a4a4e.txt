   151→func generateCode() string {
   152→        const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"
   153→        code := make([]byte, 6)
   154→        for i := range code {
   155→                n, _ := rand.Int(rand.Reader, big.NewInt(int64(len(chars))))
   156→                code[i] = chars[n.Int64()]
   157→        }
   158→        return string(code)
   159→}

   176→func RegisterHandler(c *gin.Context) {
   177→        var req RegisterRequest
   178→        if err := c.ShouldBindJSON(&req); err != nil {
   179→                c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
   180→                return
   181→        }
   182→        var existing User
   183→        if DB.Where("email = ?", req.Email).First(&existing).Error == nil {
   184→                c.JSON(http.StatusConflict, gin.H{"error": "Бүртгэлтэй имэйл байна"})
   185→                return
   186→        }
   187→        hash, _ := bcrypt.GenerateFromPassword([]byte(req.Password), bcrypt.DefaultCost)
   188→        user := User{ID: uuid.New().String(), FirstName: req.FirstName, LastName: req.LastName, Email: req.Email, Phone: req.Phone, Password: string(hash), Address: req.Address, SecondaryPhone: req.SecondaryPhone, Role: "USER"}
   189→        if err := DB.Create(&user).Error; err != nil {
   190→                c.JSON(500, gin.H{"error": "Failed to create user"})
   191→                return
   192→        }
   193→        token, _ := GenerateToken(user.ID, user.Email, user.Role, user.FirstName+" "+user.LastName)
   194→        c.JSON(201, gin.H{"user": userResponse(user), "token": token})
   195→}

   319→func QuizSubmitHandler(c *gin.Context) {
   320→        var req QuizSubmitRequest
   321→        if err := c.ShouldBindJSON(&req); err != nil {
   322→                c.JSON(400, gin.H{"error": err.Error()})
   323→                return
   324→        }
   325→        userID, _ := c.Get("userId")
   326→        var questions []QuizQuestion
   327→        DB.Where("quiz_id = ?", req.QuizID).Order("index asc").Find(&questions)
   328→
   329→        score := 0
   330→        for _, ans := range req.Answers {
   331→                for _, q := range questions {
   332→                        if q.ID == ans.QuestionID && q.Correct == ans.SelectedOption {
   333→                                score++
   334→                        }
   335→                }
   336→        }
   337→
   338→        ansJSON, _ := json.Marshal(req.Answers)
   339→        attempt := QuizAttempt{ID: uuid.New().String(), UserID: userID.(string), QuizID: req.QuizID, Score: score, Total: len(questions), Answers: string(ansJSON)}
   340→        DB.Create(&attempt)
   341→
   342→        c.JSON(200, gin.H{"score": score, "total": len(questions), "passed": score >= int(float64(len(questions))*0.8), "attemptId": attempt.ID})
   343→}

   347→func ExamHandler(c *gin.Context) {
   348→        var req ExamActionRequest
   349→        if err := c.ShouldBindJSON(&req); err != nil {
   350→                c.JSON(400, gin.H{"error": err.Error()})
   351→                return
   352→        }
   353→
   354→        if req.Action == "verify" {
   355→                var exam Exam
   356→                if DB.Where("code = ? AND is_active = ?", req.Code, true).First(&exam).Error != nil {
   357→                        c.JSON(404, gin.H{"error": "Идэвхтэй шалгалт олдсонгүй эсвэл код буруу байна"})
   358→                        return
   359→                }
   360→                c.JSON(200, gin.H{"exam": gin.H{"id": exam.ID, "title": exam.Title, "questionCount": exam.QuestionCount, "duration": exam.Duration}})
   361→                return
   362→        }
   363→
   364→        if req.Action == "submit" {
   365→                userID, _ := c.Get("userId")
   366→                // Check if already took
   367→                var existing ExamAttempt
   368→                if DB.Where("exam_id = ? AND user_id = ?", req.ExamID, userID).First(&existing).Error == nil {
   369→                        c.JSON(403, gin.H{"error": "Та энэ шалгалтад өмнө нь орсон байна. Дахин өгөх боломжгүй."})
   370→                        return
   371→                }
   372→
   373→                var questions []ExamQuestion
   374→                DB.Where("exam_id = ?", req.ExamID).Order("index asc").Find(&questions)
   375→
   376→                score := 0
   377→                for i, ans := range req.Answers {
   378→                        if i < len(questions) && ans == questions[i].Correct {
   379→                                score++
   380→                        }
   381→                }
   382→                passed := len(questions) > 0 && score >= int(float64(len(questions))*0.8)
   383→
   384→                ansJSON, _ := json.Marshal(req.Answers)
   385→                attempt := ExamAttempt{ID: uuid.New().String(), UserID: userID.(string), ExamID: req.ExamID, Score: score, Total: len(questions), Passed: passed, Answers: string(ansJSON)}
   386→                DB.Create(&attempt)
   387→
   388→                c.JSON(200, gin.H{"score": score, "total": len(questions), "passed": passed})
   389→                return
   390→        }
   391→
   392→        c.JSON(400, gin.H{"error": "Invalid action"})
   393→}

   428→func ExportExamHandler(c *gin.Context) {
   429→        attemptID := c.Param("attemptId")
   430→        var attempt ExamAttempt
   431→        if DB.Where("id = ?", attemptID).First(&attempt).Error != nil {
   432→                c.JSON(404, gin.H{"error": "Not found"})
   433→                return
   434→        }
   435→        var exam Exam
   436→        DB.Where("id = ?", attempt.ExamID).First(&exam)
   437→        var questions []ExamQuestion
   438→        DB.Where("exam_id = ?", attempt.ExamID).Order("index asc").Find(&questions)
   439→
   440→        var answers []int
   441→        json.Unmarshal([]byte(attempt.Answers), &answers)
   442→
   443→        c.Header("Content-Type", "text/csv; charset=utf-8")
   444→        c.Header("Content-Disposition", fmt.Sprintf("attachment; filename=exam_%s_result.csv", exam.Code))
   445→        writer := csv.NewWriter(c.Writer)
   446→        writer.Write([]string{"Шалгалт", exam.Title, "", "Код", exam.Code, "", "Огноо", attempt.CreatedAt.Format("2006-01-02 15:04:05")})
   447→        writer.Write([]string{"#", "Асуулт", "A", "B", "C", "D", "Таны хариулт", "Зөв хариулт", "Дүн"})
   448→        for i, q := range questions {
   449→                ansStr := "-"
   450→                correctStr := string([]string{"A", "B", "C", "D"}[q.Correct])
   451→                if i < len(answers) && answers[i] >= 0 {
   452→                        ansStr = string([]string{"A", "B", "C", "D"}[answers[i]])
   453→                }
   454→                status := "Буруу"
   455→                if i < len(answers) && answers[i] == q.Correct {
   456→                        status = "Зөв"
   457→                }
   458→                writer.Write([]string{fmt.Sprintf("%d", i+1), q.Question, q.OptionA, q.OptionB, q.OptionC, q.OptionD, ansStr, correctStr, status})
   459→        }
   460→        writer.Write([]string{"", "", "", "", "", "", "", "", ""})
   461→        writer.Write([]string{"Нийт", "", "", "", "", "", fmt.Sprintf("%d/%d", attempt.Score, attempt.Total), "", fmt.Sprintf("%v", attempt.Passed)})
   462→        writer.Flush()
   463→}

   546→func GetSurveyResultsHandler(c *gin.Context) {
   547→        var responses []SurveyResponse
   548→        DB.Order("created_at desc").Find(&responses)
   549→
   550→        type Stat struct {
   551→                QuestionID   string      `json:"questionId"`
   552→                Ratings      []int       `json:"ratings"`
   553→                AvgRating    float64     `json:"avgRating"`
   554→                YesCount     int         `json:"yesCount"`
   555→                NoCount      int         `json:"noCount"`
   556→                TextAnswers  []string    `json:"textAnswers"`
   557→        }
   558→
   559→        // Aggregate
   560→        questionStats := make(map[string]*Stat)
   561→        totalResponses := len(responses)
   562→
   563→        for _, r := range responses {
   564→                var parsed map[string]interface{}
   565→                json.Unmarshal([]byte(r.Responses), &parsed)
   566→                for qID, val := range parsed {
   567→                        if questionStats[qID] == nil {
   568→                                questionStats[qID] = &Stat{QuestionID: qID}
   569→                        }
   570→                        stat := questionStats[qID]
   571→                        switch v := val.(type) {
   572→                        case float64:
   573→                                stat.Ratings = append(stat.Ratings, int(v))
   574→                        case string:
   575→                                if v == "yes" {
   576→                                        stat.YesCount++
   577→                                } else if v == "no" {
   578→                                        stat.NoCount++
   579→                                } else {
   580→                                        stat.TextAnswers = append(stat.TextAnswers, v)
   581→                                }
   582→                        }
   583→                }
   584→        }
   585→
   586→        var stats []*Stat
   587→        for _, s := range questionStats {
   588→                if len(s.Ratings) > 0 {
   589→                        sum := 0
   590→                        for _, r := range s.Ratings { sum += r }
   591→                        s.AvgRating = float64(sum) / float64(len(s.Ratings))
   592→                }
   593→                stats = append(stats, s)
   594→        }
   595→
   596→        c.JSON(200, gin.H{"totalResponses": totalResponses, "stats": stats, "responses": responses})
   597→}
