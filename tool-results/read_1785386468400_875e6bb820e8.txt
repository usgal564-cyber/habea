   232→  quiz,
   233→  onPaid,
   234→  onBack,
   235→}: {
   236→  quiz: QuizListItem;
   237→  onPaid: () => void;
   238→  onBack: () => void;
   239→}) {
   240→  const { token } = useAuthStore();
   241→  const [paying, setPaying] = useState(false);
   242→
   243→  const handlePay = async () => {
   244→    setPaying(true);
   245→    try {
   246→      const res = await fetch("/api/quiz/payment", {
   247→        method: "POST",
   248→        headers: {
   249→          "Content-Type": "application/json",
   250→          Authorization: `Bearer ${token}`,
   251→        },
   252→        body: JSON.stringify({ quizId: quiz.id }),
   253→      });
   254→      const data = await res.json();
   255→      if (res.ok) {
   256→        toast.success("Төлбөр амжилттай");
   257→        onPaid();
   258→      } else {
   259→        toast.error("Төлбөр амжилтгүй", { description: data.error || "Дахин оролдоно уу" });

   339→/* ─── Step 3: Quiz Taking ───────────────────────────────── */
   340→
   341→function QuizTaking({
   342→  quiz,
   343→  onFinish,
   344→  onBack,
   345→}: {
   346→  quiz: QuizListItem;
   347→  onFinish: (answers: number[]) => void;
   348→  onBack: () => void;
   349→}) {
   350→  const [currentPage, setCurrentPage] = useState(1);
   351→  const [totalPages, setTotalPages] = useState(1);
   352→  const [questions, setQuestions] = useState<Question[]>([]);
   353→  const [answers, setAnswers] = useState<Record<string, number>>({});
   354→  const [isLoading, setIsLoading] = useState(true);
   355→  const [isSubmitting, setIsSubmitting] = useState(false);
   356→
   357→  const fetchQuestions = useCallback(
   358→    async (page: number) => {
   359→      setIsLoading(true);
   360→      try {
   361→        const res = await fetch(
   362→          `/api/quiz?slug=${quiz.slug}&page=${page}&limit=${QUESTIONS_PER_PAGE}`
   363→        );
   364→        const data = await res.json();
   365→        if (data.quiz && data.questions) {
   366→          setQuestions(data.questions);
   367→          setTotalPages(data.pagination.totalPages);
   368→        } else {
   369→          toast.error("Алдаа", {
   370→            description: "Асуултуудыг ачааллахад алдаа гарлаа.",
   371→          });
   372→        }
   373→      } catch {
   374→        toast.error("Холболтын алдаа");
   375→      } finally {
   376→        setIsLoading(false);
   377→      }
   378→    },
   379→    [quiz.slug]
   380→  );
   381→
   382→  useEffect(() => {
   383→    fetchQuestions(1);
   384→  }, [fetchQuestions]);
   385→
   386→  const handleSelectAnswer = (questionId: string, optionIndex: number) => {
   387→    setAnswers((prev) => ({ ...prev, [questionId]: optionIndex }));
   388→  };
   389→
   390→  const handlePageChange = (page: number) => {
   391→    setCurrentPage(page);
   392→    fetchQuestions(page);
   393→    window.scrollTo({ top: 0, behavior: "smooth" });
   394→  };
   395→
   396→  const handleFinish = async () => {
   397→    setIsSubmitting(true);
   398→    try {
   399→      const allAnswers: number[] = [];
   400→      for (let p = 1; p <= totalPages; p++) {
   401→        const res = await fetch(
   402→          `/api/quiz?slug=${quiz.slug}&page=${p}&limit=${QUESTIONS_PER_PAGE}`
   403→        );
   404→        const data = await res.json();
   405→        if (data.questions) {
   406→          for (const q of data.questions) {
   407→            allAnswers.push(answers[q.id] ?? -1);
   408→          }
   409→        }
   410→      }
   411→      onFinish(allAnswers);
   412→    } catch {
   413→      toast.error("Алдаа", {
   414→        description: "Хариултуудыг илгээхэд алдаа гарлаа.",
   415→      });
   416→    } finally {
   417→      setIsSubmitting(false);
   418→    }
   419→  };
   420→
   421→  const answeredCount = Object.keys(answers).length;
   422→
   423→  return (
   424→    <motion.div initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }}>
   425→      <div className="mb-6 flex items-center justify-between">
   426→        <button
   427→          onClick={onBack}
   428→          className="flex items-center gap-2 text-sm text-muted-foreground hover:text-brand-700 transition-colors"
   429→        >
   430→          <ArrowLeft className="size-4" />
   431→          Буцах
   432→        </button>
   433→        <h2 className="text-lg font-bold text-brand-900">{quiz.title}</h2>
   434→      </div>
   435→
   436→      <Card className="border-brand-100">
   437→        <CardHeader className="pb-3">
   438→          <div className="flex items-center justify-between text-sm">
   439→            <span className="text-muted-foreground">
   440→              Хуудас {currentPage} / {totalPages}
   441→            </span>
   442→            <span className="text-muted-foreground">
   443→              Хариулсан: {answeredCount}
   444→            </span>
   445→          </div>
   446→          <Progress value={(answeredCount / (quiz.questionCount || 20)) * 100} className="h-2" />
   447→        </CardHeader>
   448→
   449→        <CardContent>
   450→          {isLoading ? (
   451→            <div className="py-16 flex justify-center">
   452→              <Loader2 className="size-8 animate-spin text-brand-500" />
   453→            </div>
   454→          ) : (
   455→            <div className="space-y-6">
   456→              {questions.map((q, idx) => {
   457→                const globalIdx = (currentPage - 1) * QUESTIONS_PER_PAGE + idx;
   458→                return (
   459→                  <motion.div
   460→                    key={q.id}
   461→                    initial={{ opacity: 0, y: 10 }}
   462→                    animate={{ opacity: 1, y: 0 }}
   463→                    transition={{ delay: idx * 0.03 }}
   464→                    className="bg-gray-50 rounded-xl p-4"
   465→                  >
   466→                    <p className="font-medium text-brand-900 mb-3">
   467→                      {globalIdx + 1}. {q.questionText}
   468→                    </p>
   469→                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   470→                      {[
   471→                        { key: 0, label: "A", text: q.optionA },
   472→                        { key: 1, label: "B", text: q.optionB },
   473→                        { key: 2, label: "C", text: q.optionC },
   474→                        { key: 3, label: "D", text: q.optionD },
   475→                      ].map((opt) => (
   476→                        <button
   477→                          key={opt.key}
   478→                          onClick={() => handleSelectAnswer(q.id, opt.key)}
   479→                          className={`text-left p-3 rounded-lg border-2 transition-all text-sm ${
   480→                            answers[q.id] === opt.key
   481→                              ? "border-brand-500 bg-brand-50 text-brand-900"
   482→                              : "border-gray-200 hover:border-brand-300 hover:bg-brand-50/50"
   483→                          }`}
   484→                        >
   485→                          <span className="font-bold mr-2">{opt.label})</span>
   486→                          {opt.text}
   487→                        </button>
   488→                      ))}
   489→                    </div>
   490→                  </motion.div>
   491→                );
   492→              })}
   493→
   494→              {/* Pagination */}
   495→              <div className="flex items-center justify-between pt-4 border-t">
   496→                <Button
   497→                  variant="outline"
   498→                  onClick={() => handlePageChange(Math.max(1, currentPage - 1))}
   499→                  disabled={currentPage === 1}
   500→                >
   501→                  <ChevronLeft className="w-4 h-4 mr-1" /> Өмнөх
   502→                </Button>
   503→                <div className="flex gap-1 flex-wrap justify-center">
   504→                  {Array.from({ length: totalPages }, (_, i) => i + 1).map((p) => (
   505→                    <button
   506→                      key={p}
   507→                      onClick={() => handlePageChange(p)}
   508→                      className={`w-8 h-8 rounded-lg text-sm font-medium transition-all ${
   509→                        p === currentPage
   510→                          ? "bg-brand-600 text-white"
   511→                          : "bg-gray-100 hover:bg-brand-100 text-gray-700"
   512→                      }`}
   513→                    >
   514→                      {p}
   515→                    </button>
   516→                  ))}
   517→                </div>
   518→                {currentPage < totalPages ? (
   519→                  <Button
   520→                    onClick={() => handlePageChange(Math.min(totalPages, currentPage + 1))}
   521→                  >
   522→                    Дараах <ChevronRight className="w-4 h-4 ml-1" />
   523→                  </Button>
   524→                ) : (
   525→                  <Button
   526→                    onClick={handleFinish}
   527→                    className="bg-brand-600 hover:bg-brand-700"
   528→                    disabled={isSubmitting}
   529→                  >
   530→                    {isSubmitting ? (
   531→                      <>
   532→                        <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   533→                        Илгээж байна...
   534→                      </>
   535→                    ) : (
   536→                      "Дуусгах"
   537→                    )}
   538→                  </Button>
   539→                )}
   540→              </div>
   541→            </div>
   542→          )}
   543→        </CardContent>
   544→      </Card>
   545→    </motion.div>
   546→  );
   547→}

   558→  result: QuizResult;
   559→  onRetry: () => void;
   560→  onBackToList: () => void;
   561→}) {
   562→  const percentage = Math.round((result.score / result.total) * 100);

   570→            animate={{ scale: 1 }}

   573→            style={{
   574→              backgroundColor: result.passed ? "#dcfce7" : "#fef2f2",
   575→            }}
   576→          >
   577→            {result.passed ? (
   578→              <Trophy className="w-12 h-12 text-green-600" />
   579→            ) : (
   580→              <XCircle className="w-12 h-12 text-red-500" />
   581→            )}
   582→          </motion.div>
   583→
   584→          <h2 className="text-2xl font-bold mb-2 text-brand-900">
   585→            {result.passed ? "Баяр хүргэе! Тэнцэв" : "Амжилтгүй"}
   586→          </h2>

   622→/* ─── Main Component ──────────────────────────────────────── */
   623→
   624→type Step = "selection" | "payment" | "taking" | "results";
   625→
   626→export default function QuizPage() {
   627→  const [step, setStep] = useState<Step>("selection");
   628→  const [selectedQuiz, setSelectedQuiz] = useState<QuizListItem | null>(null);
   629→  const [result, setResult] = useState<QuizResult | null>(null);
   630→  const [isSubmitting, setIsSubmitting] = useState(false);
   631→
   632→  const handleSelectQuiz = useCallback((quiz: QuizListItem) => {
   633→    setSelectedQuiz(quiz);
   634→    setResult(null);
   635→    setStep("payment");
   636→  }, []);
   637→
   638→  const handlePaid = useCallback(() => {
   639→    setStep("taking");
   640→  }, []);
   641→
   642→  const handleFinish = useCallback(
   643→    async (answers: number[]) => {
   644→      if (!selectedQuiz) return;
   645→
   646→      setIsSubmitting(true);
   647→      try {
   648→        const res = await fetch("/api/quiz", {
   649→          method: "POST",
   650→          headers: { "Content-Type": "application/json" },
   651→          body: JSON.stringify({
   652→            quizId: selectedQuiz.id,
   653→            answers,
   654→          }),
   655→        });
   656→
   657→        const data = await res.json();
   658→
   659→        if (res.ok) {
   660→          setResult({
   661→            attemptId: data.attemptId,
   662→            score: data.score,
   663→            total: data.total,
   664→            passed: data.passed,
   665→          });
   666→          setStep("results");
   667→
   668→          if (data.passed) {
   669→            toast.success("Баяр хүргэе!", {
   670→              description: `Та ${data.score}/${data.total} оноо авч тэнцлээ!`,
   671→            });
   672→          } else {
   673→            toast.error("Тэнцээгүй", {
   674→              description: `Та ${data.score}/${data.total} оноо авсан байна. 70% шаардлагатай.`,
   675→            });
   676→          }
   677→        } else {
   678→          toast.error("Алдаа", {
   679→            description: data.error || "Хариултуудыг илгээхэд алдаа гарлаа.",
   680→          });
   681→        }
   682→      } catch {
   683→        toast.error("Холболтын алдаа");
   684→      } finally {
   685→        setIsSubmitting(false);
   686→      }
   687→    },
   688→    [selectedQuiz]
   689→  );
   690→
   691→  const handleRetry = useCallback(() => {
   692→    setResult(null);
   693→    setStep("payment");
   694→  }, []);
   695→
   696→  const handleBackToList = useCallback(() => {
   697→    setSelectedQuiz(null);
   698→    setResult(null);
   699→    setStep("selection");
   700→  }, []);
   701→
   702→  return (
   703→    <section className="w-full">
   704→      {step === "selection" && <QuizSelection onSelect={handleSelectQuiz} />}
   705→
   706→      {step === "payment" && selectedQuiz && (
   707→        <PaymentGate quiz={selectedQuiz} onPaid={handlePaid} onBack={handleBackToList} />
   708→      )}
   709→
   710→      {step === "taking" && selectedQuiz && (
   711→        <QuizTaking
   712→          quiz={selectedQuiz}
   713→          onFinish={handleFinish}
   714→          onBack={handleBackToList}
   715→        />
   716→      )}
   717→
   718→      {step === "results" && selectedQuiz && result && (
   719→        <QuizResults
   720→          quiz={selectedQuiz}
   721→          result={result}
   722→          onRetry={handleRetry}
   723→          onBackToList={handleBackToList}
   724→        />
   725→      )}
   726→
   727→      {/* Submit overlay */}
   728→      <AnimatePresence>
   729→        {isSubmitting && (
   730→          <motion.div
   731→            initial={{ opacity: 0 }}
   732→            animate={{ opacity: 1 }}
   733→            exit={{ opacity: 0 }}
   734→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 backdrop-blur-sm"
   735→          >
   736→            <div className="flex flex-col items-center gap-4 rounded-2xl bg-white p-8 shadow-2xl">
   737→              <Loader2 className="size-10 animate-spin text-brand-600" />
   738→              <p className="text-lg font-medium text-brand-900">
   739→                Хариултуудыг шалгаж байна...
   740→              </p>
   741→            </div>
   742→          </motion.div>
   743→        )}
   744→      </AnimatePresence>
   745→    </section>
   746→  );
   747→}
