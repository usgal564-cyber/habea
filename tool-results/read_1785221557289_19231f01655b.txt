   360→export async function POST(request: NextRequest) {
   361→  try {
   362→    const token = getTokenFromHeader(request);
   363→    if (!token) {
   364→      return NextResponse.json({ error: "Нэвтрээгүй байна" }, { status: 401 });
   365→    }
   366→    const payload = verifyToken(token);
   367→    if (!payload || payload.role !== "ADMIN") {
   368→      return NextResponse.json({ error: "Зөвхөн админ хандах боломжтой" }, { status: 403 });
   369→    }
   370→
   371→    // Check if quizzes already exist
   372→    const existing = await db.quiz.count();
   373→    if (existing > 0) {
   374→      return NextResponse.json({ message: `${existing} сорил аль хэдийн байна`, seeded: false });
   375→    }
   376→
   377→    let createdCount = 0;
   378→    for (const quiz of quizData) {
   379→      const created = await db.quiz.create({
   380→        data: {
   381→          title: quiz.title,
   382→          description: quiz.description,
   383→          slug: quiz.slug,
   384→          questionCount: quiz.questions.length,
   385→          questions: {
   386→            create: quiz.questions,
   387→          },
   388→        },
   389→      });
   390→      createdCount++;
   391→    }
   392→
   393→    return NextResponse.json({ message: `${createdCount} сорил амжилттай үүсгэлээ`, seeded: true });
   394→  } catch (error) {
   395→    console.error("Seed error:", error);
   396→    return NextResponse.json({ error: "Алдаа гарлаа" }, { status: 500 });
   397→  }
