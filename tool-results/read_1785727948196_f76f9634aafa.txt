   231→  quiz,
   232→  onPaid,
   233→  onBack,
   234→}: {
   235→  quiz: QuizListItem;
   236→  onPaid: () => void;
   237→  onBack: () => void;
   238→}) {
   239→  const { token } = useAuthStore();
   240→  const [paying, setPaying] = useState(false);
   241→
   242→  const handlePay = async () => {
   243→    setPaying(true);
   244→    try {
   245→      const res = await fetch("/api/quiz/payment", {
   246→        method: "POST",
   247→        headers: {
   248→          "Content-Type": "application/json",
   249→          Authorization: `Bearer ${token}`,
   250→        },
   251→        body: JSON.stringify({ quizId: quiz.id }),
   252→      });
   253→      const data = await res.json();
   254→      if (res.ok) {
   255→        toast.success("Төлбөр амжилттай");
   256→        onPaid();
   257→      } else {
   258→        toast.error("Төлбөр амжилтгүй", { description: data.error || "Дахин оролдоно уу" });

   338→/* ─── Step 3: Quiz Taking ───────────────────────────────── */
   339→
   340→function QuizTaking({
   341→  quiz,
   342→  onFinish,
   343→  onBack,
   344→}: {
   345→  quiz: QuizListItem;
   346→  onFinish: (answers: number[]) => void;
   347→  onBack: () => void;
   348→}) {
   349→  const [currentPage, setCurrentPage] = useState(1);
   350→  const [totalPages, setTotalPages] = useState(1);
   351→  const [questions, setQuestions] = useState<Question[]>([]);
   352→  const [answers, setAnswers] = useState<Record<string, number>>({});
   353→  const [isLoading, setIsLoading] = useState(true);
   354→  const [isSubmitting, setIsSubmitting] = useState(false);
   355→
   356→  const fetchQuestions = useCallback(
   357→    async (page: number) => {
   358→      setIsLoading(true);
   359→      try {
   360→        const res = await fetch(
   361→          `/api/quiz?slug=${quiz.slug}&page=${page}&limit=${QUESTIONS_PER_PAGE}`
   362→        );
   363→        const data = await res.json();
   364→        if (data.quiz && data.questions) {
   365→          setQuestions(data.questions);
   366→          setTotalPages(data.pagination.totalPages);
   367→        } else {
   368→          toast.error("Алдаа", {
   369→            description: "Асуултуудыг ачааллахад алдаа гарлаа.",
   370→          });
   371→        }
   372→      } catch {
   373→        toast.error("Холболтын алдаа");
   374→      } finally {
   375→        setIsLoading(false);
   376→      }
   377→    },
   378→    [quiz.slug]
   379→  );
   380→
   381→  useEffect(() => {
   382→    fetchQuestions(1);
   383→  }, [fetchQuestions]);
   384→
   385→  const handleSelectAnswer = (questionId: string, optionIndex: number) => {
   386→    setAnswers((prev) => ({ ...prev, [questionId]: optionIndex }));
   387→  };
   388→
   389→  const handlePageChange = (page: number) => {
   390→    setCurrentPage(page);
   391→    fetchQuestions(page);
   392→    window.scrollTo({ top: 0, behavior: "smooth" });
   393→  };
   394→
   395→  const handleFinish = async () => {
   396→    setIsSubmitting(true);
   397→    try {
   398→      const allAnswers: number[] = [];
   399→      for (let p = 1; p <= totalPages; p++) {
   400→        const res = await fetch(
   401→          `/api/quiz?slug=${quiz.slug}&page=${p}&limit=${QUESTIONS_PER_PAGE}`
   402→        );
   403→        const data = await res.json();
   404→        if (data.questions) {
   405→          for (const q of data.questions) {
   406→            allAnswers.push(answers[q.id] ?? -1);
   407→          }
   408→        }
   409→      }
   410→      onFinish(allAnswers);
   411→    } catch {
   412→      toast.error("Алдаа", {
   413→        description: "Хариултуудыг илгээхэд алдаа гарлаа.",
   414→      });
   415→    } finally {
   416→      setIsSubmitting(false);
   417→    }
   418→  };
   419→
   420→  const answeredCount = Object.keys(answers).length;
   421→
   422→  return (
   423→    <motion.div initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }}>
   424→      <div className="mb-6 flex items-center justify-between">
   425→        <button
   426→          onClick={onBack}
   427→          className="flex items-center gap-2 text-sm text-muted-foreground hover:text-brand-700 transition-colors"
   428→        >
   429→          <ArrowLeft className="size-4" />
   430→          Буцах
   431→        </button>
   432→        <h2 className="text-lg font-bold text-brand-900">{quiz.title}</h2>
   433→      </div>
   434→
   435→      <Card className="border-brand-100">
   436→        <CardHeader className="pb-3">
   437→          <div className="flex items-center justify-between text-sm">
   438→            <span className="text-muted-foreground">
   439→              Хуудас {currentPage} / {totalPages}
   440→            </span>
   441→            <span className="text-muted-foreground">
   442→              Хариулсан: {answeredCount}
   443→            </span>
   444→          </div>
   445→          <Progress value={(answeredCount / (quiz.questionCount || 20)) * 100} className="h-2" />
   446→        </CardHeader>
   447→
   448→        <CardContent>
   449→          {isLoading ? (
   450→            <div className="py-16 flex justify-center">
   451→              <Loader2 className="size-8 animate-spin text-brand-500" />
   452→            </div>
   453→          ) : (
   454→            <div className="space-y-6">
   455→              {questions.map((q, idx) => {
   456→                const globalIdx = (currentPage - 1) * QUESTIONS_PER_PAGE + idx;
   457→                return (
   458→                  <motion.div
   459→                    key={q.id}
   460→                    initial={{ opacity: 0, y: 10 }}
   461→                    animate={{ opacity: 1, y: 0 }}
   462→                    transition={{ delay: idx * 0.03 }}
   463→                    className="bg-gray-50 rounded-xl p-4"
   464→                  >
   465→                    <p className="font-medium text-brand-900 mb-3">
   466→                      {globalIdx + 1}. {q.questionText}
   467→                    </p>
   468→                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   469→                      {[
   470→                        { key: 0, label: "A", text: q.optionA },
   471→                        { key: 1, label: "B", text: q.optionB },
   472→                        { key: 2, label: "C", text: q.optionC },
   473→                        { key: 3, label: "D", text: q.optionD },
   474→                      ].map((opt) => (
   475→                        <button
   476→                          key={opt.key}
   477→                          onClick={() => handleSelectAnswer(q.id, opt.key)}
   478→                          className={`text-left p-3 rounded-lg border-2 transition-all text-sm ${
   479→                            answers[q.id] === opt.key
   480→                              ? "border-brand-500 bg-brand-50 text-brand-900"
   481→                              : "border-gray-200 hover:border-brand-300 hover:bg-brand-50/50"
   482→                          }`}
   483→                        >
   484→                          <span className="font-bold mr-2">{opt.label})</span>
   485→                          {opt.text}
   486→                        </button>
   487→                      ))}
   488→                    </div>
   489→                  </motion.div>
   490→                );
   491→              })}
   492→
   493→              {/* Pagination */}
   494→              <div className="flex items-center justify-between pt-4 border-t">
   495→                <Button
   496→                  variant="outline"
   497→                  onClick={() => handlePageChange(Math.max(1, currentPage - 1))}
   498→                  disabled={currentPage === 1}
   499→                >
   500→                  <ChevronLeft className="w-4 h-4 mr-1" /> Өмнөх
   501→                </Button>
   502→                <div className="flex gap-1 flex-wrap justify-center">
   503→                  {Array.from({ length: totalPages }, (_, i) => i + 1).map((p) => (
   504→                    <button
   505→                      key={p}
   506→                      onClick={() => handlePageChange(p)}
   507→                      className={`w-8 h-8 rounded-lg text-sm font-medium transition-all ${
   508→                        p === currentPage
   509→                          ? "bg-brand-600 text-white"
   510→                          : "bg-gray-100 hover:bg-brand-100 text-gray-700"
   511→                      }`}
   512→                    >
   513→                      {p}
   514→                    </button>
   515→                  ))}
   516→                </div>
   517→                {currentPage < totalPages ? (
   518→                  <Button
   519→                    onClick={() => handlePageChange(Math.min(totalPages, currentPage + 1))}
   520→                  >
   521→                    Дараах <ChevronRight className="w-4 h-4 ml-1" />
   522→                  </Button>
   523→                ) : (
   524→                  <Button
   525→                    onClick={handleFinish}
   526→                    className="bg-brand-600 hover:bg-brand-700"
   527→                    disabled={isSubmitting}
   528→                  >
   529→                    {isSubmitting ? (
   530→                      <>
   531→                        <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   532→                        Илгээж байна...
   533→                      </>
   534→                    ) : (
   535→                      "Дуусгах"
   536→                    )}
   537→                  </Button>
   538→                )}
   539→              </div>
   540→            </div>
   541→          )}
   542→        </CardContent>
   543→      </Card>
   544→    </motion.div>
   545→  );
   546→}

   557→  result: QuizResult;
   558→  onRetry: () => void;
   559→  onBackToList: () => void;
   560→}) {
   561→  const percentage = Math.round((result.score / result.total) * 100);
   562→

   564→    <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }}>

   569→            animate={{ scale: 1 }}

   572→            style={{
   573→              backgroundColor: result.passed ? "#dcfce7" : "#fef2f2",
   574→            }}
   575→          >
   576→            {result.passed ? (
   577→              <Trophy className="w-12 h-12 text-green-600" />
   578→            ) : (
   579→              <XCircle className="w-12 h-12 text-red-500" />
   580→            )}
   581→          </motion.div>
   582→
   583→          <h2 className="text-2xl font-bold mb-2 text-brand-900">
   584→            {result.passed ? "Баяр хүргэе! Тэнцэв" : "Амжилтгүй"}
   585→          </h2>

   621→/* ─── Main Component ──────────────────────────────────────── */
   622→
   623→type Step = "selection" | "payment" | "taking" | "results";
   624→
   625→export default function QuizPage() {
   626→  const [step, setStep] = useState<Step>("selection");
   627→  const [selectedQuiz, setSelectedQuiz] = useState<QuizListItem | null>(null);
   628→  const [result, setResult] = useState<QuizResult | null>(null);
   629→  const [isSubmitting, setIsSubmitting] = useState(false);
   630→
   631→  const handleSelectQuiz = useCallback((quiz: QuizListItem) => {
   632→    setSelectedQuiz(quiz);
   633→    setResult(null);
   634→    setStep("payment");
   635→  }, []);
   636→
   637→  const handlePaid = useCallback(() => {
   638→    setStep("taking");
   639→  }, []);
   640→
   641→  const handleFinish = useCallback(
   642→    async (answers: number[]) => {
   643→      if (!selectedQuiz) return;
   644→
   645→      setIsSubmitting(true);
   646→      try {
   647→        const res = await fetch("/api/quiz", {
   648→          method: "POST",
   649→          headers: { "Content-Type": "application/json" },
   650→          body: JSON.stringify({
   651→            quizId: selectedQuiz.id,
   652→            answers,
   653→          }),
   654→        });
   655→
   656→        const data = await res.json();
   657→
   658→        if (res.ok) {
   659→          setResult({
   660→            attemptId: data.attemptId,
   661→            score: data.score,
   662→            total: data.total,
   663→            passed: data.passed,
   664→          });
   665→          setStep("results");
   666→
   667→          if (data.passed) {
   668→            toast.success("Баяр хүргэе!", {
   669→              description: `Та ${data.score}/${data.total} оноо авч тэнцлээ!`,
   670→            });
   671→          } else {
   672→            toast.error("Тэнцээгүй", {
   673→              description: `Та ${data.score}/${data.total} оноо авсан байна. 70% шаардлагатай.`,
   674→            });
   675→          }
   676→        } else {
   677→          toast.error("Алдаа", {
   678→            description: data.error || "Хариултуудыг илгээхэд алдаа гарлаа.",
   679→          });
   680→        }
   681→      } catch {
   682→        toast.error("Холболтын алдаа");
   683→      } finally {
   684→        setIsSubmitting(false);
   685→      }
   686→    },
   687→    [selectedQuiz]
   688→  );
   689→
   690→  const handleRetry = useCallback(() => {
   691→    setResult(null);
   692→    setStep("payment");
   693→  }, []);
   694→
   695→  const handleBackToList = useCallback(() => {
   696→    setSelectedQuiz(null);
   697→    setResult(null);
   698→    setStep("selection");
   699→  }, []);
   700→
   701→  return (
   702→    <section className="w-full">
   703→      {step === "selection" && <QuizSelection onSelect={handleSelectQuiz} />}
   704→
   705→      {step === "payment" && selectedQuiz && (
   706→        <PaymentGate quiz={selectedQuiz} onPaid={handlePaid} onBack={handleBackToList} />
   707→      )}
   708→
   709→      {step === "taking" && selectedQuiz && (
   710→        <QuizTaking
   711→          quiz={selectedQuiz}
   712→          onFinish={handleFinish}
   713→          onBack={handleBackToList}
   714→        />
   715→      )}
   716→
   717→      {step === "results" && selectedQuiz && result && (
   718→        <QuizResults
   719→          quiz={selectedQuiz}
   720→          result={result}
   721→          onRetry={handleRetry}
   722→          onBackToList={handleBackToList}
   723→        />
   724→      )}
   725→
   726→      {/* Submit overlay */}
   727→      <AnimatePresence>
   728→        {isSubmitting && (
   729→          <motion.div
   730→            initial={{ opacity: 0 }}
   731→            animate={{ opacity: 1 }}
   732→            exit={{ opacity: 0 }}
   733→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 backdrop-blur-sm"
   734→          >
   735→            <div className="flex flex-col items-center gap-4 rounded-2xl bg-white p-8 shadow-2xl">
   736→              <Loader2 className="size-10 animate-spin text-brand-600" />
   737→              <p className="text-lg font-medium text-brand-900">
   738→                Хариултуудыг шалгаж байна...
   739→              </p>
   740→            </div>
   741→          </motion.div>
   742→        )}
   743→      </AnimatePresence>
   744→    </section>
   745→  );
   746→}
