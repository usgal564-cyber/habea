   326→/* ─── Enrollment Dialog (Multi-step) ──────────────────── */
   327→
   328→type EnrollStep = "info" | "anket" | "payment" | "confirmed";
   329→
   330→function EnrollmentDialog({
   331→  course,
   332→  onClose,
   333→  onConfirmed,
   334→}: {
   335→  course: Course;
   336→  onClose: () => void;
   337→  onConfirmed: () => void;
   338→}) {
   339→  const { user, token } = useAuthStore();
   340→  const [step, setStep] = useState<EnrollStep>("info");
   341→  const [loading, setLoading] = useState(false);
   342→
   343→  // Anket fields
   344→  const [organization, setOrganization] = useState("");
   345→  const [position, setPosition] = useState("");
   346→  const [experience, setExperience] = useState("");
   347→  const [goal, setGoal] = useState("");
   348→
   349→  const categoryKey = course.category.toLowerCase();
   350→  const price = course.price || COURSE_PRICES[categoryKey] || 30000;
   351→  const schedule = COURSE_SCHEDULES[course.title];
   352→
   353→  // Step 1: Check login & show course info
   354→  const handleStepInfo = () => {
   355→    if (!user || !token) {
   356→      toast.error("Нэвтрэх шаардлагатай", { description: "Сургалтад бүртгүүлэхийн тулд нэвтэрнө үү." });
   357→      return;
   358→    }
   359→    setStep("anket");
   360→  };
   361→
   362→  // Step 2: Fill anket & register
   363→  const handleAnket = async () => {
   364→    setLoading(true);
   365→    try {
   366→      // First register for the course
   367→      const res = await fetch(`/api/courses/${course.id}/register`, {
   368→        method: "POST",
   369→        headers: {
   370→          Authorization: `Bearer ${token}`,
   371→          "Content-Type": "application/json",
   372→        },
   373→      });
   374→      const data = await res.json();
   375→
   376→      if (res.ok) {
   377→        toast.success("Бүртгэл амжилттай!");
   378→        setStep("payment");
   379→      } else {
   380→        toast.error("Бүртгэл амжилтгүй", { description: data.error || "Алдаа гарлаа" });
   381→      }
   382→    } catch {
   383→      toast.error("Холболтын алдаа");
   384→    } finally {
   385→      setLoading(false);
   386→    }
   387→  };
   388→
   389→  // Step 3: Payment
   390→  const handlePayment = async () => {
   391→    setLoading(true);
   392→    try {
   393→      const res = await fetch("/api/courses/payment", {
   394→        method: "POST",
   395→        headers: {
   396→          Authorization: `Bearer ${token}`,
   397→          "Content-Type": "application/json",
   398→        },
   399→        body: JSON.stringify({ courseId: course.id }),
   400→      });
   401→      const data = await res.json();
   402→
   403→      if (res.ok) {
   404→        toast.success("Төлбөр амжилттай!");
   405→        setStep("confirmed");
   406→      } else {
   407→        toast.error("Төлбөр амжилтгүй", { description: data.error });
   408→      }
   409→    } catch {
   410→      toast.error("Холболтын алдаа");
   411→    } finally {
   412→      setLoading(false);
   413→    }
   414→  };
   415→
   416→  const stepLabels = [
   417→    { id: "info", label: "Мэдээлэл", icon: <ClipboardList className="w-4 h-4" /> },
   418→    { id: "anket", label: "Анкет", icon: <FileText className="w-4 h-4" /> },
   419→    { id: "payment", label: "Төлбөр", icon: <CreditCard className="w-4 h-4" /> },
   420→    { id: "confirmed", label: "Баталгаа", icon: <CheckCircle2 className="w-4 h-4" /> },
   421→  ];
   422→
   423→  const currentStepIdx = stepLabels.findIndex((s) => s.id === step);
   424→
   425→  return (
   426→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 backdrop-blur-sm p-4">
   427→      <motion.div
   428→        initial={{ opacity: 0, scale: 0.95 }}
   429→        animate={{ opacity: 1, scale: 1 }}
   430→        className="bg-white rounded-2xl shadow-2xl max-w-lg w-full max-h-[90vh] overflow-y-auto"
   431→      >
   432→        {/* Header */}
   433→        <div className="bg-gradient-to-r from-brand-700 to-brand-900 p-6 rounded-t-2xl">
   434→          <div className="flex items-center justify-between">
   435→            <div>
   436→              <h3 className="text-xl font-bold text-white">{course.title}</h3>
   437→              <p className="text-brand-200 text-sm mt-1">{COURSE_PRICES[categoryKey] ? "Төлбөртэй сургалт" : "Сургалтад бүртгүүлэх"}</p>
   438→            </div>
   439→            <button
   440→              onClick={onClose}
   441→              className="w-8 h-8 rounded-full bg-white/10 hover:bg-white/20 flex items-center justify-center text-white transition-colors"
   442→            >
   443→              ✕
   444→            </button>
   445→          </div>
   446→
   447→          {/* Step indicator */}
   448→          <div className="flex items-center gap-2 mt-4">
   449→            {stepLabels.map((s, i) => (
   450→              <div key={s.id} className="flex items-center gap-2 flex-1">
   451→                <div
   452→                  className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-medium transition-all ${
   453→                    i <= currentStepIdx
   454→                      ? "bg-white text-brand-700"
   455→                      : "bg-white/20 text-white/60"
   456→                  }`}
   457→                >
   458→                  {s.icon}
   459→                </div>
   460→                {i < stepLabels.length - 1 && (
   461→                  <div className={`flex-1 h-0.5 ${i < currentStepIdx ? "bg-white" : "bg-white/20"}`} />
   462→                )}
   463→              </div>
   464→            ))}
   465→          </div>
   466→        </div>
   467→
   468→        {/* Content */}
   469→        <div className="p-6">
   470→          <AnimatePresence mode="wait">
   471→            {/* Step 1: Course Info */}
   472→            {step === "info" && (
   473→              <motion.div key="info" initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }} className="space-y-4">
   474→                <div className="bg-brand-50 rounded-xl p-4 space-y-3">
   475→                  <div className="flex items-center gap-2 text-sm">
   476→                    <Clock className="w-4 h-4 text-brand-600" />
   477→                    <span className="text-muted-foreground">Хугацаа:</span>
   478→                    <span className="font-medium text-brand-900">{course.duration}</span>
   479→                  </div>
   480→                  <div className="flex items-center gap-2 text-sm">
   481→                    <Banknote className="w-4 h-4 text-brand-600" />
   482→                    <span className="text-muted-foreground">Үнэ:</span>
   483→                    <span className="font-bold text-brand-900">{price.toLocaleString()}₮</span>
   484→                  </div>
   485→                  {course.maxStudents && (
   486→                    <div className="flex items-center gap-2 text-sm">
   487→                      <Users className="w-4 h-4 text-brand-600" />
   488→                      <span className="text-muted-foreground">Суудлын тоо:</span>
   489→                      <span className="font-medium text-brand-900">{course.maxStudents}</span>
   490→                    </div>
   491→                  )}
   492→                </div>
   493→
   494→                {schedule && (
   495→                  <div className="bg-amber-50 border border-amber-200 rounded-xl p-4 space-y-3">
   496→                    <h4 className="font-semibold text-amber-900 text-sm flex items-center gap-2">
   497→                      <Calendar className="w-4 h-4" /> Хичээлийн хуваарь
   498→                    </h4>
   499→                    <div className="grid grid-cols-2 gap-2 text-sm">
