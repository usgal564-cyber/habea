   327→type EnrollStep = "info" | "anket" | "payment" | "confirmed";
   328→
   329→function EnrollmentDialog({
   330→  course,
   331→  onClose,
   332→  onConfirmed,
   333→}: {
   334→  course: Course;
   335→  onClose: () => void;
   336→  onConfirmed: () => void;
   337→}) {
   338→  const { user, token } = useAuthStore();
   339→  const [step, setStep] = useState<EnrollStep>("info");
   340→  const [loading, setLoading] = useState(false);
   341→
   342→  // Anket fields
   343→  const [organization, setOrganization] = useState("");
   344→  const [position, setPosition] = useState("");
   345→  const [experience, setExperience] = useState("");
   346→  const [goal, setGoal] = useState("");
   347→
   348→  const categoryKey = course.category.toLowerCase();
   349→  const price = course.price || COURSE_PRICES[categoryKey] || 30000;
   350→  const schedule = COURSE_SCHEDULES[course.title];
   351→
   352→  // Step 1: Check login & show course info
   353→  const handleStepInfo = () => {
   354→    if (!user || !token) {
   355→      toast.error("Нэвтрэх шаардлагатай", { description: "Сургалтад бүртгүүлэхийн тулд нэвтэрнө үү." });
   356→      return;
   357→    }
   358→    setStep("anket");
   359→  };
   360→
   361→  // Step 2: Fill anket & register
   362→  const handleAnket = async () => {
   363→    setLoading(true);
   364→    try {
   365→      // First register for the course
   366→      const res = await fetch(`/api/courses/${course.id}/register`, {
   367→        method: "POST",
   368→        headers: {
   369→          Authorization: `Bearer ${token}`,
   370→          "Content-Type": "application/json",
   371→        },
   372→      });
   373→      const data = await res.json();
   374→
   375→      if (res.ok) {
   376→        toast.success("Бүртгэл амжилттай!");
   377→        setStep("payment");
   378→      } else {
   379→        toast.error("Бүртгэл амжилтгүй", { description: data.error || "Алдаа гарлаа" });
   380→      }
   381→    } catch {
   382→      toast.error("Холболтын алдаа");
   383→    } finally {
   384→      setLoading(false);
   385→    }
   386→  };
   387→
   388→  // Step 3: Payment
   389→  const handlePayment = async () => {
   390→    setLoading(true);
   391→    try {
   392→      const res = await fetch("/api/courses/payment", {
   393→        method: "POST",
   394→        headers: {
   395→          Authorization: `Bearer ${token}`,
   396→          "Content-Type": "application/json",
   397→        },
   398→        body: JSON.stringify({ courseId: course.id }),
   399→      });
   400→      const data = await res.json();
   401→
   402→      if (res.ok) {
   403→        toast.success("Төлбөр амжилттай!");
   404→        setStep("confirmed");
   405→      } else {
   406→        toast.error("Төлбөр амжилтгүй", { description: data.error });
   407→      }
   408→    } catch {
   409→      toast.error("Холболтын алдаа");
   410→    } finally {
   411→      setLoading(false);
   412→    }
   413→  };
   414→
   415→  const stepLabels = [
   416→    { id: "info", label: "Мэдээлэл", icon: <ClipboardList className="w-4 h-4" /> },
   417→    { id: "anket", label: "Анкет", icon: <FileText className="w-4 h-4" /> },
   418→    { id: "payment", label: "Төлбөр", icon: <CreditCard className="w-4 h-4" /> },
   419→    { id: "confirmed", label: "Баталгаа", icon: <CheckCircle2 className="w-4 h-4" /> },
   420→  ];
   421→
   422→  const currentStepIdx = stepLabels.findIndex((s) => s.id === step);
   423→
   424→  return (
   425→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 backdrop-blur-sm p-4">
   426→      <motion.div
   427→        initial={{ opacity: 0, scale: 0.95 }}
   428→        animate={{ opacity: 1, scale: 1 }}
   429→        className="bg-white rounded-2xl shadow-2xl max-w-lg w-full max-h-[90vh] overflow-y-auto"
   430→      >
   431→        {/* Header */}
   432→        <div className="bg-gradient-to-r from-brand-700 to-brand-900 p-6 rounded-t-2xl">
   433→          <div className="flex items-center justify-between">
   434→            <div>
   435→              <h3 className="text-xl font-bold text-white">{course.title}</h3>
   436→              <p className="text-brand-200 text-sm mt-1">{COURSE_PRICES[categoryKey] ? "Төлбөртэй сургалт" : "Сургалтад бүртгүүлэх"}</p>
   437→            </div>
   438→            <button
   439→              onClick={onClose}
   440→              className="w-8 h-8 rounded-full bg-white/10 hover:bg-white/20 flex items-center justify-center text-white transition-colors"
   441→            >
   442→              ✕
   443→            </button>
   444→          </div>
   445→
   446→          {/* Step indicator */}
   447→          <div className="mt-5">
   448→            <div className="flex items-center justify-between relative">
   449→              {/* Background line */}
   450→              <div className="absolute top-[18px] left-[45px] right-[45px] h-[2px] bg-white/20" />
   451→              {/* Active line */}
   452→              <div
   453→                className="absolute top-[18px] left-[45px] h-[2px] bg-white transition-all duration-500"
   454→                style={{ width: currentStepIdx > 0 ? `calc(${currentStepIdx * 100}% / 3 + ${currentStepIdx * 20}px / 3)` : "0%" }}
   455→              />
   456→              {/* Steps */}
   457→              {stepLabels.map((s, i) => (
   458→                <div key={s.id} className="flex flex-col items-center gap-1.5 relative z-10">
   459→                  <div
   460→                    className={`w-9 h-9 rounded-full flex items-center justify-center text-sm font-medium transition-all ${
   461→                      i <= currentStepIdx
   462→                        ? "bg-white text-brand-700 shadow-sm"
   463→                        : "bg-white/20 text-white/60"
   464→                    }`}
   465→                  >
   466→                    {s.icon}
   467→                  </div>
   468→                  <span className={`text-[10px] font-medium transition-colors ${
   469→                    i <= currentStepIdx ? "text-white/90" : "text-white/40"
   470→                  }`}>
   471→                    {s.label}
   472→                  </span>
   473→                </div>
   474→              ))}
   475→            </div>
   476→          </div>
   477→        </div>
   478→
   479→        {/* Content */}
   480→        <div className="p-6">
   481→          <AnimatePresence mode="wait">
   482→            {/* Step 1: Course Info */}
   483→            {step === "info" && (
   484→              <motion.div key="info" initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }} className="space-y-4">
   485→                <div className="bg-brand-50 rounded-xl p-4 space-y-3">
   486→                  <div className="flex items-center gap-2 text-sm">
   487→                    <Clock className="w-4 h-4 text-brand-600" />
   488→                    <span className="text-muted-foreground">Хугацаа:</span>
   489→                    <span className="font-medium text-brand-900">{course.duration}</span>
   490→                  </div>
   491→                  <div className="flex items-center gap-2 text-sm">
   492→                    <Banknote className="w-4 h-4 text-brand-600" />
   493→                    <span className="text-muted-foreground">Үнэ:</span>
   494→                    <span className="font-bold text-brand-900">{price.toLocaleString()}₮</span>
   495→                  </div>
   496→                  {course.maxStudents && (
   497→                    <div className="flex items-center gap-2 text-sm">
   498→                      <Users className="w-4 h-4 text-brand-600" />
