   246→   246→        method: "POST",

   338→   338→/* ─── Step 3: Quiz Taking ───────────────────────────────── */
   339→   339→
   340→   340→function QuizTaking({
   341→   341→  quiz,
   342→   342→  onFinish,
   343→   343→  onBack,
   344→   344→}: {
   345→   345→  quiz: QuizListItem;
   346→   346→  onFinish: (answers: number[]) => void;
   347→   347→  onBack: () => void;
   348→   348→}) {
   349→   349→  const [currentPage, setCurrentPage] = useState(1);
   350→   350→  const [totalPages, setTotalPages] = useState(1);
   351→   351→  const [questions, setQuestions] = useState<Question[]>([]);
   352→   352→  const [answers, setAnswers] = useState<Record<string, number>>({});
   353→   353→  const [isLoading, setIsLoading] = useState(true);
   354→   354→  const [isSubmitting, setIsSubmitting] = useState(false);
   355→   355→
   356→   356→  const fetchQuestions = useCallback(
   357→   357→    async (page: number) => {
   358→   358→      setIsLoading(true);
   359→   359→      try {
   360→   360→        const res = await fetch(
   361→   361→          `/api/quiz?slug=${quiz.slug}&page=${page}&limit=${QUESTIONS_PER_PAGE}`
   362→   362→        );
   363→   363→        const data = await res.json();
   364→   364→        if (data.quiz && data.questions) {
   365→   365→          setQuestions(data.questions);
   366→   366→          setTotalPages(data.pagination.totalPages);
   367→   367→        } else {
   368→   368→          toast.error("Алдаа", {
   369→   369→            description: "Асуултуудыг ачааллахад алдаа гарлаа.",
   370→   370→          });
   371→   371→        }
   372→   372→      } catch {
   373→   373→        toast.error("Холболтын алдаа");
   374→   374→      } finally {
   375→   375→        setIsLoading(false);
   376→   376→      }
   377→   377→    },
   378→   378→    [quiz.slug]
   379→   379→  );
   380→   380→
   381→   381→  useEffect(() => {
   382→   382→    fetchQuestions(1);
   383→   383→  }, [fetchQuestions]);
   384→   384→
   385→   385→  const handleSelectAnswer = (questionId: string, optionIndex: number) => {
   386→   386→    setAnswers((prev) => ({ ...prev, [questionId]: optionIndex }));
   387→   387→  };
   388→   388→
   389→   389→  const handlePageChange = (page: number) => {
   390→   390→    setCurrentPage(page);
   391→   391→    fetchQuestions(page);
   392→   392→    window.scrollTo({ top: 0, behavior: "smooth" });
   393→   393→  };
   394→   394→
   395→   395→  const handleFinish = async () => {
   396→   396→    setIsSubmitting(true);
   397→   397→    try {
   398→   398→      const allAnswers: number[] = [];
   399→   399→      for (let p = 1; p <= totalPages; p++) {
   400→   400→        const res = await fetch(
   401→   401→          `/api/quiz?slug=${quiz.slug}&page=${p}&limit=${QUESTIONS_PER_PAGE}`
   402→   402→        );
   403→   403→        const data = await res.json();
   404→   404→        if (data.questions) {
   405→   405→          for (const q of data.questions) {
   406→   406→            allAnswers.push(answers[q.id] ?? -1);
   407→   407→          }
   408→   408→        }
   409→   409→      }
   410→   410→      onFinish(allAnswers);
   411→   411→    } catch {
   412→   412→      toast.error("Алдаа", {
   413→   413→        description: "Хариултуудыг илгээхэд алдаа гарлаа.",
   414→   414→      });
   415→   415→    } finally {
   416→   416→      setIsSubmitting(false);
   417→   417→    }
   418→   418→  };
   419→   419→
   420→   420→  const answeredCount = Object.keys(answers).length;
   421→   421→
   422→   422→  return (
   423→   423→    <motion.div initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }}>
   424→   424→      <div className="mb-6 flex items-center justify-between">
   425→   425→        <button
   426→   426→          onClick={onBack}
   427→   427→          className="flex items-center gap-2 text-sm text-muted-foreground hover:text-brand-700 transition-colors"
   428→   428→        >
   429→   429→          <ArrowLeft className="size-4" />
   430→   430→          Буцах
   431→   431→        </button>
   432→   432→        <h2 className="text-lg font-bold text-brand-900">{quiz.title}</h2>
   433→   433→      </div>
   434→   434→
   435→   435→      <Card className="border-brand-100">
   436→   436→        <CardHeader className="pb-3">
   437→   437→          <div className="flex items-center justify-between text-sm">
   438→   438→            <span className="text-muted-foreground">
   439→   439→              Хуудас {currentPage} / {totalPages}
   440→   440→            </span>
   441→   441→            <span className="text-muted-foreground">
   442→   442→              Хариулсан: {answeredCount}
   443→   443→            </span>
   444→   444→          </div>
   445→   445→          <Progress value={(answeredCount / (quiz.questionCount || 20)) * 100} className="h-2" />
   446→   446→        </CardHeader>
   447→   447→
   448→   448→        <CardContent>
   449→   449→          {isLoading ? (
   450→   450→            <div className="py-16 flex justify-center">
   451→   451→              <Loader2 className="size-8 animate-spin text-brand-500" />
   452→   452→            </div>
   453→   453→          ) : (
   454→   454→            <div className="space-y-6">
   455→   455→              {questions.map((q, idx) => {
   456→   456→                const globalIdx = (currentPage - 1) * QUESTIONS_PER_PAGE + idx;
   457→   457→                return (
   458→   458→                  <motion.div

   516→   516→                </div>

   540→   540→            </div>

   542→   542→        </CardContent>

   621→   621→/* ─── Main Component ──────────────────────────────────────── */
   622→   622→
   623→   623→type Step = "selection" | "payment" | "taking" | "results";
   624→   624→
   625→   625→export default function QuizPage() {
   626→   626→  const [step, setStep] = useState<Step>("selection");
   627→   627→  const [selectedQuiz, setSelectedQuiz] = useState<QuizListItem | null>(null);
   628→   628→  const [result, setResult] = useState<QuizResult | null>(null);
   629→   629→  const [isSubmitting, setIsSubmitting] = useState(false);
   630→   630→
   631→   631→  const handleSelectQuiz = useCallback((quiz: QuizListItem) => {
   632→   632→    setSelectedQuiz(quiz);
   633→   633→    setResult(null);
   634→   634→    setStep("payment");
   635→   635→  }, []);
   636→   636→
   637→   637→  const handlePaid = useCallback(() => {
   638→   638→    setStep("taking");
   639→   639→  }, []);
   640→   640→
   641→   641→  const handleFinish = useCallback(
   642→   642→    async (answers: number[]) => {
   643→   643→      if (!selectedQuiz) return;
   644→   644→
   645→   645→      setIsSubmitting(true);
   646→   646→      try {
   647→   647→        const res = await fetch("/api/quiz", {
   648→   648→          method: "POST",
   649→   649→          headers: { "Content-Type": "application/json" },
   650→   650→          body: JSON.stringify({
   651→   651→            quizId: selectedQuiz.id,
   652→   652→            answers,
   653→   653→          }),
   654→   654→        });
   655→   655→
   656→   656→        const data = await res.json();
   657→   657→
   658→   658→        if (res.ok) {
   659→   659→          setResult({
   660→   660→            attemptId: data.attemptId,
   661→   661→            score: data.score,
   662→   662→            total: data.total,
   663→   663→            passed: data.passed,
   664→   664→          });
   665→   665→          setStep("results");
   666→   666→
   667→   667→          if (data.passed) {
   668→   668→            toast.success("Баяр хүргэе!", {
   669→   669→              description: `Та ${data.score}/${data.total} оноо авч тэнцлээ!`,
   670→   670→            });
   671→   671→          } else {
   672→   672→            toast.error("Тэнцээгүй", {
   673→   673→              description: `Та ${data.score}/${data.total} оноо авсан байна. 70% шаардлагатай.`,
   674→   674→            });
   675→   675→          }
   676→   676→        } else {
   677→   677→          toast.error("Алдаа", {
   678→   678→            description: data.error || "Хариултуудыг илгээхэд алдаа гарлаа.",
   679→   679→          });
   680→   680→        }
   681→   681→      } catch {
   682→   682→        toast.error("Холболтын алдаа");
   683→   683→      } finally {
   684→   684→        setIsSubmitting(false);
   685→   685→      }
   686→   686→    },
   687→   687→    [selectedQuiz]
   688→   688→  );
   689→   689→
   690→   690→  const handleRetry = useCallback(() => {
   691→   691→    setResult(null);
   692→   692→    setStep("payment");
   693→   693→  }, []);
   694→   694→
   695→   695→  const handleBackToList = useCallback(() => {
   696→   696→    setSelectedQuiz(null);
   697→   697→    setResult(null);
   698→   698→    setStep("selection");
   699→   699→  }, []);
   700→   700→
   701→   701→  return (
   702→   702→    <section className="w-full">
   703→   703→      {step === "selection" && <QuizSelection onSelect={handleSelectQuiz} />}
   704→   704→
   705→   705→      {step === "payment" && selectedQuiz && (
   706→   706→        <PaymentGate quiz={selectedQuiz} onPaid={handlePaid} onBack={handleBackToList} />
   707→   707→      )}
   708→   708→
   709→   709→      {step === "taking" && selectedQuiz && (
   710→   710→        <QuizTaking
   711→   711→          quiz={selectedQuiz}
   712→   712→          onFinish={handleFinish}
   713→   713→          onBack={handleBackToList}
   714→   714→        />
   715→   715→      )}
   716→   716→
   717→   717→      {step === "results" && selectedQuiz && result && (
   718→   718→        <QuizResults
   719→   719→          quiz={selectedQuiz}
   720→   720→          result={result}
   721→   721→          onRetry={handleRetry}
   722→   722→          onBackToList={handleBackToList}
   723→   723→        />
   724→   724→      )}
   725→   725→
   726→   726→      {/* Submit overlay */}
   727→   727→      <AnimatePresence>
   728→   728→        {isSubmitting && (
   729→   729→          <motion.div
   730→   730→            initial={{ opacity: 0 }}
   731→   731→            animate={{ opacity: 1 }}
   732→   732→            exit={{ opacity: 0 }}
   733→   733→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 backdrop-blur-sm"
   734→   734→          >
   735→   735→            <div className="flex flex-col items-center gap-4 rounded-2xl bg-white p-8 shadow-2xl">
   736→   736→              <Loader2 className="size-10 animate-spin text-brand-600" />
   737→   737→              <p className="text-lg font-medium text-brand-900">
   738→   738→                Хариултуудыг шалгаж байна...
   739→   739→              </p>
   740→   740→            </div>
   741→   741→          </motion.div>
   742→   742→        )}
   743→   743→      </AnimatePresence>
   744→   744→    </section>
   745→   745→  );
   746→   746→}
