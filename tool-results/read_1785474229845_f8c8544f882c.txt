   126→   126→func RegisterHandler(c *gin.Context) {
   127→   127→        var req RegisterRequest
   128→   128→        if err := c.ShouldBindJSON(&req); err != nil {
   129→   129→                c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
   130→   130→                return
   131→   131→        }
   132→   132→
   133→   133→        var existing User
   134→   134→        if err := DB.Where("email = ?", req.Email).First(&existing).Error; err == nil {
   135→   135→                c.JSON(http.StatusConflict, gin.H{"error": "Email already registered"})
   136→   136→                return
   137→   137→        }
   138→   138→
   139→   139→        hashedPassword, err := bcrypt.GenerateFromPassword([]byte(req.Password), bcrypt.DefaultCost)
   140→   140→        if err != nil {
   141→   141→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to hash password"})
   142→   142→                return
   143→   143→        }
   144→   144→
   145→   145→        user := User{
   146→   146→                ID:             uuid.New().String(),
   147→   147→                FirstName:      req.FirstName,
   148→   148→                LastName:       req.LastName,
   149→   149→                Email:          req.Email,
   150→   150→                Phone:          req.Phone,
   151→   151→                Password:       string(hashedPassword),
   152→   152→                Address:        req.Address,
   153→   153→                SecondaryPhone: req.SecondaryPhone,
   154→   154→                Role:           "USER",
   155→   155→        }
   156→   156→
   157→   157→        if err := DB.Create(&user).Error; err != nil {
   158→   158→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to create user"})
   159→   159→                return
   160→   160→        }
   161→   161→
   162→   162→        token, err := GenerateToken(user.ID, user.Email, user.Role)
   163→   163→        if err != nil {
   164→   164→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to generate token"})
   165→   165→                return
   166→   166→        }
   167→   167→
   168→   168→        c.JSON(http.StatusCreated, gin.H{
   169→   169→                "user": gin.H{
   170→   170→                        "userId": user.ID,
   171→   171→                        "email":  user.Email,
   172→   172→                        "role":   user.Role,
   173→   173→                },
   174→   174→                "token": token,
   175→   175→        })
   176→   176→}

   294→   294→func QuizSubmitHandler(c *gin.Context) {
   295→   295→        var req QuizSubmitRequest
   296→   296→        if err := c.ShouldBindJSON(&req); err != nil {
   297→   297→                c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
   298→   298→                return
   299→   299→        }
   300→   300→
   301→   301→        var questions []QuizQuestion
   302→   302→        if err := DB.Where("quiz_id = ?", req.QuizID).Order("index ASC").Find(&questions).Error; err != nil {
   303→   303→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to fetch questions"})
   304→   304→                return
   305→   305→        }
   306→   306→
   307→   307→        if len(questions) == 0 {
   308→   308→                c.JSON(http.StatusNotFound, gin.H{"error": "Quiz has no questions"})
   309→   309→                return
   310→   310→        }
   311→   311→
   312→   312→        questionMap := make(map[string]QuizQuestion)
   313→   313→        for _, q := range questions {
   314→   314→                questionMap[q.ID] = q
   315→   315→        }
   316→   316→
   317→   317→        correct := 0
   318→   318→        total := len(questions)
   319→   319→        for _, ans := range req.Answers {
   320→   320→                if q, ok := questionMap[ans.QuestionID]; ok {
   321→   321→                        if ans.SelectedOption == q.Correct {
   322→   322→                                correct++
   323→   323→                        }
   324→   324→                }
   325→   325→        }
   326→   326→
   327→   327→        answersJSON, _ := json.Marshal(req.Answers)
   328→   328→        userID, _ := c.Get("userId")
   329→   329→
   330→   330→        attempt := QuizAttempt{
   331→   331→                ID:      uuid.New().String(),
   332→   332→                UserID:  userID.(string),
   333→   333→                QuizID:  req.QuizID,
   334→   334→                Score:   correct,
   335→   335→                Total:   total,
   336→   336→                Answers: string(answersJSON),
   337→   337→        }
   338→   338→        DB.Create(&attempt)
   339→   339→
   340→   340→        percentage := 0
   341→   341→        if total > 0 {
   342→   342→                percentage = (correct * 100) / total
   343→   343→        }
   344→   344→
   345→   345→        c.JSON(http.StatusOK, gin.H{
   346→   346→                "score":      correct,
   347→   347→                "total":      total,
   348→   348→                "correct":    correct,
   349→   349→                "percentage": percentage,
   350→   350→        })
   351→   351→}

   391→   391→func submitExam(c *gin.Context, req ExamActionRequest) {
   392→   392→        userID, _ := c.Get("userId")
   393→   393→
   394→   394→        var exam Exam
   395→   395→        if err := DB.Where("id = ?", req.ExamID).First(&exam).Error; err != nil {
   396→   396→                c.JSON(http.StatusNotFound, gin.H{"error": "Exam not found"})
   397→   397→                return
   398→   398→        }
   399→   399→
   400→   400→        if !exam.IsActive {
   401→   401→                c.JSON(http.StatusBadRequest, gin.H{"error": "Exam is not active"})
   402→   402→                return
   403→   403→        }
   404→   404→
   405→   405→        var questions []QuizQuestion
   406→   406→        DB.Where("quiz_id = ?", req.ExamID).Order("index ASC").Find(&questions)
   407→   407→
   408→   408→        score := 0
   409→   409→        total := len(questions)
   410→   410→        for i, answer := range req.Answers {
   411→   411→                if i < len(questions) && answer == questions[i].Correct {
   412→   412→                        score++
   413→   413→                }
   414→   414→        }
   415→   415→
   416→   416→        passed := false
   417→   417→        if total > 0 {
   418→   418→                passed = float64(score)/float64(total) >= 0.7
   419→   419→        }
   420→   420→
   421→   421→        answersJSON, _ := json.Marshal(req.Answers)
   422→   422→
   423→   423→        attempt := ExamAttempt{
   424→   424→                ID:      uuid.New().String(),
   425→   425→                ExamID:  req.ExamID,
   426→   426→                UserID:  userID.(string),
   427→   427→                Score:   score,
   428→   428→                Total:   total,
   429→   429→                Passed:  passed,
   430→   430→                Answers: string(answersJSON),
   431→   431→        }
   432→   432→
   433→   433→        if err := DB.Create(&attempt).Error; err != nil {
   434→   434→                c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to save attempt"})
   435→   435→                return
   436→   436→        }
   437→   437→
   438→   438→        c.JSON(http.StatusOK, gin.H{
   439→   439→                "score":  score,
   440→   440→                "total":  total,
   441→   441→                "passed": passed,
   442→   442→        })
   443→   443→}
