   334→/* ─── Enrollment Dialog (Multi-step) ──────────────────── */
   335→
   336→type EnrollStep = "info" | "anket" | "payment" | "confirmed";
   337→
   338→function EnrollmentDialog({
   339→  course,
   340→  onClose,
   341→  onConfirmed,
   342→}: {
   343→  course: Course;
   344→  onClose: () => void;
   345→  onConfirmed: () => void;
   346→}) {
   347→  const { user, token } = useAuthStore();
   348→  const [step, setStep] = useState<EnrollStep>("info");
   349→  const [loading, setLoading] = useState(false);
   350→
   351→  // Anket fields
   352→  const [organization, setOrganization] = useState("");
   353→  const [position, setPosition] = useState("");
   354→  const [experience, setExperience] = useState("");
   355→  const [goal, setGoal] = useState("");
   356→
   357→  const categoryKey = course.category.toLowerCase();
   358→  const price = course.price || COURSE_PRICES[categoryKey] || 30000;
   359→  const schedule = COURSE_SCHEDULES[course.title];
   360→
   361→  // Step 1: Check login & show course info
   362→  const handleStepInfo = () => {
   363→    if (!user || !token) {
   364→      toast.error("Нэвтрэх шаардлагатай", { description: "Сургалтад бүртгүүлэхийн тулд нэвтэрнө үү." });
   365→      return;
   366→    }
   367→    setStep("anket");
   368→  };
   369→
   370→  // Step 2: Fill anket & register
   371→  const handleAnket = async () => {
   372→    setLoading(true);
   373→    try {
   374→      // First register for the course
   375→      const res = await fetch(`/api/courses/${course.id}/register`, {
   376→        method: "POST",
   377→        headers: {
   378→          Authorization: `Bearer ${token}`,
   379→          "Content-Type": "application/json",
   380→        },
   381→      });
   382→      const data = await res.json();
   383→
   384→      if (res.ok) {
   385→        toast.success("Бүртгэл амжилттай!");
   386→        setStep("payment");
   387→      } else {
   388→        toast.error("Бүртгэл амжилтгүй", { description: data.error || "Алдаа гарлаа" });
   389→      }
   390→    } catch {
   391→      toast.error("Холболтын алдаа");
   392→    } finally {
   393→      setLoading(false);
   394→    }
   395→  };
   396→
   397→  // Step 3: Payment
   398→  const handlePayment = async () => {
   399→    setLoading(true);
   400→    try {
   401→      const res = await fetch("/api/courses/payment", {
   402→        method: "POST",
   403→        headers: {
   404→          Authorization: `Bearer ${token}`,
   405→          "Content-Type": "application/json",
   406→        },
   407→        body: JSON.stringify({ courseId: course.id }),
   408→      });
   409→      const data = await res.json();
   410→
   411→      if (res.ok) {
   412→        toast.success("Төлбөр амжилттай!");
   413→        setStep("confirmed");
   414→      } else {
   415→        toast.error("Төлбөр амжилтгүй", { description: data.error });
   416→      }
   417→    } catch {
   418→      toast.error("Холболтын алдаа");
   419→    } finally {
   420→      setLoading(false);
   421→    }
   422→  };
   423→
   424→  const stepLabels = [
   425→    { id: "info", label: "Мэдээлэл", icon: <ClipboardList className="w-4 h-4" /> },
   426→    { id: "anket", label: "Анкет", icon: <FileText className="w-4 h-4" /> },
   427→    { id: "payment", label: "Төлбөр", icon: <CreditCard className="w-4 h-4" /> },
   428→    { id: "confirmed", label: "Баталгаа", icon: <CheckCircle2 className="w-4 h-4" /> },
   429→  ];
   430→
   431→  const currentStepIdx = stepLabels.findIndex((s) => s.id === step);
   432→
   433→  return (
   434→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 backdrop-blur-sm p-4">
   435→      <motion.div
   436→        initial={{ opacity: 0, scale: 0.95 }}
   437→        animate={{ opacity: 1, scale: 1 }}
   438→        className="bg-white rounded-2xl shadow-2xl max-w-lg w-full max-h-[90vh] overflow-y-auto"
   439→      >
   440→        {/* Header */}
   441→        <div className="bg-gradient-to-r from-brand-700 to-brand-900 p-6 rounded-t-2xl">
   442→          <div className="flex items-center justify-between">
   443→            <div>
   444→              <h3 className="text-xl font-bold text-white">{course.title}</h3>
   445→              <p className="text-brand-200 text-sm mt-1">{COURSE_PRICES[categoryKey] ? "Төлбөртэй сургалт" : "Сургалтад бүртгүүлэх"}</p>
   446→            </div>
   447→            <button
   448→              onClick={onClose}
   449→              className="w-8 h-8 rounded-full bg-white/10 hover:bg-white/20 flex items-center justify-center text-white transition-colors"
   450→            >
   451→              ✕
   452→            </button>
   453→          </div>
   454→
   455→          {/* Step indicator */}
   456→          <div className="mt-5">
   457→            <div className="flex items-center justify-between relative">
   458→              {/* Background line */}
   459→              <div className="absolute top-[18px] left-[45px] right-[45px] h-[2px] bg-white/20" />
   460→              {/* Active line */}
   461→              <div
   462→                className="absolute top-[18px] left-[45px] h-[2px] bg-white transition-all duration-500"
   463→                style={{ width: currentStepIdx > 0 ? `calc(${currentStepIdx * 100}% / 3 + ${currentStepIdx * 20}px / 3)` : "0%" }}
   464→              />
   465→              {/* Steps */}
   466→              {stepLabels.map((s, i) => (
   467→                <div key={s.id} className="flex flex-col items-center gap-1.5 relative z-10">
   468→                  <div
   469→                    className={`w-9 h-9 rounded-full flex items-center justify-center text-sm font-medium transition-all ${
   470→                      i <= currentStepIdx
   471→                        ? "bg-white text-brand-700 shadow-sm"
   472→                        : "bg-white/20 text-white/60"
   473→                    }`}
   474→                  >
   475→                    {s.icon}
   476→                  </div>
   477→                  <span className={`text-[10px] font-medium transition-colors ${
   478→                    i <= currentStepIdx ? "text-white/90" : "text-white/40"
   479→                  }`}>
   480→                    {s.label}
   481→                  </span>
   482→                </div>
   483→              ))}
   484→            </div>
   485→          </div>
   486→        </div>
   487→
   488→        {/* Content */}
   489→        <div className="p-6">
   490→          <AnimatePresence mode="wait">
   491→            {/* Step 1: Course Info */}
   492→            {step === "info" && (
   493→              <motion.div key="info" initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }} className="space-y-4">
   494→                <div className="bg-brand-50 rounded-xl p-4 space-y-3">
   495→                  <div className="flex items-center gap-2 text-sm">
   496→                    <Clock className="w-4 h-4 text-brand-600" />
   497→                    <span className="text-muted-foreground">Хугацаа:</span>
   498→                    <span className="font-medium text-brand-900">{course.duration}</span>
   499→                  </div>
   500→                  <div className="flex items-center gap-2 text-sm">
   501→                    <Banknote className="w-4 h-4 text-brand-600" />
   502→                    <span className="text-muted-foreground">Үнэ:</span>
