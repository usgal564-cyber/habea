   219→   219→/* ─── Step 2: Quiz Taking ───────────────────────────────── */
   220→   220→
   221→   221→function QuizTaking({
   222→   222→  quiz,
   223→   223→  onFinish,
   224→   224→  onBack,
   225→   225→}: {
   226→   226→  quiz: QuizListItem;
   227→   227→  onFinish: (answers: number[]) => void;
   228→   228→  onBack: () => void;
   229→   229→}) {
   230→   230→  const [currentPage, setCurrentPage] = useState(1);
   231→   231→  const [totalPages, setTotalPages] = useState(1);
   232→   232→  const [questions, setQuestions] = useState<Question[]>([]);
   233→   233→  const [answers, setAnswers] = useState<Record<string, number>>({});
   234→   234→  const [isLoading, setIsLoading] = useState(true);
   235→   235→  const [isSubmitting, setIsSubmitting] = useState(false);
   236→   236→
   237→   237→  const fetchQuestions = useCallback(
   238→   238→    async (page: number) => {
   239→   239→      setIsLoading(true);
   240→   240→      try {
   241→   241→        const res = await fetch(
   242→   242→          `/api/quiz?slug=${quiz.slug}&page=${page}&limit=${QUESTIONS_PER_PAGE}`
   243→   243→        );
   244→   244→        const data = await res.json();
   245→   245→        if (data.quiz && data.questions) {
   246→   246→          setQuestions(data.questions);
   247→   247→          setTotalPages(data.pagination.totalPages);
   248→   248→        } else {
   249→   249→          toast.error("Алдаа", {
   250→   250→            description: "Асуултуудыг ачааллахад алдаа гарлаа.",
   251→   251→          });
   252→   252→        }
   253→   253→      } catch {
   254→   254→        toast.error("Холболтын алдаа");
   255→   255→      } finally {
   256→   256→        setIsLoading(false);
   257→   257→      }
   258→   258→    },
   259→   259→    [quiz.slug]
   260→   260→  );
   261→   261→
   262→   262→  // Fetch first page
   263→   263→  useEffect(() => {
   264→   264→    fetchQuestions(1);
   265→   265→  }, [fetchQuestions]);
   266→   266→
   267→   267→  const handleSelectAnswer = (questionId: string, optionIndex: number) => {
   268→   268→    setAnswers((prev) => ({ ...prev, [questionId]: optionIndex }));
   269→   269→  };
   270→   270→
   271→   271→  const handlePageChange = (page: number) => {
   272→   272→    setCurrentPage(page);
   273→   273→    fetchQuestions(page);
   274→   274→    window.scrollTo({ top: 0, behavior: "smooth" });
   275→   275→  };
   276→   276→
   277→   277→  const handleFinish = async () => {
   278→   278→    setIsSubmitting(true);
   279→   279→    try {
   280→   280→      const allAnswers: number[] = [];
   281→   281→      // Fetch all pages to build ordered answer array
   282→   282→      for (let p = 1; p <= totalPages; p++) {
   283→   283→        const res = await fetch(
   284→   284→          `/api/quiz?slug=${quiz.slug}&page=${p}&limit=${QUESTIONS_PER_PAGE}`
   285→   285→        );
   286→   286→        const data = await res.json();
   287→   287→        if (data.questions) {
   288→   288→          for (const q of data.questions) {
   289→   289→            allAnswers.push(answers[q.id] ?? -1);
   290→   290→          }
   291→   291→        }
   292→   292→      }
   293→   293→      onFinish(allAnswers);
   294→   294→    } catch {
   295→   295→      toast.error("Алдаа", {
   296→   296→        description: "Хариултуудыг илгээхэд алдаа гарлаа.",
   297→   297→      });
   298→   298→    } finally {
   299→   299→      setIsSubmitting(false);
   300→   300→    }
   301→   301→  };
   302→   302→
   303→   303→  const answeredOnPage = questions.filter(
   304→   304→    (q) => answers[q.id] !== undefined
   305→   305→  ).length;
   306→   306→  const progressPercent =
   307→   307→    totalPages > 0
   308→   308→      ? ((currentPage - 1) / totalPages) * 100 +
   309→   309→        (answeredOnPage / (questions.length || 1) / totalPages) * 100
   310→   310→      : 0;
   311→   311→
   312→   312→  return (
   313→   313→    <motion.div
   314→   314→      initial={{ opacity: 0, y: 20 }}
   315→   315→      animate={{ opacity: 1, y: 0 }}
   316→   316→      transition={{ duration: 0.4 }}
   317→   317→    >
   318→   318→      {/* Top Bar */}
   319→   319→      <div className="mb-6 flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   320→   320→        <div className="flex items-center gap-3">
   321→   321→          <Button
   322→   322→            variant="ghost"
   323→   323→            size="icon"
   324→   324→            className="text-brand-600 hover:bg-brand-50 hover:text-brand-800"
   325→   325→            onClick={onBack}
   326→   326→          >
   327→   327→            <ArrowLeft className="size-5" />
   328→   328→          </Button>
   329→   329→          <div>
   330→   330→            <h2 className="text-xl font-bold text-brand-900 sm:text-2xl">
   331→   331→              {quiz.title}
   332→   332→            </h2>
   333→   333→            <p className="text-sm text-muted-foreground">{quiz.description}</p>
   334→   334→          </div>
   335→   335→        </div>
   336→   336→        <Badge className="w-fit bg-brand-50 text-brand-700 hover:bg-brand-100">
   337→   337→          <ListChecks className="mr-1 size-3.5" />
   338→   338→          Хуудас {currentPage} / {totalPages}
   339→   339→        </Badge>
   340→   340→      </div>
   341→   341→
   342→   342→      {/* Progress Bar */}
   343→   343→      <div className="mb-6 space-y-2">
   344→   344→        <div className="flex items-center justify-between text-sm">
   345→   345→          <span className="text-muted-foreground">
   346→   346→            Хуудасны дэвсгэр: {currentPage} / {totalPages}
   347→   347→          </span>
   348→   348→          <span className="font-medium text-brand-700">
   349→   349→            Энэ хуудас: {answeredOnPage}/{questions.length} хариулсан
   350→   350→          </span>
   351→   351→        </div>
   352→   352→        <Progress
   353→   353→          value={progressPercent}
   354→   354→          className="h-2.5 bg-brand-100"
   355→   355→        />
   356→   356→      </div>
   357→   357→
   358→   358→      {/* Questions */}
   359→   359→      {isLoading ? (
   360→   360→        <div className="space-y-6">
   361→   361→          {Array.from({ length: 5 }).map((_, i) => (
   362→   362→            <Card key={i} className="p-6">
   363→   363→              <Skeleton className="mb-4 h-5 w-3/4" />
   364→   364→              <div className="space-y-3">
   365→   365→                {Array.from({ length: 4 }).map((_, j) => (
   366→   366→                  <Skeleton key={j} className="h-10 w-full" />
   367→   367→                ))}
   368→   368→              </div>
   369→   369→            </Card>
   370→   370→          ))}
   371→   371→        </div>
   372→   372→      ) : (
   373→   373→        <AnimatePresence mode="wait">
   374→   374→          <motion.div
   375→   375→            key={currentPage}
   376→   376→            variants={pageTransition}
   377→   377→            initial="initial"
   378→   378→            animate="animate"
   379→   379→            exit="exit"
   380→   380→            className="space-y-4"
   381→   381→          >
   382→   382→            {questions.map((question, qIdx) => {
   383→   383→              const globalIndex =
   384→   384→                (currentPage - 1) * QUESTIONS_PER_PAGE + qIdx + 1;
   385→   385→              const selectedAnswer = answers[question.id];
   386→   386→
   387→   387→              return (
   388→   388→                <motion.div
   389→   389→                  key={question.id}
   390→   390→                  variants={itemVariants}
   391→   391→                  initial="hidden"
   392→   392→                  animate="visible"
   393→   393→                >
   394→   394→                  <Card className="overflow-hidden border-brand-100 bg-white">
   395→   395→                    <CardContent className="p-5 sm:p-6">
   396→   396→                      {/* Question Number & Text */}
   397→   397→                      <div className="mb-4 flex items-start gap-3">
   398→   398→                        <span className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-brand-600 text-xs font-bold text-white">
   399→   399→                          {globalIndex}
   400→   400→                        </span>
   401→   401→                        <p className="text-base font-medium leading-relaxed text-foreground sm:text-lg">
   402→   402→                          {question.questionText}
   403→   403→                        </p>
   404→   404→                      </div>
   405→   405→
   406→   406→                      {/* Options */}
   407→   407→                      <div className="ml-10 space-y-2.5">
   408→   408→                        {OPTION_KEYS.map((key, optIdx) => {
   409→   409→                          const isSelected = selectedAnswer === optIdx;
   410→   410→                          return (
   411→   411→                            <button
   412→   412→                              key={key}
   413→   413→                              type="button"
   414→   414→                              onClick={() =>
   415→   415→                                handleSelectAnswer(question.id, optIdx)
   416→   416→                              }
   417→   417→                              className={[
   418→   418→                                "flex w-full items-start gap-3 rounded-lg border-2 px-4 py-3 text-left text-sm transition-all duration-200 sm:text-base",
   419→   419→                                isSelected
   420→   420→                                  ? "border-brand-500 bg-brand-50 text-brand-900 shadow-sm ring-1 ring-brand-200"
   421→   421→                                  : "border-transparent bg-muted/50 text-foreground hover:border-brand-200 hover:bg-brand-50/50",
   422→   422→                              ].join(" ")}
   423→   423→                            >
   424→   424→                              <span
   425→   425→                                className={[
   426→   426→                                  "flex h-6 w-6 shrink-0 items-center justify-center rounded-full border-2 text-xs font-bold transition-colors",
   427→   427→                                  isSelected
   428→   428→                                    ? "border-brand-600 bg-brand-600 text-white"
   429→   429→                                    : "border-muted-foreground/30 text-muted-foreground",
   430→   430→                                ].join(" ")}
   431→   431→                              >
   432→   432→                                {OPTION_LABELS[optIdx]}
   433→   433→                              </span>
   434→   434→                              <span className="pt-0.5 leading-relaxed">
   435→   435→                                {question[key]}
   436→   436→                              </span>
   437→   437→                            </button>
   438→   438→                          );
   439→   439→                        })}
   440→   440→                      </div>
   441→   441→                    </CardContent>
   442→   442→                  </Card>
   443→   443→                </motion.div>
   444→   444→              );
   445→   445→            })}
   446→   446→          </motion.div>
   447→   447→        </AnimatePresence>
   448→   448→      )}
   449→   449→
   450→   450→      {/* Page Navigation */}
   451→   451→      <div className="mt-8 flex flex-col items-center gap-4 sm:flex-row sm:justify-between">
   452→   452→        {/* Previous Button */}
   453→   453→        <Button
   454→   454→          variant="outline"
   455→   455→          className="border-brand-200 text-brand-700 hover:bg-brand-50 hover:text-brand-800"
   456→   456→          onClick={() => handlePageChange(currentPage - 1)}
   457→   457→          disabled={currentPage <= 1 || isLoading}
   458→   458→        >
   459→   459→          <ChevronLeft className="size-4" />
   460→   460→          Өмнөх
   461→   461→        </Button>
   462→   462→
   463→   463→        {/* Page Numbers */}
   464→   464→        <div className="flex items-center gap-1.5">
   465→   465→          {Array.from({ length: totalPages }, (_, i) => i + 1).map((page) => (
   466→   466→            <button
   467→   467→              key={page}
   468→   468→              type="button"
   469→   469→              onClick={() => handlePageChange(page)}
   470→   470→              disabled={isLoading}
   471→   471→              className={[
   472→   472→                "flex h-9 w-9 items-center justify-center rounded-md text-sm font-medium transition-all disabled:opacity-50",
   473→   473→                page === currentPage

   713→   713→
   714→   714→type Step = "selection" | "taking" | "results";
   715→   715→
   716→   716→export default function QuizPage() {
   717→   717→  const [step, setStep] = useState<Step>("selection");
   718→   718→  const [selectedQuiz, setSelectedQuiz] = useState<QuizListItem | null>(null);
   719→   719→  const [result, setResult] = useState<QuizResult | null>(null);
   720→   720→  const [isSubmitting, setIsSubmitting] = useState(false);
   721→   721→
   722→   722→  const handleSelectQuiz = useCallback((quiz: QuizListItem) => {
   723→   723→    setSelectedQuiz(quiz);
   724→   724→    setResult(null);
   725→   725→    setStep("taking");
   726→   726→  }, []);
   727→   727→
   728→   728→  const handleFinish = useCallback(
   729→   729→    async (answers: number[]) => {
   730→   730→      if (!selectedQuiz) return;
   731→   731→
   732→   732→      setIsSubmitting(true);
   733→   733→      try {
   734→   734→        const res = await fetch("/api/quiz", {
   735→   735→          method: "POST",
   736→   736→          headers: { "Content-Type": "application/json" },
   737→   737→          body: JSON.stringify({
   738→   738→            quizId: selectedQuiz.id,
   739→   739→            answers,
   740→   740→          }),
   741→   741→        });
   742→   742→
   743→   743→        const data = await res.json();
   744→   744→
   745→   745→        if (res.ok) {
   746→   746→          setResult({
   747→   747→            attemptId: data.attemptId,
   748→   748→            score: data.score,
   749→   749→            total: data.total,
   750→   750→            passed: data.passed,
   751→   751→          });
   752→   752→          setStep("results");
   753→   753→
   754→   754→          if (data.passed) {
   755→   755→            toast.success("Баяр хүргэе!", {
   756→   756→              description: `Та ${data.score}/${data.total} оноо авч тэнцлээ!`,
   757→   757→            });
   758→   758→          } else {
   759→   759→            toast.error("Тэнцээгүй", {
   760→   760→              description: `Та ${data.score}/${data.total} оноо авсан байна. 70% шаардлагатай.`,
   761→   761→            });
   762→   762→          }
   763→   763→        } else {
   764→   764→          toast.error("Алдаа", {
   765→   765→            description: data.error || "Хариултуудыг илгээхэд алдаа гарлаа.",
   766→   766→          });
   767→   767→        }
   768→   768→      } catch {
   769→   769→        toast.error("Холболтын алдаа");
   770→   770→      } finally {
   771→   771→        setIsSubmitting(false);
   772→   772→      }
   773→   773→    },
   774→   774→    [selectedQuiz]
   775→   775→  );
   776→   776→
   777→   777→  const handleRetry = useCallback(() => {
   778→   778→    setResult(null);
   779→   779→    setStep("taking");
   780→   780→  }, []);
   781→   781→
   782→   782→  const handleBackToList = useCallback(() => {
   783→   783→    setSelectedQuiz(null);
   784→   784→    setResult(null);
   785→   785→    setStep("selection");
   786→   786→  }, []);
   787→   787→
   788→   788→  return (
   789→   789→    <section className="w-full">
   790→   790→      {step === "selection" && <QuizSelection onSelect={handleSelectQuiz} />}
   791→   791→
   792→   792→      {step === "taking" && selectedQuiz && (
   793→   793→        <QuizTaking
   794→   794→          quiz={selectedQuiz}
   795→   795→          onFinish={handleFinish}
   796→   796→          onBack={handleBackToList}
   797→   797→        />
   798→   798→      )}
   799→   799→
   800→   800→      {step === "results" && selectedQuiz && result && (
   801→   801→        <QuizResults
   802→   802→          quiz={selectedQuiz}
   803→   803→          result={result}
   804→   804→          onRetry={handleRetry}
   805→   805→          onBackToList={handleBackToList}
   806→   806→        />
   807→   807→      )}
   808→   808→
   809→   809→      {/* Submit overlay */}
   810→   810→      <AnimatePresence>
   811→   811→        {isSubmitting && (
   812→   812→          <motion.div
   813→   813→            initial={{ opacity: 0 }}
   814→   814→            animate={{ opacity: 1 }}
   815→   815→            exit={{ opacity: 0 }}
   816→   816→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 backdrop-blur-sm"
   817→   817→          >
   818→   818→            <div className="flex flex-col items-center gap-4 rounded-2xl bg-white p-8 shadow-2xl">
   819→   819→              <Loader2 className="size-10 animate-spin text-brand-600" />
   820→   820→              <p className="text-lg font-medium text-brand-900">
   821→   821→                Хариултуудыг шалгаж байна...
   822→   822→              </p>

   827→   827→    </section>
   828→   828→  );
   829→   829→}
