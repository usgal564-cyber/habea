   334→/* ─── Enrollment Dialog (Multi-step) ──────────────────── */
   335→
   336→type EnrollStep = "register" | "info" | "payment" | "confirmed";
   337→
   338→function EnrollmentDialog({
   339→  course,
   340→  onClose,
   341→  onConfirmed,
   342→}: {
   343→  course: Course;
   344→  onClose: () => void;
   345→  onConfirmed: () => void;
   346→}) {
   347→  const { user, token, setAuth } = useAuthStore();
   348→  const [step, setStep] = useState<EnrollStep>(user ? "info" : "register");
   349→  const [loading, setLoading] = useState(false);
   350→
   351→  // Registration fields (for non-logged-in users)
   352→  const [regLastName, setRegLastName] = useState("");
   353→  const [regFirstName, setRegFirstName] = useState("");
   354→  const [regPhone, setRegPhone] = useState("");
   355→  const [regEmail, setRegEmail] = useState("");
   356→  const [regPassword, setRegPassword] = useState("");
   357→  const [regConfirmPassword, setRegConfirmPassword] = useState("");
   358→
   359→  const categoryKey = course.category.toLowerCase();
   360→  const price = course.price || COURSE_PRICES[categoryKey] || 30000;
   361→  const schedule = COURSE_SCHEDULES[course.title];
   362→
   363→  // Step 1: Register (if not logged in)
   364→  const handleRegister = async () => {
   365→    if (!regLastName.trim() || !regFirstName.trim()) {
   366→      toast.error("Овог, нэр оруулна уу");
   367→      return;
   368→    }
   369→    if (!regPhone.trim()) {
   370→      toast.error("Утасны дугаар оруулна уу");
   371→      return;
   372→    }
   373→    if (!regEmail.trim()) {
   374→      toast.error("Имэйл оруулна уу");
   375→      return;
   376→    }
   377→    if (regPassword.length < 6) {
   378→      toast.error("Нууц үг дор хаяж 6 тэмдэгт байх ёстой");
   379→      return;
   380→    }
   381→    if (regPassword !== regConfirmPassword) {
   382→      toast.error("Нууц үг таарахгүй байна");
   383→      return;
   384→    }
   385→
   386→    setLoading(true);
   387→    try {
   388→      const res = await fetch("/api/auth/register", {
   389→        method: "POST",
   390→        headers: { "Content-Type": "application/json" },
   391→        body: JSON.stringify({
   392→          firstName: regFirstName,
   393→          lastName: regLastName,
   394→          email: regEmail,
   395→          phone: regPhone,
   396→          password: regPassword,
   397→        }),
   398→      });
   399→      const data = await res.json();
   400→      if (!res.ok) {
   401→        toast.error(data.error || "Бүртгэл амжилтгүй");
   402→        return;
   403→      }
   404→      // Auto-login after registration
   405→      if (data.user && data.token) {
   406→        setAuth(data.user, data.token);
   407→      }
   408→      toast.success("Амжилттай бүртгэгдлээ!");
   409→      // Now register for the course automatically
   410→      const enrollRes = await fetch(`/api/courses/${course.id}/register`, {
   411→        method: "POST",
   412→        headers: {
   413→          Authorization: `Bearer ${data.token}`,
   414→          "Content-Type": "application/json",
   415→        },
   416→      });
   417→      const enrollData = await enrollRes.json();
   418→      if (enrollRes.ok) {
   419→        toast.success("Сургалтад бүртгэгдлээ!");
   420→        setStep("payment");
   421→      } else {
   422→        toast.error(enrollData.error || "Сургалтад бүртгэхэд алдаа гарлаа");
   423→      }
   424→    } catch {
   425→      toast.error("Холболтын алдаа");
   426→    } finally {
   427→      setLoading(false);
   428→    }
   429→  };
   430→
   431→  // Step: Login & enroll (for already logged-in users)
   432→  const handleEnrollCourse = async () => {
   433→    if (!user || !token) return;
   434→    setLoading(true);
   435→    try {
   436→      const res = await fetch(`/api/courses/${course.id}/register`, {
   437→        method: "POST",
   438→        headers: {
   439→          Authorization: `Bearer ${token}`,
   440→          "Content-Type": "application/json",
   441→        },
   442→      });
   443→      const data = await res.json();
   444→      if (res.ok) {
   445→        toast.success("Сургалтад бүртгэгдлээ!");
   446→        setStep("payment");
   447→      } else {
   448→        toast.error(data.error || "Сургалтад бүртгэхэд алдаа");
   449→      }
   450→    } catch {
   451→      toast.error("Холболтын алдаа");
   452→    } finally {
   453→      setLoading(false);
   454→    }
   455→  };
   456→
   457→  // Step: Payment
   458→  const handlePayment = async () => {
   459→    setLoading(true);
   460→    try {
   461→      const currentToken = useAuthStore.getState().token;
   462→      const res = await fetch("/api/courses/payment", {
   463→        method: "POST",
   464→        headers: {
   465→          Authorization: `Bearer ${currentToken}`,
   466→          "Content-Type": "application/json",
   467→        },
   468→        body: JSON.stringify({ courseId: course.id }),
   469→      });
   470→      const data = await res.json();
   471→
   472→      if (res.ok) {
   473→        toast.success("Төлбөр амжилттай!");
   474→        setStep("confirmed");
   475→      } else {
   476→        toast.error("Төлбөр амжилтгүй", { description: data.error });
   477→      }
   478→    } catch {
   479→      toast.error("Холболтын алдаа");
   480→    } finally {
   481→      setLoading(false);
   482→    }
   483→  };
   484→
   485→  // Dynamic step labels based on login state
   486→  const stepLabels = user
   487→    ? [
   488→        { id: "info", label: "Мэдээлэл", icon: <ClipboardList className="w-4 h-4" /> },
   489→        { id: "payment", label: "Төлбөр", icon: <CreditCard className="w-4 h-4" /> },
   490→        { id: "confirmed", label: "Баталгаа", icon: <CheckCircle2 className="w-4 h-4" /> },
   491→      ]
   492→    : [
   493→        { id: "register", label: "Бүртгэл", icon: <UserPlus className="w-4 h-4" /> },
   494→        { id: "payment", label: "Төлбөр", icon: <CreditCard className="w-4 h-4" /> },
   495→        { id: "confirmed", label: "Баталгаа", icon: <CheckCircle2 className="w-4 h-4" /> },
   496→      ];
   497→
   498→  const currentStepIdx = stepLabels.findIndex((s) => s.id === step);
   499→
   500→  return (
   501→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 backdrop-blur-sm p-4">
   502→      <motion.div
   503→        initial={{ opacity: 0, scale: 0.95 }}
   504→        animate={{ opacity: 1, scale: 1 }}
   505→        className="bg-white rounded-2xl shadow-2xl max-w-lg w-full max-h-[90vh] overflow-y-auto"
   506→      >
   507→        {/* Header */}
   508→        <div className="bg-gradient-to-r from-brand-700 to-brand-900 p-6 rounded-t-2xl">
   509→          <div className="flex items-center justify-between">
