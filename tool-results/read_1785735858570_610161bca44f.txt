   336→   336→type EnrollStep = "info" | "anket" | "payment" | "confirmed";

   338→   338→function EnrollmentDialog({
   339→   339→  course,
   340→   340→  onClose,
   341→   341→  onConfirmed,
   342→   342→}: {
   343→   343→  course: Course;
   344→   344→  onClose: () => void;
   345→   345→  onConfirmed: () => void;
   346→   346→}) {
   347→   347→  const { user, token } = useAuthStore();
   348→   348→  const [step, setStep] = useState<EnrollStep>("info");
   349→   349→  const [loading, setLoading] = useState(false);
   350→   350→
   351→   351→  // Anket fields
   352→   352→  const [organization, setOrganization] = useState("");
   353→   353→  const [position, setPosition] = useState("");

   355→   355→  const [goal, setGoal] = useState("");
   356→   356→
   357→   357→  const categoryKey = course.category.toLowerCase();
   358→   358→  const price = course.price || COURSE_PRICES[categoryKey] || 30000;
   359→   359→  const schedule = COURSE_SCHEDULES[course.title];
   360→   360→
   361→   361→  // Step 1: Check login & show course info
   362→   362→  const handleStepInfo = () => {
   363→   363→    if (!user || !token) {
   364→   364→      toast.error("Нэвтрэх шаардлагатай", { description: "Сургалтад бүртгүүлэхийн тулд нэвтэрнө үү." });
   365→   365→      return;
   366→   366→    }
   367→   367→    setStep("anket");
   368→   368→  };
   369→   369→
   370→   370→  // Step 2: Fill anket & register
   371→   371→  const handleAnket = async () => {
   372→   372→    setLoading(true);
   373→   373→    try {
   374→   374→      // First register for the course
   375→   375→      const res = await fetch(`/api/courses/${course.id}/register`, {
   376→   376→        method: "POST",
   377→   377→        headers: {
   378→   378→          Authorization: `Bearer ${token}`,
   379→   379→          "Content-Type": "application/json",
   380→   380→        },
   381→   381→      });
   382→   382→      const data = await res.json();
   383→   383→
   384→   384→      if (res.ok) {
   385→   385→        toast.success("Бүртгэл амжилттай!");
   386→   386→        setStep("payment");
   387→   387→      } else {
   388→   388→        toast.error("Бүртгэл амжилтгүй", { description: data.error || "Алдаа гарлаа" });
   389→   389→      }
   390→   390→    } catch {
   391→   391→      toast.error("Холболтын алдаа");
   392→   392→    } finally {
   393→   393→      setLoading(false);
   394→   394→    }
   395→   395→  };
   396→   396→
   397→   397→  // Step 3: Payment
   398→   398→  const handlePayment = async () => {
   399→   399→    setLoading(true);
   400→   400→    try {
   401→   401→      const res = await fetch("/api/courses/payment", {
   402→   402→        method: "POST",
   403→   403→        headers: {
   404→   404→          Authorization: `Bearer ${token}`,
   405→   405→          "Content-Type": "application/json",
   406→   406→        },
   407→   407→        body: JSON.stringify({ courseId: course.id }),
   408→   408→      });
   409→   409→      const data = await res.json();
   410→   410→
   411→   411→      if (res.ok) {
   412→   412→        toast.success("Төлбөр амжилттай!");
   413→   413→        setStep("confirmed");
   414→   414→      } else {
   415→   415→        toast.error("Төлбөр амжилтгүй", { description: data.error });
   416→   416→      }
   417→   417→    } catch {
   418→   418→      toast.error("Холболтын алдаа");
   419→   419→    } finally {
   420→   420→      setLoading(false);
   421→   421→    }
   422→   422→  };
   423→   423→
   424→   424→  const stepLabels = [
   425→   425→    { id: "info", label: "Мэдээлэл", icon: <ClipboardList className="w-4 h-4" /> },
   426→   426→    { id: "anket", label: "Анкет", icon: <FileText className="w-4 h-4" /> },
   427→   427→    { id: "payment", label: "Төлбөр", icon: <CreditCard className="w-4 h-4" /> },
   428→   428→    { id: "confirmed", label: "Баталгаа", icon: <CheckCircle2 className="w-4 h-4" /> },
   429→   429→  ];
   430→   430→
   431→   431→  const currentStepIdx = stepLabels.findIndex((s) => s.id === step);
