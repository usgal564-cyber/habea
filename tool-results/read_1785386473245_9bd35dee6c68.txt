   339→   339→/* ─── Step 3: Quiz Taking ───────────────────────────────── */
   340→   340→
   341→   341→function QuizTaking({
   342→   342→  quiz,
   343→   343→  onFinish,
   344→   344→  onBack,
   345→   345→}: {
   346→   346→  quiz: QuizListItem;
   347→   347→  onFinish: (answers: number[]) => void;
   348→   348→  onBack: () => void;
   349→   349→}) {
   350→   350→  const [currentPage, setCurrentPage] = useState(1);
   351→   351→  const [totalPages, setTotalPages] = useState(1);
   352→   352→  const [questions, setQuestions] = useState<Question[]>([]);
   353→   353→  const [answers, setAnswers] = useState<Record<string, number>>({});
   354→   354→  const [isLoading, setIsLoading] = useState(true);
   355→   355→  const [isSubmitting, setIsSubmitting] = useState(false);
   356→   356→
   357→   357→  const fetchQuestions = useCallback(
   358→   358→    async (page: number) => {
   359→   359→      setIsLoading(true);
   360→   360→      try {
   361→   361→        const res = await fetch(
   362→   362→          `/api/quiz?slug=${quiz.slug}&page=${page}&limit=${QUESTIONS_PER_PAGE}`
   363→   363→        );
   364→   364→        const data = await res.json();
   365→   365→        if (data.quiz && data.questions) {
   366→   366→          setQuestions(data.questions);
   367→   367→          setTotalPages(data.pagination.totalPages);
   368→   368→        } else {
   369→   369→          toast.error("Алдаа", {
   370→   370→            description: "Асуултуудыг ачааллахад алдаа гарлаа.",
   371→   371→          });
   372→   372→        }
   373→   373→      } catch {
   374→   374→        toast.error("Холболтын алдаа");
   375→   375→      } finally {
   376→   376→        setIsLoading(false);
   377→   377→      }
   378→   378→    },
   379→   379→    [quiz.slug]
   380→   380→  );
   381→   381→
   382→   382→  useEffect(() => {
   383→   383→    fetchQuestions(1);
   384→   384→  }, [fetchQuestions]);
   385→   385→
   386→   386→  const handleSelectAnswer = (questionId: string, optionIndex: number) => {
   387→   387→    setAnswers((prev) => ({ ...prev, [questionId]: optionIndex }));
   388→   388→  };
   389→   389→
   390→   390→  const handlePageChange = (page: number) => {
   391→   391→    setCurrentPage(page);
   392→   392→    fetchQuestions(page);
   393→   393→    window.scrollTo({ top: 0, behavior: "smooth" });
   394→   394→  };
   395→   395→
   396→   396→  const handleFinish = async () => {
   397→   397→    setIsSubmitting(true);
   398→   398→    try {
   399→   399→      const allAnswers: number[] = [];
   400→   400→      for (let p = 1; p <= totalPages; p++) {
   401→   401→        const res = await fetch(
   402→   402→          `/api/quiz?slug=${quiz.slug}&page=${p}&limit=${QUESTIONS_PER_PAGE}`
   403→   403→        );
   404→   404→        const data = await res.json();
   405→   405→        if (data.questions) {
   406→   406→          for (const q of data.questions) {
   407→   407→            allAnswers.push(answers[q.id] ?? -1);
   408→   408→          }
   409→   409→        }
   410→   410→      }
   411→   411→      onFinish(allAnswers);
   412→   412→    } catch {
   413→   413→      toast.error("Алдаа", {
   414→   414→        description: "Хариултуудыг илгээхэд алдаа гарлаа.",
   415→   415→      });
   416→   416→    } finally {
   417→   417→      setIsSubmitting(false);
   418→   418→    }
   419→   419→  };
   420→   420→
   421→   421→  const answeredCount = Object.keys(answers).length;
   422→   422→
   423→   423→  return (
   424→   424→    <motion.div initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }}>
   425→   425→      <div className="mb-6 flex items-center justify-between">
   426→   426→        <button
   427→   427→          onClick={onBack}
   428→   428→          className="flex items-center gap-2 text-sm text-muted-foreground hover:text-brand-700 transition-colors"
   429→   429→        >
   430→   430→          <ArrowLeft className="size-4" />
   431→   431→          Буцах
   432→   432→        </button>
   433→   433→        <h2 className="text-lg font-bold text-brand-900">{quiz.title}</h2>
   434→   434→      </div>
   435→   435→
   436→   436→      <Card className="border-brand-100">
   437→   437→        <CardHeader className="pb-3">
   438→   438→          <div className="flex items-center justify-between text-sm">
   439→   439→            <span className="text-muted-foreground">
   440→   440→              Хуудас {currentPage} / {totalPages}
   441→   441→            </span>
   442→   442→            <span className="text-muted-foreground">
   443→   443→              Хариулсан: {answeredCount}
   444→   444→            </span>
   445→   445→          </div>
   446→   446→          <Progress value={(answeredCount / (quiz.questionCount || 20)) * 100} className="h-2" />
   447→   447→        </CardHeader>
   448→   448→
   449→   449→        <CardContent>
   450→   450→          {isLoading ? (
   451→   451→            <div className="py-16 flex justify-center">
   452→   452→              <Loader2 className="size-8 animate-spin text-brand-500" />
   453→   453→            </div>
   454→   454→          ) : (
   455→   455→            <div className="space-y-6">
   456→   456→              {questions.map((q, idx) => {
   457→   457→                const globalIdx = (currentPage - 1) * QUESTIONS_PER_PAGE + idx;
   458→   458→                return (
   459→   459→                  <motion.div

   623→   623→
   624→   624→type Step = "selection" | "payment" | "taking" | "results";
   625→   625→
   626→   626→export default function QuizPage() {
   627→   627→  const [step, setStep] = useState<Step>("selection");
   628→   628→  const [selectedQuiz, setSelectedQuiz] = useState<QuizListItem | null>(null);
   629→   629→  const [result, setResult] = useState<QuizResult | null>(null);
   630→   630→  const [isSubmitting, setIsSubmitting] = useState(false);
   631→   631→
   632→   632→  const handleSelectQuiz = useCallback((quiz: QuizListItem) => {
   633→   633→    setSelectedQuiz(quiz);
   634→   634→    setResult(null);
   635→   635→    setStep("payment");
   636→   636→  }, []);
   637→   637→
   638→   638→  const handlePaid = useCallback(() => {
   639→   639→    setStep("taking");
   640→   640→  }, []);
   641→   641→
   642→   642→  const handleFinish = useCallback(
   643→   643→    async (answers: number[]) => {
   644→   644→      if (!selectedQuiz) return;
   645→   645→
   646→   646→      setIsSubmitting(true);
   647→   647→      try {
   648→   648→        const res = await fetch("/api/quiz", {
   649→   649→          method: "POST",
   650→   650→          headers: { "Content-Type": "application/json" },
   651→   651→          body: JSON.stringify({
   652→   652→            quizId: selectedQuiz.id,
   653→   653→            answers,
   654→   654→          }),
   655→   655→        });
   656→   656→
   657→   657→        const data = await res.json();
   658→   658→
   659→   659→        if (res.ok) {
   660→   660→          setResult({
   661→   661→            attemptId: data.attemptId,
   662→   662→            score: data.score,
   663→   663→            total: data.total,
   664→   664→            passed: data.passed,
   665→   665→          });
   666→   666→          setStep("results");
   667→   667→
   668→   668→          if (data.passed) {
   669→   669→            toast.success("Баяр хүргэе!", {
   670→   670→              description: `Та ${data.score}/${data.total} оноо авч тэнцлээ!`,
   671→   671→            });
   672→   672→          } else {
   673→   673→            toast.error("Тэнцээгүй", {
   674→   674→              description: `Та ${data.score}/${data.total} оноо авсан байна. 70% шаардлагатай.`,
   675→   675→            });
   676→   676→          }
   677→   677→        } else {
   678→   678→          toast.error("Алдаа", {
   679→   679→            description: data.error || "Хариултуудыг илгээхэд алдаа гарлаа.",
   680→   680→          });
   681→   681→        }
   682→   682→      } catch {
   683→   683→        toast.error("Холболтын алдаа");
   684→   684→      } finally {
   685→   685→        setIsSubmitting(false);
   686→   686→      }
   687→   687→    },
   688→   688→    [selectedQuiz]
   689→   689→  );
   690→   690→
   691→   691→  const handleRetry = useCallback(() => {
   692→   692→    setResult(null);
   693→   693→    setStep("payment");
   694→   694→  }, []);
   695→   695→
   696→   696→  const handleBackToList = useCallback(() => {
   697→   697→    setSelectedQuiz(null);
   698→   698→    setResult(null);
   699→   699→    setStep("selection");
   700→   700→  }, []);
   701→   701→
   702→   702→  return (
   703→   703→    <section className="w-full">
   704→   704→      {step === "selection" && <QuizSelection onSelect={handleSelectQuiz} />}
   705→   705→
   706→   706→      {step === "payment" && selectedQuiz && (
   707→   707→        <PaymentGate quiz={selectedQuiz} onPaid={handlePaid} onBack={handleBackToList} />
   708→   708→      )}
   709→   709→
   710→   710→      {step === "taking" && selectedQuiz && (
   711→   711→        <QuizTaking
   712→   712→          quiz={selectedQuiz}
   713→   713→          onFinish={handleFinish}
   714→   714→          onBack={handleBackToList}
   715→   715→        />
   716→   716→      )}
   717→   717→
   718→   718→      {step === "results" && selectedQuiz && result && (
   719→   719→        <QuizResults
   720→   720→          quiz={selectedQuiz}
   721→   721→          result={result}
   722→   722→          onRetry={handleRetry}
   723→   723→          onBackToList={handleBackToList}
   724→   724→        />
   725→   725→      )}
   726→   726→
   727→   727→      {/* Submit overlay */}
   728→   728→      <AnimatePresence>
   729→   729→        {isSubmitting && (
   730→   730→          <motion.div
   731→   731→            initial={{ opacity: 0 }}
   732→   732→            animate={{ opacity: 1 }}
   733→   733→            exit={{ opacity: 0 }}
   734→   734→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 backdrop-blur-sm"
   735→   735→          >
   736→   736→            <div className="flex flex-col items-center gap-4 rounded-2xl bg-white p-8 shadow-2xl">
   737→   737→              <Loader2 className="size-10 animate-spin text-brand-600" />
   738→   738→              <p className="text-lg font-medium text-brand-900">
   739→   739→                Хариултуудыг шалгаж байна...
   740→   740→              </p>
   741→   741→            </div>
   742→   742→          </motion.div>
   743→   743→        )}
   744→   744→      </AnimatePresence>
   745→   745→    </section>
   746→   746→  );
   747→   747→}
