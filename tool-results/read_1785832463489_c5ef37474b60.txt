   154→function SimpleBarChart({ responses }: { responses: Record<string, string | number> }) {
   155→  // Only show rating-type questions in the chart
   156→  const ratingQuestions = surveyQuestions.filter((q) => q.type === "rating");
   157→  const maxRating = 5;
   158→
   159→  return (
   160→    <div className="space-y-4">
   161→      {ratingQuestions.map((q, i) => {
   162→        const val = (responses[q.id] as number) || 0;
   163→        const percent = (val / maxRating) * 100;
   164→        return (
   165→          <motion.div
   166→            key={q.id}
   167→            initial={{ opacity: 0, x: -20 }}
   168→            animate={{ opacity: 1, x: 0 }}
   169→            transition={{ duration: 0.4, delay: i * 0.1 }}
   170→          >
   171→            <div className="flex items-center justify-between mb-1.5">
   172→              <span className="text-sm font-medium text-foreground">
   173→                {q.label || q.question}
   174→              </span>
   175→              <span className="text-sm font-bold text-brand-600">
   176→                {val}/{maxRating}
   177→              </span>
   178→            </div>
   179→            <div className="h-4 bg-brand-100 rounded-full overflow-hidden">
   180→              <motion.div
   181→                className="h-full bg-gradient-to-r from-brand-500 to-brand-600 rounded-full"
   182→                initial={{ width: 0 }}
   183→                animate={{ width: `${percent}%` }}
   184→                transition={{ duration: 0.6, delay: 0.3 + i * 0.1, ease: "easeOut" }}
   185→              />
   186→            </div>
   187→          </motion.div>
   188→        );
   189→      })}
   190→      {/* Average */}
   191→      {(() => {
   192→        const ratingVals = ratingQuestions
   193→          .map((q) => responses[q.id] as number)
   194→          .filter((v) => typeof v === "number" && v > 0);
   195→        const avg = ratingVals.length
   196→          ? ratingVals.reduce((a, b) => a + b, 0) / ratingVals.length
   197→          : 0;
   198→        return (
   199→          <motion.div
   200→            initial={{ opacity: 0, y: 10 }}
   201→            animate={{ opacity: 1, y: 0 }}
   202→            transition={{ duration: 0.4, delay: 0.6 }}
   203→            className="mt-6 pt-4 border-t border-brand-200"
   204→          >
   205→            <div className="flex items-center justify-between mb-2">
   206→              <span className="text-base font-bold text-foreground">
   207→                Дундаж үнэлгээ
   208→              </span>
   209→              <span className="text-lg font-bold text-brand-600">
   210→                {avg.toFixed(1)}/{maxRating}
   211→              </span>
   212→            </div>
   213→            <div className="h-5 bg-brand-100 rounded-full overflow-hidden">
   214→              <motion.div
   215→                className="h-full bg-gradient-to-r from-brand-600 to-brand-500 rounded-full"
   216→                initial={{ width: 0 }}
   217→                animate={{ width: `${(avg / maxRating) * 100}%` }}
   218→                transition={{ duration: 0.8, delay: 0.8, ease: "easeOut" }}
   219→              />
   220→            </div>
   221→          </motion.div>
   222→        );
   223→      })()}
   224→    </div>
   225→  );
   226→}
