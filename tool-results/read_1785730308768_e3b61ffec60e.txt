   327→/* ─── Enrollment Dialog (Multi-step) ──────────────────── */

   329→type EnrollStep = "info" | "anket" | "payment" | "confirmed";
   330→
   331→function EnrollmentDialog({
   332→  course,
   333→  onClose,
   334→  onConfirmed,
   335→}: {
   336→  course: Course;
   337→  onClose: () => void;
   338→  onConfirmed: () => void;
   339→}) {
   340→  const { user, token } = useAuthStore();
   341→  const [step, setStep] = useState<EnrollStep>("info");
   342→  const [loading, setLoading] = useState(false);
   343→
   344→  // Anket fields
   345→  const [organization, setOrganization] = useState("");
   346→  const [position, setPosition] = useState("");
   347→  const [experience, setExperience] = useState("");
   348→  const [goal, setGoal] = useState("");
   349→
   350→  const categoryKey = course.category.toLowerCase();
   351→  const price = course.price || COURSE_PRICES[categoryKey] || 30000;
   352→  const schedule = COURSE_SCHEDULES[course.title];
   353→
   354→  // Step 1: Check login & show course info
   355→  const handleStepInfo = () => {
   356→    if (!user || !token) {
   357→      toast.error("Нэвтрэх шаардлагатай", { description: "Сургалтад бүртгүүлэхийн тулд нэвтэрнө үү." });
   358→      return;
   359→    }
   360→    setStep("anket");
   361→  };
   362→
   363→  // Step 2: Fill anket & register
   364→  const handleAnket = async () => {
   365→    setLoading(true);
   366→    try {
   367→      // First register for the course
   368→      const res = await fetch(`/api/courses/${course.id}/register`, {
   369→        method: "POST",
   370→        headers: {
   371→          Authorization: `Bearer ${token}`,
   372→          "Content-Type": "application/json",
   373→        },
   374→      });
   375→      const data = await res.json();
   376→
   377→      if (res.ok) {
   378→        toast.success("Бүртгэл амжилттай!");
   379→        setStep("payment");
   380→      } else {
   381→        toast.error("Бүртгэл амжилтгүй", { description: data.error || "Алдаа гарлаа" });
   382→      }
   383→    } catch {
   384→      toast.error("Холболтын алдаа");
   385→    } finally {
   386→      setLoading(false);
   387→    }
   388→  };
   389→
   390→  // Step 3: Payment
   391→  const handlePayment = async () => {
   392→    setLoading(true);
   393→    try {
   394→      const res = await fetch("/api/courses/payment", {
   395→        method: "POST",
   396→        headers: {
   397→          Authorization: `Bearer ${token}`,
   398→          "Content-Type": "application/json",
   399→        },
   400→        body: JSON.stringify({ courseId: course.id }),
   401→      });
   402→      const data = await res.json();
   403→
   404→      if (res.ok) {
   405→        toast.success("Төлбөр амжилттай!");
   406→        setStep("confirmed");
   407→      } else {
   408→        toast.error("Төлбөр амжилтгүй", { description: data.error });
   409→      }
   410→    } catch {
   411→      toast.error("Холболтын алдаа");
   412→    } finally {
   413→      setLoading(false);
   414→    }
   415→  };
   416→
   417→  const stepLabels = [
   418→    { id: "info", label: "Мэдээлэл", icon: <ClipboardList className="w-4 h-4" /> },
   419→    { id: "anket", label: "Анкет", icon: <FileText className="w-4 h-4" /> },
   420→    { id: "payment", label: "Төлбөр", icon: <CreditCard className="w-4 h-4" /> },
   421→    { id: "confirmed", label: "Баталгаа", icon: <CheckCircle2 className="w-4 h-4" /> },
   422→  ];
   423→
   424→  const currentStepIdx = stepLabels.findIndex((s) => s.id === step);
   425→
   426→  return (
   427→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 backdrop-blur-sm p-4">
   428→      <motion.div
   429→        initial={{ opacity: 0, scale: 0.95 }}
   430→        animate={{ opacity: 1, scale: 1 }}
   431→        className="bg-white rounded-2xl shadow-2xl max-w-lg w-full max-h-[90vh] overflow-y-auto"
   432→      >
   433→        {/* Header */}
   434→        <div className="bg-gradient-to-r from-brand-700 to-brand-900 p-6 rounded-t-2xl">
   435→          <div className="flex items-center justify-between">
   436→            <div>
   437→              <h3 className="text-xl font-bold text-white">{course.title}</h3>
   438→              <p className="text-brand-200 text-sm mt-1">{COURSE_PRICES[categoryKey] ? "Төлбөртэй сургалт" : "Сургалтад бүртгүүлэх"}</p>
   439→            </div>
   440→            <button
   441→              onClick={onClose}
   442→              className="w-8 h-8 rounded-full bg-white/10 hover:bg-white/20 flex items-center justify-center text-white transition-colors"
   443→            >
   444→              ✕
   445→            </button>
   446→          </div>
   447→
   448→          {/* Step indicator */}
   449→          <div className="mt-5">
   450→            <div className="flex items-center justify-between relative">
   451→              {/* Background line */}
   452→              <div className="absolute top-[18px] left-[45px] right-[45px] h-[2px] bg-white/20" />
   453→              {/* Active line */}
   454→              <div
   455→                className="absolute top-[18px] left-[45px] h-[2px] bg-white transition-all duration-500"
   456→                style={{ width: currentStepIdx > 0 ? `calc(${currentStepIdx * 100}% / 3 + ${currentStepIdx * 20}px / 3)` : "0%" }}
   457→              />
   458→              {/* Steps */}
   459→              {stepLabels.map((s, i) => (
   460→                <div key={s.id} className="flex flex-col items-center gap-1.5 relative z-10">
   461→                  <div
   462→                    className={`w-9 h-9 rounded-full flex items-center justify-center text-sm font-medium transition-all ${
   463→                      i <= currentStepIdx
   464→                        ? "bg-white text-brand-700 shadow-sm"
   465→                        : "bg-white/20 text-white/60"
   466→                    }`}
   467→                  >
   468→                    {s.icon}
   469→                  </div>
   470→                  <span className={`text-[10px] font-medium transition-colors ${
   471→                    i <= currentStepIdx ? "text-white/90" : "text-white/40"
   472→                  }`}>
   473→                    {s.label}
   474→                  </span>
   475→                </div>
   476→              ))}
   477→            </div>
   478→          </div>
   479→        </div>
   480→
   481→        {/* Content */}
   482→        <div className="p-6">
   483→          <AnimatePresence mode="wait">
   484→            {/* Step 1: Course Info */}
   485→            {step === "info" && (
   486→              <motion.div key="info" initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }} className="space-y-4">
   487→                <div className="bg-brand-50 rounded-xl p-4 space-y-3">
   488→                  <div className="flex items-center gap-2 text-sm">
   489→                    <Clock className="w-4 h-4 text-brand-600" />
   490→                    <span className="text-muted-foreground">Хугацаа:</span>
   491→                    <span className="font-medium text-brand-900">{course.duration}</span>
   492→                  </div>
   493→                  <div className="flex items-center gap-2 text-sm">
   494→                    <Banknote className="w-4 h-4 text-brand-600" />
   495→                    <span className="text-muted-foreground">Үнэ:</span>
   496→                    <span className="font-bold text-brand-900">{price.toLocaleString()}₮</span>
   497→                  </div>
   498→                  {course.maxStudents && (
   499→                    <div className="flex items-center gap-2 text-sm">
