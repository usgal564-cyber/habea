   154→func generateCode() string {
   155→        const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"
   156→        code := make([]byte, 6)
   157→        for i := range code {
   158→                n, _ := rand.Int(rand.Reader, big.NewInt(int64(len(chars))))
   159→                code[i] = chars[n.Int64()]
   160→        }
   161→        return string(code)
   162→}

   179→func RegisterHandler(c *gin.Context) {
   180→        var req RegisterRequest
   181→        if err := c.ShouldBindJSON(&req); err != nil {
   182→                c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
   183→                return
   184→        }
   185→        var existing User
   186→        if DB.Where("email = ?", req.Email).First(&existing).Error == nil {
   187→                c.JSON(http.StatusConflict, gin.H{"error": "Бүртгэлтэй имэйл байна"})
   188→                return
   189→        }
   190→        hash, _ := bcrypt.GenerateFromPassword([]byte(req.Password), bcrypt.DefaultCost)
   191→        user := User{ID: uuid.New().String(), FirstName: req.FirstName, LastName: req.LastName, Email: req.Email, Phone: req.Phone, Password: string(hash), Address: req.Address, SecondaryPhone: req.SecondaryPhone, Role: "USER"}
   192→        if err := DB.Create(&user).Error; err != nil {
   193→                c.JSON(500, gin.H{"error": "Failed to create user"})
   194→                return
   195→        }
   196→        token, _ := GenerateToken(user.ID, user.Email, user.Role, user.FirstName+" "+user.LastName)
   197→        c.JSON(201, gin.H{"user": userResponse(user), "token": token})
   198→}

   376→func QuizSubmitHandler(c *gin.Context) {

   394→
   395→        ansJSON, _ := json.Marshal(req.Answers)
   396→        attempt := QuizAttempt{ID: uuid.New().String(), UserID: userID.(string), QuizID: req.QuizID, Score: score, Total: len(questions), Answers: string(ansJSON)}
   397→        DB.Create(&attempt)
   398→
   399→        c.JSON(200, gin.H{"score": score, "total": len(questions), "passed": score >= int(float64(len(questions))*0.8), "attemptId": attempt.ID})

   404→func ExamHandler(c *gin.Context) {
   405→        var req ExamActionRequest
   406→        if err := c.ShouldBindJSON(&req); err != nil {
   407→                c.JSON(400, gin.H{"error": err.Error()})
   408→                return
   409→        }
   410→
   411→        if req.Action == "verify" {
   412→                var exam Exam
   413→                if DB.Where("code = ? AND is_active = ?", req.Code, true).First(&exam).Error != nil {
   414→                        c.JSON(404, gin.H{"error": "Идэвхтэй шалгалт олдсонгүй эсвэл код буруу байна"})
   415→                        return
   416→                }
   417→                c.JSON(200, gin.H{"exam": gin.H{"id": exam.ID, "title": exam.Title, "questionCount": exam.QuestionCount, "duration": exam.Duration}})
   418→                return
   419→        }
   420→
   421→        if req.Action == "submit" {
   422→                userID, _ := c.Get("userId")
   423→                // Check if already took
   424→                var existing ExamAttempt
   425→                if DB.Where("exam_id = ? AND user_id = ?", req.ExamID, userID).First(&existing).Error == nil {
   426→                        c.JSON(403, gin.H{"error": "Та энэ шалгалтад өмнө нь орсон байна. Дахин өгөх боломжгүй."})
   427→                        return
   428→                }
   429→
   430→                var questions []ExamQuestion
   431→                DB.Where("exam_id = ?", req.ExamID).Order("index asc").Find(&questions)
   432→
   433→                score := 0
   434→                for i, ans := range req.Answers {
   435→                        if i < len(questions) && ans == questions[i].Correct {
   436→                                score++
   437→                        }
   438→                }
   439→                passed := len(questions) > 0 && score >= int(float64(len(questions))*0.8)
   440→
   441→                ansJSON, _ := json.Marshal(req.Answers)
   442→                attempt := ExamAttempt{ID: uuid.New().String(), UserID: userID.(string), ExamID: req.ExamID, Score: score, Total: len(questions), Passed: passed, Answers: string(ansJSON), TimeSpent: req.TimeSpent}
   443→                DB.Create(&attempt)
   444→
   445→                c.JSON(200, gin.H{"score": score, "total": len(questions), "passed": passed, "timeSpent": req.TimeSpent})
   446→                return
   447→        }
   448→
   449→        c.JSON(400, gin.H{"error": "Invalid action"})
   450→}

   485→func ExportExamHandler(c *gin.Context) {
   486→        attemptID := c.Param("attemptId")
   487→        var attempt ExamAttempt
   488→        if DB.Where("id = ?", attemptID).First(&attempt).Error != nil {
   489→                c.JSON(404, gin.H{"error": "Not found"})
   490→                return
   491→        }
   492→        var exam Exam
   493→        DB.Where("id = ?", attempt.ExamID).First(&exam)
   494→        var questions []ExamQuestion
   495→        DB.Where("exam_id = ?", attempt.ExamID).Order("index asc").Find(&questions)
   496→
   497→        var answers []int
   498→        json.Unmarshal([]byte(attempt.Answers), &answers)
   499→
   500→        c.Header("Content-Type", "text/csv; charset=utf-8")
   501→        c.Header("Content-Disposition", fmt.Sprintf("attachment; filename=exam_%s_result.csv", exam.Code))
   502→        writer := csv.NewWriter(c.Writer)
   503→        writer.Write([]string{"Шалгалт", exam.Title, "", "Код", exam.Code, "", "Огноо", attempt.CreatedAt.Format("2006-01-02 15:04:05")})
   504→        writer.Write([]string{"#", "Асуулт", "A", "B", "C", "D", "Таны хариулт", "Зөв хариулт", "Дүн"})
   505→        for i, q := range questions {
   506→                ansStr := "-"
   507→                correctStr := string([]string{"A", "B", "C", "D"}[q.Correct])
   508→                if i < len(answers) && answers[i] >= 0 {
   509→                        ansStr = string([]string{"A", "B", "C", "D"}[answers[i]])
   510→                }
   511→                status := "Буруу"
   512→                if i < len(answers) && answers[i] == q.Correct {
   513→                        status = "Зөв"
   514→                }
   515→                writer.Write([]string{fmt.Sprintf("%d", i+1), q.Question, q.OptionA, q.OptionB, q.OptionC, q.OptionD, ansStr, correctStr, status})
   516→        }
   517→        writer.Write([]string{"", "", "", "", "", "", "", "", ""})
   518→        writer.Write([]string{"Нийт", "", "", "", "", "", fmt.Sprintf("%d/%d", attempt.Score, attempt.Total), "", fmt.Sprintf("%v", attempt.Passed)})
   519→        writer.Flush()
   520→}

   603→func GetSurveyResultsHandler(c *gin.Context) {
   604→        var responses []SurveyResponse
   605→        DB.Order("created_at desc").Find(&responses)
   606→
   607→        type Stat struct {
   608→                QuestionID   string      `json:"questionId"`
   609→                Ratings      []int       `json:"ratings"`
   610→                AvgRating    float64     `json:"avgRating"`
   611→                YesCount     int         `json:"yesCount"`
   612→                NoCount      int         `json:"noCount"`
   613→                TextAnswers  []string    `json:"textAnswers"`
   614→        }
   615→
   616→        // Aggregate
   617→        questionStats := make(map[string]*Stat)
   618→        totalResponses := len(responses)
   619→
   620→        for _, r := range responses {
   621→                var parsed map[string]interface{}
   622→                json.Unmarshal([]byte(r.Responses), &parsed)
   623→                for qID, val := range parsed {
   624→                        if questionStats[qID] == nil {
   625→                                questionStats[qID] = &Stat{QuestionID: qID}
   626→                        }
   627→                        stat := questionStats[qID]
   628→                        switch v := val.(type) {
   629→                        case float64:
   630→                                stat.Ratings = append(stat.Ratings, int(v))
   631→                        case string:
   632→                                if v == "yes" {
   633→                                        stat.YesCount++
   634→                                } else if v == "no" {
   635→                                        stat.NoCount++
   636→                                } else {
   637→                                        stat.TextAnswers = append(stat.TextAnswers, v)
   638→                                }
   639→                        }
   640→                }
   641→        }
   642→
   643→        var stats []*Stat
   644→        for _, s := range questionStats {
   645→                if len(s.Ratings) > 0 {
   646→                        sum := 0
   647→                        for _, r := range s.Ratings { sum += r }
   648→                        s.AvgRating = float64(sum) / float64(len(s.Ratings))
   649→                }
   650→                stats = append(stats, s)
   651→        }
   652→
   653→        c.JSON(200, gin.H{"totalResponses": totalResponses, "stats": stats, "responses": responses})
   654→}
