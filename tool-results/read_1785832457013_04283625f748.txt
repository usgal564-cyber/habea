   347→/* ─── Step 2: Payment Dialog ──────────────────────────────── */
   348→
   349→function PaymentGate({
   350→  quiz,
   351→  onPaid,
   352→  onBack,
   353→}: {
   354→  quiz: QuizListItem;
   355→  onPaid: () => void;
   356→  onBack: () => void;
   357→}) {
   358→  const { token } = useAuthStore();
   359→  const [paying, setPaying] = useState(false);
   360→
   361→  const handlePay = async () => {
   362→    setPaying(true);
   363→    try {
   364→      const res = await fetch("/api/quiz/payment", {
   365→        method: "POST",
   366→        headers: {
   367→          "Content-Type": "application/json",
   368→          Authorization: `Bearer ${token}`,
   369→        },
   370→        body: JSON.stringify({ quizId: quiz.id }),
   371→      });
   372→      const data = await res.json();
   373→      if (res.ok) {
   374→        toast.success("Төлбөр амжилттай");
   375→        onPaid();
   376→      } else {
   377→        toast.error("Төлбөр амжилтгүй", { description: data.error || "Дахин оролдоно уу" });
   378→      }
   379→    } catch {
   380→      toast.error("Холболтын алдаа");
   381→    } finally {
   382→      setPaying(false);
   383→    }
   384→  };
   385→
   386→  return (
   387→    <Dialog open onOpenChange={() => onBack()}>
   388→      <DialogContent className="sm:max-w-md bg-white">
   389→        <DialogHeader>
   390→          <DialogTitle className="text-brand-900 font-bold text-xl flex items-center gap-2">
   391→            <Wallet className="w-5 h-5 text-brand-600" />
   392→            Төлбөр төлөх
   393→          </DialogTitle>
   394→        </DialogHeader>
   395→
   396→        <div className="space-y-5 pt-2">
   397→          {/* Quiz Info */}
   398→          <div className="bg-brand-50 rounded-xl p-4">
   399→            <p className="text-sm text-muted-foreground">Сорил</p>
   400→            <p className="text-lg font-bold text-brand-900">{quiz.title}</p>
   401→            <div className="flex items-center gap-3 mt-2 text-sm text-muted-foreground">
   402→              <span className="flex items-center gap-1">
   403→                <ListChecks className="w-4 h-4" />
   404→                {quiz.questionCount} асуулт
   405→              </span>
   406→            </div>
   407→          </div>
   408→
   409→          {/* Price */}
   410→          <div className="flex items-center justify-between bg-amber-50 rounded-xl p-4 border border-amber-200">
   411→            <span className="text-sm font-medium text-amber-800">Төлбөрийн дүн:</span>
   412→            <span className="text-2xl font-bold text-amber-900">{quiz.price.toLocaleString()}₮</span>
   413→          </div>
   414→
   415→          {/* Payment Info */}
   416→          <div className="bg-gray-50 rounded-xl p-4 text-sm text-muted-foreground space-y-2">
   417→            <p className="flex items-center gap-2">
   418→              <ShieldCheck className="w-4 h-4 text-brand-600" />
   419→              Төлбөр амжилттай хийгдсэний дараа сорил эхлүүлнэ
   420→            </p>
   421→            <p className="flex items-center gap-2">
   422→              <CreditCard className="w-4 h-4 text-brand-600" />
   423→              70% бөгөөд тэнцсэн тохиолдод сертификат олгогдоно
   424→            </p>
   425→          </div>
   426→
   427→          {/* Actions */}
   428→          <div className="flex gap-3">
   429→            <Button variant="outline" onClick={onBack} className="flex-1" disabled={paying}>
   430→              <ArrowLeft className="w-4 h-4 mr-1" />
   431→              Буцах
   432→            </Button>
   433→            <Button
   434→              onClick={handlePay}
   435→              className="flex-1 bg-brand-600 hover:bg-brand-700 text-white"
   436→              disabled={paying}
   437→            >
   438→              {paying ? (
   439→                <>
   440→                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   441→                  Төлбөр хийгдэж байна...
   442→                </>
   443→              ) : (
   444→                <>
   445→                  <CreditCard className="w-4 h-4 mr-1" />
   446→                  {quiz.price.toLocaleString()}₮ төлөх
   447→                </>
   448→              )}
   449→            </Button>
   450→          </div>
   451→        </div>
   452→      </DialogContent>
   453→    </Dialog>
   454→  );
   455→}

   457→/* ─── Step 3: Quiz Taking ───────────────────────────────── */
   458→
   459→function QuizTaking({
   460→  quiz,
   461→  onFinish,
   462→  onBack,
   463→}: {
   464→  quiz: QuizListItem;
   465→  onFinish: (answers: number[]) => void;
   466→  onBack: () => void;
   467→}) {
   468→  const [currentPage, setCurrentPage] = useState(1);
   469→  const [totalPages, setTotalPages] = useState(1);
   470→  const [questions, setQuestions] = useState<Question[]>([]);
   471→  const [answers, setAnswers] = useState<Record<string, number>>({});
   472→  const [isLoading, setIsLoading] = useState(true);
   473→  const [isSubmitting, setIsSubmitting] = useState(false);
   474→
   475→  const fetchQuestions = useCallback(
   476→    async (page: number) => {
   477→      setIsLoading(true);
   478→      try {
   479→        const res = await fetch(
   480→          `/api/quiz?slug=${quiz.slug}&page=${page}&limit=${QUESTIONS_PER_PAGE}`
   481→        );
   482→        const data = await res.json();
   483→        if (data.quiz && data.questions) {
   484→          setQuestions(data.questions);
   485→          setTotalPages(data.pagination.totalPages);
   486→        } else {
   487→          toast.error("Алдаа", {
   488→            description: "Асуултуудыг ачааллахад алдаа гарлаа.",
   489→          });
   490→        }
   491→      } catch {
   492→        toast.error("Холболтын алдаа");
   493→      } finally {
   494→        setIsLoading(false);
   495→      }
   496→    },
   497→    [quiz.slug]
   498→  );
   499→
   500→  useEffect(() => {
   501→    fetchQuestions(1);
   502→  }, [fetchQuestions]);
   503→
   504→  const handleSelectAnswer = (questionId: string, optionIndex: number) => {
   505→    setAnswers((prev) => ({ ...prev, [questionId]: optionIndex }));
   506→  };
   507→
   508→  const handlePageChange = (page: number) => {
   509→    setCurrentPage(page);
   510→    fetchQuestions(page);
   511→    window.scrollTo({ top: 0, behavior: "smooth" });
   512→  };
   513→
   514→  const handleFinish = async () => {
   515→    setIsSubmitting(true);
   516→    try {
   517→      const allAnswers: number[] = [];
   518→      for (let p = 1; p <= totalPages; p++) {
   519→        const res = await fetch(
   520→          `/api/quiz?slug=${quiz.slug}&page=${p}&limit=${QUESTIONS_PER_PAGE}`
   521→        );
   522→        const data = await res.json();
   523→        if (data.questions) {
   524→          for (const q of data.questions) {
   525→            allAnswers.push(answers[q.id] ?? -1);
   526→          }
   527→        }
   528→      }
   529→      onFinish(allAnswers);
   530→    } catch {
   531→      toast.error("Алдаа", {
   532→        description: "Хариултуудыг илгээхэд алдаа гарлаа.",
   533→      });
   534→    } finally {
   535→      setIsSubmitting(false);
   536→    }
   537→  };
   538→
   539→  const answeredCount = Object.keys(answers).length;
   540→
   541→  return (
   542→    <motion.div initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }}>
   543→      <div className="mb-6 flex items-center justify-between">
   544→        <button
   545→          onClick={onBack}
   546→          className="flex items-center gap-2 text-sm text-muted-foreground hover:text-brand-700 transition-colors"
   547→        >
   548→          <ArrowLeft className="size-4" />
   549→          Буцах
   550→        </button>
   551→        <h2 className="text-lg font-bold text-brand-900">{quiz.title}</h2>
   552→      </div>
   553→
   554→      <Card className="border-brand-100">
   555→        <CardHeader className="pb-3">
   556→          <div className="flex items-center justify-between text-sm">
   557→            <span className="text-muted-foreground">
   558→              Хуудас {currentPage} / {totalPages}
   559→            </span>
   560→            <span className="text-muted-foreground">
   561→              Хариулсан: {answeredCount}
   562→            </span>
   563→          </div>
   564→          <Progress value={(answeredCount / (quiz.questionCount || 20)) * 100} className="h-2" />
   565→        </CardHeader>
   566→
   567→        <CardContent>
   568→          {isLoading ? (
   569→            <div className="py-16 flex justify-center">
   570→              <Loader2 className="size-8 animate-spin text-brand-500" />
   571→            </div>
   572→          ) : (
   573→            <div className="space-y-6">
   574→              {questions.map((q, idx) => {
   575→                const globalIdx = (currentPage - 1) * QUESTIONS_PER_PAGE + idx;
   576→                return (
   577→                  <motion.div
   578→                    key={q.id}
   579→                    initial={{ opacity: 0, y: 10 }}
   580→                    animate={{ opacity: 1, y: 0 }}
   581→                    transition={{ delay: idx * 0.03 }}
   582→                    className="bg-gray-50 rounded-xl p-4"
   583→                  >
   584→                    <p className="font-medium text-brand-900 mb-3">
   585→                      {globalIdx + 1}. {q.questionText}
   586→                    </p>
   587→                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   588→                      {[
   589→                        { key: 0, label: "A", text: q.optionA },
   590→                        { key: 1, label: "B", text: q.optionB },
   591→                        { key: 2, label: "C", text: q.optionC },
   592→                        { key: 3, label: "D", text: q.optionD },
   593→                      ].map((opt) => (
   594→                        <button
   595→                          key={opt.key}
   596→                          onClick={() => handleSelectAnswer(q.id, opt.key)}
   597→                          className={`text-left p-3 rounded-lg border-2 transition-all text-sm ${
   598→                            answers[q.id] === opt.key
   599→                              ? "border-brand-500 bg-brand-50 text-brand-900"
   600→                              : "border-gray-200 hover:border-brand-300 hover:bg-brand-50/50"
   601→                          }`}
   602→                        >
   603→                          <span className="font-bold mr-2">{opt.label})</span>
   604→                          {opt.text}
   605→                        </button>
   606→                      ))}
   607→                    </div>
   608→                  </motion.div>
   609→                );
   610→              })}
   611→
   612→              {/* Pagination */}
   613→              <div className="flex items-center justify-between pt-4 border-t">
   614→                <Button
   615→                  variant="outline"
   616→                  onClick={() => handlePageChange(Math.max(1, currentPage - 1))}
   617→                  disabled={currentPage === 1}
   618→                >
   619→                  <ChevronLeft className="w-4 h-4 mr-1" /> Өмнөх
   620→                </Button>
   621→                <div className="flex gap-1 flex-wrap justify-center">
   622→                  {Array.from({ length: totalPages }, (_, i) => i + 1).map((p) => (
   623→                    <button
   624→                      key={p}
   625→                      onClick={() => handlePageChange(p)}
   626→                      className={`w-8 h-8 rounded-lg text-sm font-medium transition-all ${
   627→                        p === currentPage
   628→                          ? "bg-brand-600 text-white"
   629→                          : "bg-gray-100 hover:bg-brand-100 text-gray-700"
   630→                      }`}
   631→                    >
   632→                      {p}
   633→                    </button>
   634→                  ))}
   635→                </div>
   636→                {currentPage < totalPages ? (
   637→                  <Button
   638→                    onClick={() => handlePageChange(Math.min(totalPages, currentPage + 1))}
   639→                  >
   640→                    Дараах <ChevronRight className="w-4 h-4 ml-1" />
   641→                  </Button>
   642→                ) : (
   643→                  <Button
   644→                    onClick={handleFinish}
   645→                    className="bg-brand-600 hover:bg-brand-700"
   646→                    disabled={isSubmitting}
   647→                  >
   648→                    {isSubmitting ? (
   649→                      <>
   650→                        <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   651→                        Илгээж байна...
   652→                      </>
   653→                    ) : (
   654→                      "Дуусгах"
   655→                    )}
   656→                  </Button>
   657→                )}
   658→              </div>
   659→            </div>
   660→          )}
   661→        </CardContent>
   662→      </Card>
   663→    </motion.div>
   664→  );
   665→}

   740→/* ─── Main Component ──────────────────────────────────────── */
   741→
   742→type Step = "selection" | "payment" | "taking" | "results";
   743→
   744→export default function QuizPage() {
   745→  const [step, setStep] = useState<Step>("selection");
   746→  const [selectedQuiz, setSelectedQuiz] = useState<QuizListItem | null>(null);
   747→  const [result, setResult] = useState<QuizResult | null>(null);
   748→  const [isSubmitting, setIsSubmitting] = useState(false);
   749→
   750→  const handleSelectQuiz = useCallback((quiz: QuizListItem) => {
   751→    setSelectedQuiz(quiz);
   752→    setResult(null);
   753→    setStep("payment");
   754→  }, []);
   755→
   756→  const handlePaid = useCallback(() => {
   757→    setStep("taking");
   758→  }, []);
   759→
   760→  const handleFinish = useCallback(
   761→    async (answers: number[]) => {
   762→      if (!selectedQuiz) return;
   763→
   764→      setIsSubmitting(true);
   765→      try {
   766→        const res = await fetch("/api/quiz", {
   767→          method: "POST",
   768→          headers: { "Content-Type": "application/json" },
   769→          body: JSON.stringify({
   770→            quizId: selectedQuiz.id,
   771→            answers,
   772→          }),
   773→        });
   774→
   775→        const data = await res.json();
   776→
   777→        if (res.ok) {
   778→          setResult({
   779→            attemptId: data.attemptId,
   780→            score: data.score,
   781→            total: data.total,
   782→            passed: data.passed,
   783→          });
   784→          setStep("results");
   785→
   786→          if (data.passed) {
   787→            toast.success("Баяр хүргэе!", {
   788→              description: `Та ${data.score}/${data.total} оноо авч тэнцлээ!`,
   789→            });
   790→          } else {
   791→            toast.error("Тэнцээгүй", {
   792→              description: `Та ${data.score}/${data.total} оноо авсан байна. 70% шаардлагатай.`,
   793→            });
   794→          }
   795→        } else {
   796→          toast.error("Алдаа", {
   797→            description: data.error || "Хариултуудыг илгээхэд алдаа гарлаа.",
   798→          });
   799→        }
   800→      } catch {
   801→        toast.error("Холболтын алдаа");
   802→      } finally {
   803→        setIsSubmitting(false);
   804→      }
   805→    },
   806→    [selectedQuiz]
   807→  );
   808→
   809→  const handleRetry = useCallback(() => {
   810→    setResult(null);
   811→    setStep("payment");
   812→  }, []);
   813→
   814→  const handleBackToList = useCallback(() => {
   815→    setSelectedQuiz(null);
   816→    setResult(null);
   817→    setStep("selection");
   818→  }, []);
   819→
   820→  return (
   821→    <section className="w-full">
   822→      {step === "selection" && <QuizSelection onSelect={handleSelectQuiz} />}
   823→
   824→      {step === "payment" && selectedQuiz && (
   825→        <PaymentGate quiz={selectedQuiz} onPaid={handlePaid} onBack={handleBackToList} />
   826→      )}
   827→
   828→      {step === "taking" && selectedQuiz && (
   829→        <QuizTaking
   830→          quiz={selectedQuiz}
   831→          onFinish={handleFinish}
   832→          onBack={handleBackToList}
   833→        />
   834→      )}
   835→
   836→      {step === "results" && selectedQuiz && result && (
   837→        <QuizResults
   838→          quiz={selectedQuiz}
   839→          result={result}
   840→          onRetry={handleRetry}
   841→          onBackToList={handleBackToList}
   842→        />
   843→      )}
   844→
   845→      {/* Submit overlay */}
   846→      <AnimatePresence>
   847→        {isSubmitting && (
   848→          <motion.div
   849→            initial={{ opacity: 0 }}
   850→            animate={{ opacity: 1 }}
   851→            exit={{ opacity: 0 }}
   852→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 backdrop-blur-sm"
   853→          >
   854→            <div className="flex flex-col items-center gap-4 rounded-2xl bg-white p-8 shadow-2xl">
   855→              <Loader2 className="size-10 animate-spin text-brand-600" />
   856→              <p className="text-lg font-medium text-brand-900">
   857→                Хариултуудыг шалгаж байна...
   858→              </p>
   859→            </div>
   860→          </motion.div>
   861→        )}
   862→      </AnimatePresence>
   863→    </section>
   864→  );
   865→}
